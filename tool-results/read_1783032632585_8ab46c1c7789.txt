   142→   142→function timeAgo(iso: string): string {
   143→   143→  try {
   144→   144→    const date = new Date(iso);
   145→   145→    const now = Date.now();
   146→   146→    const diff = Math.max(0, now - date.getTime());
   147→   147→    const seconds = Math.floor(diff / 1000);
   148→   148→    const minutes = Math.floor(seconds / 60);
   149→   149→    const hours = Math.floor(minutes / 60);
   150→   150→    const days = Math.floor(hours / 24);
   151→   151→    if (days >= 30) return date.toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric' });
   152→   152→    if (days >= 1) return `قبل ${days} ${days === 1 ? 'يوم' : days === 2 ? 'يومين' : 'أيام'}`;
   153→   153→    if (hours >= 1) return `قبل ${hours} ${hours === 1 ? 'ساعة' : hours === 2 ? 'ساعتين' : 'ساعات'}`;
   154→   154→    if (minutes >= 1) return `قبل ${minutes} ${minutes === 1 ? 'دقيقة' : 'دقائق'}`;
   155→   155→    return 'الآن';
   156→   156→  } catch {
   157→   157→    return '';
   158→   158→  }
   159→   159→}

   309→   309→function ParentLogin({
   310→   310→  schoolId, onLogin,
   311→   311→}: {
   312→   312→  schoolId: string;
   313→   313→  onLogin: (s: ParentSession) => void;
   314→   314→}) {
   315→   315→  const [studentNumber, setStudentNumber] = useState('');
   316→   316→  const [parentPhone, setParentPhone] = useState('');
   317→   317→  const [loading, setLoading] = useState(false);
   318→   318→  const [error, setError] = useState<string | null>(null);
   319→   319→  const [fadeIn, setFadeIn] = useState(false);
   320→   320→
   321→   321→  useEffect(() => {
   322→   322→    const t = setTimeout(() => setFadeIn(true), 100);
   323→   323→    return () => clearTimeout(t);
   324→   324→  }, []);
   325→   325→
   326→   326→  const handleSubmit = async (e: React.FormEvent) => {
   327→   327→    e.preventDefault();
   328→   328→    if (!studentNumber.trim() || !parentPhone.trim()) {
   329→   329→      setError('يرجى إدخال رقم الطالب وهاتف ولي الأمر');
   330→   330→      return;
   331→   331→    }
   332→   332→    setLoading(true);
   333→   333→    setError(null);
   334→   334→    try {
   335→   335→      const res = await fetch(`/api/parent/login?schoolId=${encodeURIComponent(schoolId)}`, {
   336→   336→        method: 'POST',
   337→   337→        headers: { 'Content-Type': 'application/json' },
   338→   338→        body: JSON.stringify({
   339→   339→          schoolId,
   340→   340→          studentNumber: studentNumber.trim(),
   341→   341→          parentPhone: parentPhone.trim(),
   342→   342→        }),
   343→   343→      });
   344→   344→      const data = await res.json();
   345→   345→      if (!res.ok) {
   346→   346→        const msg = data?.error || (res.status === 404 ? 'رقم الطالب أو هاتف ولي الأمر غير صحيح' : 'تعذّر تسجيل الدخول');
   347→   347→        throw new Error(typeof msg === 'string' ? msg : 'تعذّر تسجيل الدخول');
   348→   348→      }
   349→   349→      if (!data?.children || !Array.isArray(data.children) || data.children.length === 0) {
   350→   350→        throw new Error('لم يتم العثور على أبناء مرتبطين بهذا الحساب');
   351→   351→      }
   352→   352→      const sessionData: ParentSession = {
   353→   353→        parentName: data.parentName || 'ولي الأمر',
   354→   354→        parentPhone: data.parentPhone || parentPhone.trim(),
   355→   355→        children: data.children,
   356→   356→        testMode: !!data.testMode,
   357→   357→        fakeStudent: !!data.fakeStudent,
   358→   358→        originalStudentNumber: data.originalStudentNumber ?? null,
   359→   359→        warning: data.warning ?? null,
   360→   360→      };
   361→   361→      onLogin(sessionData);
   362→   362→      toast.success(`مرحباً ${sessionData.parentName}`);
   363→   363→    } catch (err) {
   364→   364→      const msg = err instanceof Error ? err.message : 'تعذّر تسجيل الدخول';
   365→   365→      setError(msg);
   366→   366→    } finally {
   367→   367→      setLoading(false);
   368→   368→    }
   369→   369→  };
   370→   370→
   371→   371→  return (
   372→   372→    <div className={`max-w-md mx-auto transition-all duration-700 ${fadeIn ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'}`}>
   373→   373→      <Card className="overflow-hidden border-0 shadow-2xl">
   374→   374→        {/* Gradient header */}
   375→   375→        <div className="bg-gradient-to-l from-emerald-600 via-emerald-600 to-teal-600 p-6 text-white relative overflow-hidden">
   376→   376→          <div className="absolute top-0 left-0 w-40 h-40 bg-white/10 rounded-full -translate-x-1/2 -translate-y-1/2" />
   377→   377→          <div className="absolute bottom-0 right-0 w-32 h-32 bg-teal-300/10 rounded-full translate-x-1/4 translate-y-1/4" />
   378→   378→          <div className="relative z-10 text-center">
   379→   379→            <div className="w-16 h-16 mx-auto bg-white/20 rounded-full flex items-center justify-center mb-3 backdrop-blur">
   380→   380→              <Users className="w-9 h-9 text-white" />
   381→   381→            </div>
   382→   382→            <h2 className="text-xl font-bold mb-1">تسجيل دخول ولي الأمر</h2>
   383→   383→            <p className="text-emerald-50 text-sm">أدخل بيانات الطالب للوصول إلى لوحة المتابعة</p>
   384→   384→          </div>
   385→   385→        </div>
   386→   386→
   387→   387→        <CardContent className="p-6 space-y-4">
   388→   388→          <form onSubmit={handleSubmit} className="space-y-4">
   389→   389→            <div className="space-y-1.5">
   390→   390→              <Label htmlFor="studentNumber" className="text-sm font-semibold">
   391→   391→                رقم الطالب
   392→   392→              </Label>
   393→   393→              <Input
   394→   394→                id="studentNumber"
   395→   395→                value={studentNumber}
   396→   396→                onChange={(e) => setStudentNumber(e.target.value)}
   397→   397→                placeholder="مثال: 2024002"
   398→   398→                disabled={loading}
   399→   399→                autoComplete="off"
   400→   400→                inputMode="numeric"
   401→   401→                className="text-base h-11"
   402→   402→              />
   403→   403→            </div>
   404→   404→            <div className="space-y-1.5">
