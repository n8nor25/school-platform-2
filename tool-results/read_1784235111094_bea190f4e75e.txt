   108→function HomePage() {
   109→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated, setHasHydrated } = useAdminStore()
   110→  const searchParams = useSearchParams()
   111→  const router = useRouter()

   114→  const urlAction = searchParams.get('action')

   116→
   117→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   118→  const [news, setNews] = useState<NewsItem[]>([])
   119→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   120→  const [teachers, setTeachers] = useState<Teacher[]>([])
   121→  const [sliders, setSliders] = useState<SliderItem[]>([])
   122→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   123→  const [loading, setLoading] = useState(true)
   124→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)

   127→  const [showResultsPage, setShowResultsPage] = useState(false)
   128→  const [showStudentLife, setShowStudentLife] = useState(false)
   129→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   130→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   131→  const [showParentsPage, setShowParentsPage] = useState(false)
   132→  const [showTeacherPage, setShowTeacherPage] = useState(false)
   133→  const [showExamsPage, setShowExamsPage] = useState(false)
   134→  const [showParentExamsPage, setShowParentExamsPage] = useState(false)
   135→  const [parentExamsChild, setParentExamsChild] = useState<{
   136→    id: string;
   137→    name: string;
   138→    studentNumber: string;
   139→    classroomName?: string | null;
   140→    gradeName?: string | null;
   141→  } | null>(null)
   142→  const [showAnalyticsPage, setShowAnalyticsPage] = useState(false)

   144→  const [showScrollTop, setShowScrollTop] = useState(false)

   146→  const logoClickCount = useRef(0)
   147→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   148→
   149→  const hydrated = _hasHydrated
   150→
   151→  // Fallback: ensure hydration flag is set on the client (Zustand v5
   152→  // onRehydrateStorage may not fire in all environments).
   153→  useEffect(() => {
   154→    if (!_hasHydrated) setHasHydrated(true)
   155→  }, [_hasHydrated, setHasHydrated])
   156→
   157→  // Resolve school from subdomain only (not from store's school list)
   158→  useEffect(() => {
   159→    if (!hydrated) return
   160→    let cancelled = false
   161→    async function resolveSchool() {
   162→      try {
   163→        if (urlSubdomain) {
   164→          // If subdomain is in URL, fetch by subdomain
   165→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   166→          if (res.ok && !cancelled) {
   167→            const data = await res.json()
   168→            if (data.school) {
   169→              setSchoolData(data)
   170→              setSelectedSchoolId(data.school.id)
   171→              return
   172→            }
   173→          }
   174→        }
   175→        // No subdomain in URL — fetch first available school
   176→        const schoolsRes = await fetch('/api/schools')
   177→        if (schoolsRes.ok && !cancelled) {
   178→          const schoolsList = await schoolsRes.json()
   179→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   180→            setSchools(schoolsList)
   181→            const firstSchool = schoolsList[0]
   182→            // Fetch full data for the first school
   183→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   184→            if (res.ok && !cancelled) {
   185→              const data = await res.json()
   186→              if (data.school) {
   187→                setSchoolData(data)
   188→                setSelectedSchoolId(data.school.id)
   189→              }
   190→            }
   191→          }
   192→        }
   193→      } catch {
   194→        // fallback to default data
   195→      }
   196→    }
   197→    resolveSchool()
   198→    return () => { cancelled = true }
   199→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   200→
   201→  // Fetch content for selected school
   202→  useEffect(() => {
   203→    if (!hydrated || !selectedSchoolId) return
   204→    let cancelled = false
   205→    const sid = selectedSchoolId
   206→    async function loadData() {
   207→      setLoading(true)
   208→      try {
   209→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   210→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   211→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   212→          fetch(`/api/teachers?schoolId=${sid}`),
   213→          fetch(`/api/sliders?schoolId=${sid}`),
   214→          fetch(`/api/custom-sections?schoolId=${sid}`),
   215→        ])
   216→        if (cancelled) return
   217→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   218→          const d = await newsRes.value.json()
   219→          if (Array.isArray(d) && !cancelled) setNews(d)
   220→        }
   221→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   222→          const d = await galleryRes.value.json()
   223→          if (Array.isArray(d) && !cancelled) setGallery(d)
   224→        }
   225→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   226→          const d = await teachersRes.value.json()
   227→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   228→        }
   229→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   230→          const d = await slidersRes.value.json()
   231→          if (Array.isArray(d) && !cancelled) setSliders(d)
   232→        }
   233→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   234→          const d = await csRes.value.json()
   235→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   236→        }
   237→      } catch {
   238→        // silently fail
   239→      } finally {
   240→        if (!cancelled) setLoading(false)
   241→      }
   242→    }
   243→    loadData()
   244→    return () => { cancelled = true }
   245→  }, [selectedSchoolId, hydrated])
   246→
   247→  // Auto-rotate slider
   248→  const activeSliders = sliders.filter(s => s.active)
   249→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   250→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   251→  useEffect(() => {
   252→    if (slideItems.length <= 1) return
   253→    const timer = setInterval(() => setCurrentSlide(i => (i + 1) % slideItems.length), 5000)
   254→    return () => clearInterval(timer)
   255→  }, [slideItems.length])
   256→
   257→  // Scroll to top
   258→  useEffect(() => {
   259→    const h = () => setShowScrollTop(window.scrollY > 500)
   260→    window.addEventListener('scroll', h)
   261→    return () => window.removeEventListener('scroll', h)
   262→  }, [])
   263→
   264→  const handleLogoClick = useCallback(() => {
   265→    logoClickCount.current += 1
   266→    if (logoClickTimer.current) clearTimeout(logoClickTimer.current)
   267→    logoClickTimer.current = setTimeout(() => { logoClickCount.current = 0 }, 3000)
   268→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   269→  }, [])
   270→
   271→  // معالجة اختصارات PWA (?action=...) عند فتح التطبيق من الشاشة الرئيسية
   272→  const urlActionProcessed = useRef(false)
   273→  useEffect(() => {
   274→    if (!urlAction || !hydrated || urlActionProcessed.current) return
   275→    urlActionProcessed.current = true
   276→    // تنظيف الـ URL من معامل action بعد تفعيله
   277→    router.replace('/')
   278→    // استخدام microtask لتجنّب تحذير set-state-in-effect
   279→    queueMicrotask(() => {
   280→      if (urlAction === 'exams') { setShowExamsPage(true) }
   281→      else if (urlAction === 'results') setShowResultsPage(true)
   282→      else if (urlAction === 'schedules') setShowSchedulesPage(true)
   283→      else if (urlAction === 'library') setShowLibraryPage(true)
   284→      else if (urlAction === 'parents') setShowParentsPage(true)
   285→      else if (urlAction === 'teacher') setShowTeacherPage(true)
   286→      else if (urlAction === 'coordinator') setShowCoordinatorPage(true)
   287→    })
   288→  }, [urlAction, hydrated, router])
   289→
   290→  const handleServiceAction = (action: string) => {
   291→    setServicesOpen(false); setMobileMenuOpen(false)
   292→    if (action === 'results') setShowResultsPage(true)
   293→    if (action === 'schedules') setShowSchedulesPage(true)
   294→    if (action === 'library') setShowLibraryPage(true)
   295→    if (action === 'parents') setShowParentsPage(true)
   296→    if (action === 'teacher') setShowTeacherPage(true)
   297→    if (action === 'exams') setShowExamsPage(true)
   298→    if (action === 'teacher') setShowTeacherPage(true)
   299→    if (action === 'coordinator') setShowCoordinatorPage(true)
   300→  }
   301→
   302→  // الانتقال من بوابة أولياء الأمور إلى صفحة الامتحانات الإلكترونية

   305→    setShowExamsPage(true)
   306→  }, [])
   307→
   308→  // الانتقال من بوابة أولياء الأمور إلى صفحة التحليلات المقارنة
   309→  const handleOpenAnalyticsFromParents = useCallback(() => {
   310→    setShowParentsPage(false)
   311→    setShowAnalyticsPage(true)
   312→  }, [])
   313→
   314→  const school = schoolData?.school || defaultSchoolData.school
   315→  const settings = schoolData?.settings || defaultSchoolData.settings
   316→  const stats = schoolData?.stats || defaultSchoolData.stats
   317→  const alertNews = news.filter(n => n.category === 'تنبيه')
   318→
   319→  // Build ticker text for seamless scrolling
   320→  const tickerText = news.map(item => item.title).join('     ◆     ')
   321→
   322→  // Sub-page rendering
   323→  if (isAdminMode) return <AdminLayout />
   324→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   325→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   326→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   327→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />

   329→    return (
   330→      <ParentExamsPage
   331→        onBack={() => {
   332→          setShowParentExamsPage(false)

   334→        }}
   335→        schoolId={selectedSchoolId}
   336→        child={parentExamsChild}
   337→      />
   338→    )
   339→  }
   340→  if (showAnalyticsPage) return <ParentAnalyticsPage onBack={() => setShowAnalyticsPage(false)} schoolId={selectedSchoolId} />
   341→  if (showCoordinatorPage) return <CoordinatorDashboardPage onBack={() => setShowCoordinatorPage(false)} schoolId={selectedSchoolId} />
   342→  if (showParentsPage) return (
   343→    <ParentsPortalPage

   347→        setParentExamsChild(child)
   348→        setShowParentExamsPage(true)
   349→      }}
   350→      onOpenAnalytics={() => setShowAnalyticsPage(true)}
   351→    />

   353→  if (showTeacherPage) return <TeacherPortalPage onBack={() => setShowTeacherPage(false)} schoolId={selectedSchoolId} />
   354→  if (showExamsPage) return <StudentExamsPage onBack={() => setShowExamsPage(false)} schoolId={selectedSchoolId} />
   355→
   356→  if (!hydrated) {
   357→    return (
   358→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   359→        <div className="text-center">
   360→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />
   361→          <p className="text-gray-500 text-sm">جاري التحميل...</p>
   362→        </div>
   363→      </div>
   364→    )
   365→  }
   366→
   367→  return (
   368→    <div className="min-h-screen flex flex-col" dir="rtl">
   369→
   370→      {/* ===== STICKY TOP ===== */}
   371→      <div className="sticky top-0 z-50 shadow-lg">

   373→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">

   387→          </div>
   388→        </div>

   390→        {/* Header */}

   410→        {/* Navigation */}

   412→          <div className="max-w-[1280px] mx-auto px-4">

   414→            <div className="hidden lg:flex items-center h-11">
   415→              {navLinks.map((link) => (
