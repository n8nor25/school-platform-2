   104→function HomePage() {
   105→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated } = useAdminStore()
   106→  const searchParams = useSearchParams()
   107→  const router = useRouter()
   108→  const pathname = usePathname()
   109→  const urlSubdomain = searchParams.get('school')
   110→  const hasAppliedUrlSchool = useRef(false)
   111→
   112→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   113→  const [news, setNews] = useState<NewsItem[]>([])
   114→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   115→  const [teachers, setTeachers] = useState<Teacher[]>([])
   116→  const [sliders, setSliders] = useState<SliderItem[]>([])
   117→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   118→  const [loading, setLoading] = useState(true)
   119→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
   120→  const [servicesOpen, setServicesOpen] = useState(false)
   121→  const [showAdminLogin, setShowAdminLogin] = useState(false)
   122→  const [showResultsPage, setShowResultsPage] = useState(false)
   123→  const [showStudentLife, setShowStudentLife] = useState(false)
   124→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   125→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   126→  const [showParentsPage, setShowParentsPage] = useState(false)
   127→  const [showTeacherExamsPage, setShowTeacherExamsPage] = useState(false)
   128→  const [showStudentExamsPage, setShowStudentExamsPage] = useState(false)
   129→  const [showScrollTop, setShowScrollTop] = useState(false)
   130→  const [currentSlide, setCurrentSlide] = useState(0)
   131→  const logoClickCount = useRef(0)
   132→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   133→
   134→  const hydrated = _hasHydrated
   135→
   136→  // Resolve school from subdomain only (not from store's school list)
   137→  useEffect(() => {
   138→    if (!hydrated) return
   139→    let cancelled = false
   140→    async function resolveSchool() {
   141→      try {
   142→        if (urlSubdomain) {
   143→          // If subdomain is in URL, fetch by subdomain
   144→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   145→          if (res.ok && !cancelled) {
   146→            const data = await res.json()
   147→            if (data.school) {
   148→              setSchoolData(data)
   149→              setSelectedSchoolId(data.school.id)
   150→              return
   151→            }
   152→          }
   153→        }
   154→        // No subdomain in URL — fetch first available school
   155→        const schoolsRes = await fetch('/api/schools')
   156→        if (schoolsRes.ok && !cancelled) {
   157→          const schoolsList = await schoolsRes.json()
   158→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   159→            setSchools(schoolsList)
   160→            const firstSchool = schoolsList[0]
   161→            // Fetch full data for the first school
   162→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   163→            if (res.ok && !cancelled) {
   164→              const data = await res.json()
   165→              if (data.school) {
   166→                setSchoolData(data)
   167→                setSelectedSchoolId(data.school.id)
   168→              }
   169→            }
   170→          }
   171→        }
   172→      } catch {
   173→        // fallback to default data
   174→      }
   175→    }
   176→    resolveSchool()
   177→    return () => { cancelled = true }
   178→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   179→
   180→  // Fetch content for selected school
   181→  useEffect(() => {
   182→    if (!hydrated || !selectedSchoolId) return
   183→    let cancelled = false
   184→    const sid = selectedSchoolId
   185→    async function loadData() {
   186→      setLoading(true)
   187→      try {
   188→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   189→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   190→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   191→          fetch(`/api/teachers?schoolId=${sid}`),
   192→          fetch(`/api/sliders?schoolId=${sid}`),
   193→          fetch(`/api/custom-sections?schoolId=${sid}`),
   194→        ])
   195→        if (cancelled) return
   196→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   197→          const d = await newsRes.value.json()
   198→          if (Array.isArray(d) && !cancelled) setNews(d)
   199→        }
   200→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   201→          const d = await galleryRes.value.json()
   202→          if (Array.isArray(d) && !cancelled) setGallery(d)
   203→        }
   204→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   205→          const d = await teachersRes.value.json()
   206→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   207→        }
   208→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   209→          const d = await slidersRes.value.json()
   210→          if (Array.isArray(d) && !cancelled) setSliders(d)
   211→        }
   212→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   213→          const d = await csRes.value.json()
   214→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   215→        }
   216→      } catch {
   217→        // silently fail
   218→      } finally {
   219→        if (!cancelled) setLoading(false)
   220→      }
   221→    }
   222→    loadData()
   223→    return () => { cancelled = true }
   224→  }, [selectedSchoolId, hydrated])
   225→
   226→  // Auto-rotate slider
   227→  const activeSliders = sliders.filter(s => s.active)
   228→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   229→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   230→  useEffect(() => {
   231→    if (slideItems.length <= 1) return
   232→    const timer = setInterval(() => setCurrentSlide(i => (i + 1) % slideItems.length), 5000)
   233→    return () => clearInterval(timer)
   234→  }, [slideItems.length])
   235→
   236→  // Scroll to top
   237→  useEffect(() => {
   238→    const h = () => setShowScrollTop(window.scrollY > 500)
   239→    window.addEventListener('scroll', h)
   240→    return () => window.removeEventListener('scroll', h)
   241→  }, [])
   242→
   243→  const handleLogoClick = useCallback(() => {
   244→    logoClickCount.current += 1
   245→    if (logoClickTimer.current) clearTimeout(logoClickTimer.current)
   246→    logoClickTimer.current = setTimeout(() => { logoClickCount.current = 0 }, 3000)
   247→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   248→  }, [])
   249→
   250→  const handleServiceAction = (action: string) => {
   251→    setServicesOpen(false); setMobileMenuOpen(false)
   252→    if (action === 'results') setShowResultsPage(true)
   253→    if (action === 'schedules') setShowSchedulesPage(true)
   254→    if (action === 'library') setShowLibraryPage(true)
   255→    if (action === 'parents') setShowParentsPage(true)
   256→    if (action === 'teacher-exams') setShowTeacherExamsPage(true)
   257→    if (action === 'student-exams') setShowStudentExamsPage(true)
   258→  }
   259→
   260→  const school = schoolData?.school || defaultSchoolData.school
   261→  const settings = schoolData?.settings || defaultSchoolData.settings
   262→  const stats = schoolData?.stats || defaultSchoolData.stats
   263→  const alertNews = news.filter(n => n.category === 'تنبيه')
   264→
   265→  // Build ticker text for seamless scrolling
   266→  const tickerText = news.map(item => item.title).join('     ◆     ')
   267→
   268→  // Sub-page rendering
   269→  if (isAdminMode) return <AdminLayout />
   270→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   271→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   272→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   273→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />
   274→  if (showParentsPage) return <ParentsPortalPage onBack={() => setShowParentsPage(false)} schoolId={selectedSchoolId} />
   275→  if (showTeacherExamsPage) return <TeacherExamsPage onBack={() => setShowTeacherExamsPage(false)} schoolId={selectedSchoolId} />
   276→  if (showStudentExamsPage) return <StudentExamsPage onBack={() => setShowStudentExamsPage(false)} schoolId={selectedSchoolId} />
   277→
   278→  if (!hydrated) {
   279→    return (
   280→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   281→        <div className="text-center">
   282→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />
   283→          <p className="text-gray-500 text-sm">جاري التحميل...</p>
   284→        </div>
   285→      </div>
   286→    )
   287→  }
   288→
   289→  return (
   290→    <div className="min-h-screen flex flex-col" dir="rtl">
   291→
   292→      {/* ===== STICKY TOP ===== */}
   293→      <div className="sticky top-0 z-50 shadow-lg">
   294→        {/* Top Bar - Contact info only, NO school switcher for visitors */}
   295→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">
   296→          <div className="max-w-[1280px] mx-auto px-4 py-1.5 flex items-center justify-between">
   297→            <div className="flex items-center gap-4">
   298→              {school.phone && <a href={`tel:${school.phone}`} className="hover:opacity-80 flex items-center gap-1">📞 {school.phone}</a>}
   299→              {school.email && <a href={`mailto:${school.email}`} className="hover:opacity-80 flex items-center gap-1">✉️ {school.email}</a>}
   300→            </div>
   301→            <div className="flex items-center gap-2">
   302→              {school.facebookUrl && (
   303→                <a href={school.facebookUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📘</a>
   304→              )}
   305→              {settings?.youtubeUrl && (
   306→                <a href={settings.youtubeUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📺</a>
   307→              )}
   308→            </div>
   309→          </div>
   310→        </div>
   311→
   312→        {/* Header */}
   313→        <header className="bg-white shadow-sm">
   314→          <div className="max-w-[1280px] mx-auto px-4 py-2 flex items-center gap-3">
   315→            <div onClick={handleLogoClick} className="cursor-pointer shrink-0" title="اضغط 5 مرات للإدارة">
   316→              {school.logoUrl ? (
   317→                <img src={school.logoUrl} alt={school.name} className="w-12 h-12 rounded-full object-cover border-2" style={{ borderColor: school.primaryColor }} />
   318→              ) : (
   319→                <div className="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl" style={{ backgroundColor: school.primaryColor }}>🏫</div>
   320→              )}
   321→            </div>
   322→            <div className="flex-1 min-w-0">
   323→              <h1 className="text-base md:text-xl font-bold truncate" style={{ color: school.primaryColor }}>{school.name}</h1>
   324→              {settings?.heroSubtitle && <p className="text-xs text-gray-500 truncate">{settings.heroSubtitle}</p>}
   325→            </div>
   326→            <button className="lg:hidden p-2 text-2xl text-gray-600" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   327→              {mobileMenuOpen ? '✕' : '☰'}
   328→            </button>
   329→          </div>
   330→        </header>
   331→
   332→        {/* Navigation */}
   333→        <nav style={{ backgroundColor: '#1a2332' }} className="text-white shadow-lg">
   334→          <div className="max-w-[1280px] mx-auto px-4">
   335→            {/* Desktop Nav */}
   336→            <div className="hidden lg:flex items-center h-11">
   337→              {navLinks.map((link) => (
   338→                link.isServices ? (
   339→                  <div key={link.href} className="relative" onMouseEnter={() => setServicesOpen(true)} onMouseLeave={() => setServicesOpen(false)}>
   340→                    <button className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center gap-1 transition-colors">
   341→                      {link.label} ▾

   349→                          </button>
   350→                        ))}
   351→                      </div>
   352→                    )}
   353→                  </div>
   354→                ) : (
   355→                  <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   356→                    onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true) } : undefined}
   357→                    className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center transition-colors">
   358→                    {link.label}
   359→                  </a>
   360→                )
   361→              ))}
   362→            </div>
   363→            {/* Mobile Nav */}
   364→            {mobileMenuOpen && (
   365→              <div className="lg:hidden py-2 border-t border-white/10">
   366→                {navLinks.map((link) => (
   367→                  link.isServices ? (
   368→                    <div key={link.href}>
   369→                      <div className="px-4 py-2 text-sm font-medium text-white/70">{link.label}</div>
   370→                      {serviceItems.map(item => (
   371→                        <button key={item.action} onClick={() => handleServiceAction(item.action)}
   372→                          className="w-full flex items-center gap-2 px-8 py-2 text-sm text-white/80 hover:bg-white/10">
   373→                          <span>{item.icon}</span> {item.label}
   374→                        </button>
   375→                      ))}
   376→                    </div>
   377→                  ) : (
   378→                    <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   379→                      onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true); setMobileMenuOpen(false) } : () => setMobileMenuOpen(false)}
   380→                      className="block px-4 py-2 text-sm hover:bg-white/10">{link.label}</a>
   381→                  )
   382→                ))}
   383→              </div>
   384→            )}
   385→          </div>
   386→        </nav>
   387→
   388→        {/* News Ticker - Seamless scrolling */}
   389→        {settings?.showNewsTicker && news.length > 0 && (

   391→            <div className="flex items-center">
   392→              <span className="bg-red-600 text-white px-4 py-2 text-xs font-bold shrink-0 flex items-center gap-1 z-10">
   393→                🔴 عاجل
   394→              </span>
   395→              <div className="overflow-hidden flex-1 relative">
   396→                <div className="animate-news-ticker-scroll whitespace-nowrap py-2">
   397→                  {/* Duplicate content for seamless loop */}
   398→                  <span className="inline-block text-sm mx-4">
   399→                    {tickerText}
   400→                  </span>
   401→                  <span className="inline-block text-sm mx-4">
   402→                    {tickerText}
   403→                  </span>
   404→                </div>
   405→              </div>
   406→            </div>
   407→          </div>
   408→        )}
   409→      </div>
   410→
   411→      {/* ===== MAIN CONTENT ===== */}
   412→      <main className="flex-1">
   413→        {/* Hero Slider */}
   414→        {(settings?.showSlider ?? true) && (
   415→          <section className="max-w-[1280px] mx-auto px-4 py-6">
   416→            <div className="grid grid-cols-1 md:grid-cols-12 gap-4">
   417→              <div className="md:col-span-7 rounded-lg overflow-hidden shadow-lg relative h-[300px] md:h-[400px] bg-gray-200">
   418→                {loading ? (
   419→                  <div className="w-full h-full flex items-center justify-center text-gray-400">جاري التحميل...</div>
   420→                ) : slideItems.length > 0 ? (
