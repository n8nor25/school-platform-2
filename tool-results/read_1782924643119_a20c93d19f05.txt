   318→async function teacherFetch<T>(url: string, options?: RequestInit): Promise<T> {
   319→  const controller = new AbortController();
   320→  const timeout = setTimeout(() => controller.abort(), 25000);
   321→  try {
   322→    const res = await fetch(url, {
   323→      ...options,
   324→      signal: controller.signal,
   325→      headers: {
   326→        'Content-Type': 'application/json',
   327→        ...(options?.headers || {}),
   328→      },
   329→    });
   330→    const data = await res.json().catch(() => ({}));
   331→    if (!res.ok) throw new Error((data as Record<string, unknown>).error as string || `HTTP ${res.status}`);
   332→    return data as T;
   333→  } catch (err: unknown) {
   334→    if (err instanceof Error && err.name === 'AbortError') throw new Error('انتهت مهلة الطلب');
   335→    throw err;
   336→  } finally {
   337→    clearTimeout(timeout);
   338→  }
   339→}

   341→function buildUrl(base: string, teacher: TeacherInfo, extra?: Record<string, string>): string {
   342→  const params = new URLSearchParams();
   343→  params.set('schoolId', teacher.schoolId);
   344→  params.set('teacherId', teacher.teacherId);
   345→  params.set('teacherName', teacher.teacherName);
   346→  if (extra) for (const [k, v] of Object.entries(extra)) params.set(k, v);
   347→  return `${base}?${params.toString()}`;
   348→}

   350→function formatDate(iso: string | null): string {
   351→  if (!iso) return '—';
   352→  try {
   353→    return new Date(iso).toLocaleDateString('ar-EG', {
   354→      year: 'numeric', month: 'long', day: 'numeric', hour: '2-digit', minute: '2-digit',
   355→    });
   356→  } catch {
   357→    return iso;
   358→  }
   359→}

   361→function toDateTimeLocal(iso: string): string {
   362→  try {
   363→    const d = new Date(iso);
   364→    const off = d.getTimezoneOffset();
   365→    return new Date(d.getTime() - off * 60000).toISOString().slice(0, 16);
   366→  } catch {
   367→    return '';
   368→  }
   369→}
   370→
   371→function defaultDateTimeLocal(minutesFromNow: number): string {
   372→  const d = new Date(Date.now() + minutesFromNow * 60000);
   373→  const off = d.getTimezoneOffset();
   374→  return new Date(d.getTime() - off * 60000).toISOString().slice(0, 16);
   375→}

   710→function TeacherDashboard({
   711→  teacher, onOpenExam, showToast,
   712→}: {
   713→  teacher: TeacherInfo;
   714→  onOpenExam: (examId: string) => void;
   715→  showToast: (type: ToastMessage['type'], text: string) => void;
   716→}) {
   717→  const [exams, setExams] = useState<ExamListItem[]>([]);
   718→  const [loading, setLoading] = useState(true);
   719→  const [error, setError] = useState<string | null>(null);
   720→  const [statusFilter, setStatusFilter] = useState<string>('all');
   721→  const [search, setSearch] = useState('');
   722→  const [showCreate, setShowCreate] = useState(false);
   723→  const [confirmDelete, setConfirmDelete] = useState<ExamListItem | null>(null);
   724→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   725→
   726→  const fetchExams = useCallback(async () => {
   727→    setLoading(true);
   728→    setError(null);
   729→    try {
   730→      const extra: Record<string, string> = {};
   731→      if (statusFilter !== 'all') extra.status = statusFilter;
   732→      if (search.trim()) extra.search = search.trim();
   733→      const url = buildUrl('/api/exams/teacher', teacher, extra);
   734→      const data = await teacherFetch<{ success: boolean; exams: ExamListItem[] }>(url);
   735→      setExams(data.exams || []);
   736→    } catch (err: unknown) {
   737→      setError(err instanceof Error ? err.message : 'فشل جلب الامتحانات');
   738→    } finally {
   739→      setLoading(false);
   740→    }
   741→  }, [teacher, statusFilter, search]);
   742→
   743→  useEffect(() => {
   744→    const t = setTimeout(fetchExams, 350);
   745→    return () => clearTimeout(t);
   746→  }, [fetchExams]);
   747→
   748→  const handlePublish = async (exam: ExamListItem) => {
   749→    setActionLoading(exam.id);
   750→    try {
   751→      const url = buildUrl(`/api/exams/teacher/${exam.id}/publish`, teacher);
   752→      await teacherFetch<{ success: boolean }>(url, { method: 'POST' });
   753→      showToast('success', 'تم نشر الامتحان بنجاح');
   754→      fetchExams();
   755→    } catch (err: unknown) {
   756→      showToast('error', err instanceof Error ? err.message : 'فشل النشر');
   757→    } finally {
   758→      setActionLoading(null);
   759→    }
   760→  };
   761→
   762→  const handleClose = async (exam: ExamListItem) => {
   763→    if (!confirm(`هل تريد إغلاق الامتحان "${exam.title}"؟ سيتم إغلاق جميع المحاولات الجارية.`)) {
   764→      return;
   765→    }
   766→    setActionLoading(exam.id);
   767→    try {
   768→      const url = buildUrl(`/api/exams/teacher/${exam.id}/close`, teacher);
   769→      const data = await teacherFetch<{ success: boolean; closedSubmissions: number }>(url, { method: 'POST' });
   770→      showToast('success', `تم الإغلاق. أُغلق ${data.closedSubmissions || 0} محاولة جارية.`);
   771→      fetchExams();
   772→    } catch (err: unknown) {
   773→      showToast('error', err instanceof Error ? err.message : 'فشل الإغلاق');
   774→    } finally {
   775→      setActionLoading(null);
   776→    }
   777→  };
   778→
   779→  const handleArchive = async (exam: ExamListItem) => {
   780→    if (!confirm(`هل تريد أرشفة الامتحان "${exam.title}"؟`)) return;
   781→    setActionLoading(exam.id);
   782→    try {
   783→      const url = buildUrl(`/api/exams/teacher/${exam.id}/archive`, teacher);
   784→      await teacherFetch<{ success: boolean }>(url, { method: 'POST' });
   785→      showToast('success', 'تمت الأرشفة بنجاح');
   786→      fetchExams();
   787→    } catch (err: unknown) {
   788→      showToast('error', err instanceof Error ? err.message : 'فشل الأرشفة');
   789→    } finally {
   790→      setActionLoading(null);
   791→    }
   792→  };

   800→      showToast('success', 'تم حذف الامتحان');
   801→      setConfirmDelete(null);
   802→      fetchExams();
   803→    } catch (err: unknown) {
   804→      showToast('error', err instanceof Error ? err.message : 'فشل الحذف (قد يكون منشوراً وله تسليمات)');
   805→    } finally {
   806→      setActionLoading(null);
   807→    }
   808→  };
   809→
   810→  // KPI stats
   811→  const kpis = {

   813→    published: exams.filter(e => e.status === 'PUBLISHED').length,
   814→    draft: exams.filter(e => e.status === 'DRAFT').length,

   816→    submissionsTotal: exams.reduce((s, e) => s + e.submissionsCount, 0),
   817→  };

   824→          icon={<FileText className="w-5 h-5" />}
   825→          label="إجمالي الامتحانات"
   826→          value={kpis.total}
   827→          color="bg-gray-100 text-gray-700"
   828→        />
   829→        <KpiCard
   830→          icon={<CheckCircle2 className="w-5 h-5" />}
   831→          label="منشور"
   832→          value={kpis.published}
   833→          color="bg-emerald-100 text-emerald-700"
   834→        />
   835→        <KpiCard
   836→          icon={<Edit3 className="w-5 h-5" />}
   837→          label="مسودة"
   838→          value={kpis.draft}
   839→          color="bg-amber-100 text-amber-700"
   840→        />
   841→        <KpiCard
   842→          icon={<ClipboardList className="w-5 h-5" />}
   843→          label="إجمالي التسليمات"
   844→          value={kpis.submissionsTotal}
   845→          color="bg-purple-100 text-purple-700"
   846→        />
   847→      </div>
   848→
   849→      {/* Filter Bar */}
   850→      <Card className="shadow-sm">
   851→        <CardContent className="p-4">
   852→          <div className="flex flex-col sm:flex-row gap-3 items-stretch sm:items-center">
   853→            <div className="flex-1 relative">
   854→              <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   855→              <Input
   856→                value={search}
   857→                onChange={(e) => setSearch(e.target.value)}
   858→                placeholder="ابحث بالعنوان أو الوصف..."
   859→                className="pr-9"

  1780→function QuestionEditor({
  1781→  teacher, examId, question, onClose, onSaved, showToast,
  1782→}: {
  1783→  teacher: TeacherInfo;
  1784→  examId: string;
  1785→  question: Question | null;
  1786→  onClose: () => void;
  1787→  onSaved: () => void;
  1788→  showToast: (type: ToastMessage['type'], text: string) => void;
  1789→}) {
  1790→  const [type, setType] = useState<QuestionType>(question?.type || 'MCQ');
  1791→  const [text, setText] = useState(question?.text || '');
  1792→  const [points, setPoints] = useState(question?.points || 1);
  1793→  const [options, setOptions] = useState<string[]>(question?.options?.length ? question.options : ['', '']);
  1794→  const [correctAnswer, setCorrectAnswer] = useState(question?.correctAnswer || '');
  1795→  const [correctText, setCorrectText] = useState(question?.correctText || '');
  1796→  const [rubric, setRubric] = useState<string>(
  1797→    typeof question?.rubric === 'string' ? question.rubric : (question?.rubric ? JSON.stringify(question.rubric) : '')
  1798→  );
  1799→  const [explanation, setExplanation] = useState(question?.explanation || '');
  1800→  const [saving, setSaving] = useState(false);
  1801→  const [error, setError] = useState<string | null>(null);
  1802→  const [moderation, setModeration] = useState<{ decision: string; reasons: string[] } | null>(null);
  1803→
  1804→  const isEdit = !!question;
  1805→
  1806→  const handleAddOption = () => {
  1807→    if (options.length >= 8) return;
  1808→    setOptions([...options, '']);
  1809→  };
  1810→  const handleRemoveOption = (idx: number) => {
  1811→    if (options.length <= 2) return;
  1812→    const next = options.filter((_, i) => i !== idx);
  1813→    setOptions(next);
  1814→    if (String(correctAnswer) === String(idx)) setCorrectAnswer('');
  1815→    else if (Number(correctAnswer) > idx) setCorrectAnswer(String(Number(correctAnswer) - 1));
  1816→  };
  1817→  const handleOptionChange = (idx: number, value: string) => {
  1818→    const next = [...options];
  1819→    next[idx] = value;
  1820→    setOptions(next);
  1821→  };
  1822→
  1823→  const handleSubmit = async (e: React.FormEvent) => {
  1824→    e.preventDefault();
  1825→    setError(null);
  1826→    setModeration(null);
  1827→    if (!text.trim()) { setError('نص السؤال مطلوب'); return; }
  1828→    if (type === 'MCQ') {
  1829→      const filled = options.filter(o => o.trim());
  1830→      if (filled.length < 2) { setError('خياران على الأقل مطلوبان للاختيار من متعدد'); return; }
  1831→      if (correctAnswer === '' || Number(correctAnswer) < 0 || Number(correctAnswer) >= filled.length) {
  1832→        setError('حدّد الإجابة الصحيحة'); return;
  1833→      }
  1834→    }
  1835→    if (type === 'TRUE_FALSE' && !correctAnswer) {
  1836→      setError('حدّد الإجابة الصحيحة (صح أو خطأ)'); return;
  1837→    }
  1838→
  1839→    setSaving(true);
  1840→    try {
  1841→      const body: Record<string, unknown> = {
  1842→        type,
  1843→        text: text.trim(),
  1844→        points: Number(points),
  1845→        explanation: explanation.trim() || undefined,
  1846→      };
  1847→      if (type === 'MCQ') {
  1848→        body.options = options.filter(o => o.trim());
  1849→        body.correctAnswer = String(correctAnswer);
  1850→      } else if (type === 'TRUE_FALSE') {
  1851→        body.correctAnswer = correctAnswer;
  1852→      } else if (type === 'SHORT') {
  1853→        if (correctText.trim()) body.correctText = correctText.trim();
  1854→      } else if (type === 'ESSAY') {
  1855→        if (rubric.trim()) body.rubric = rubric.trim();
  1856→      }
  1857→
  1858→      if (isEdit && question) {
  1859→        const url = buildUrl(`/api/exams/teacher/${examId}/questions/${question.id}`, teacher);
  1860→        await teacherFetch<{ success: boolean }>(url, {
  1861→          method: 'PUT',
  1862→          body: JSON.stringify(body),
  1863→        });
  1864→        showToast('success', 'تم تحديث السؤال');
  1865→      } else {
  1866→        const url = buildUrl(`/api/exams/teacher/${examId}/questions`, teacher);
  1867→        const data = await teacherFetch<{ success: boolean; moderation?: { decision: string; reasons: string[] } }>(url, {
  1868→          method: 'POST',
  1869→          body: JSON.stringify(body),
  1870→        });
  1871→        showToast('success', 'تمت إضافة السؤال');
  1872→        if (data.moderation && data.moderation.decision !== 'SAFE') {
  1873→          setModeration(data.moderation);
  1874→          setSaving(false);
  1875→          return; // keep dialog open to show moderation warning
  1876→        }
  1877→      }
  1878→      onSaved();
  1879→    } catch (err: unknown) {
  1880→      const msg = err instanceof Error ? err.message : 'فشل حفظ السؤال';
  1881→      setError(msg);
  1882→    } finally {
  1883→      setSaving(false);
  1884→    }
  1885→  };
  1886→
  1887→  return (
  1888→    <Dialog open onOpenChange={(o) => !o && onClose()}>
  1889→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
  1890→        <DialogHeader>
  1891→          <DialogTitle className="text-right">
  1892→            {isEdit ? 'تحرير السؤال' : 'إضافة سؤال جديد'}
  1893→          </DialogTitle>
  1894→          <DialogDescription className="text-right">
  1895→            {isEdit ? 'عدّل بيانات السؤال ثم احفظ' : 'اختر نوع السؤال وأدخل بياناته'}
  1896→          </DialogDescription>
  1897→        </DialogHeader>
  1898→
  1899→        <form onSubmit={handleSubmit} className="space-y-4">
  1900→          {/* Type */}
  1901→          <div className="space-y-2">
  1902→            <Label className="text-sm font-medium">نوع السؤال</Label>
  1903→            <Select value={type} onValueChange={(v) => { setType(v as QuestionType); setCorrectAnswer(''); }} disabled={isEdit}>
  1904→              <SelectTrigger>
  1905→                <SelectValue />
  1906→              </SelectTrigger>
  1907→              <SelectContent>
  1908→                {(Object.keys(QUESTION_TYPE_LABELS) as QuestionType[]).map(t => (
  1909→                  <SelectItem key={t} value={t}>{QUESTION_TYPE_LABELS[t]}</SelectItem>
  1910→                ))}
  1911→              </SelectContent>
  1912→            </Select>
  1913→          </div>
  1914→
  1915→          {/* Text */}
  1916→          <div className="space-y-2">
  1917→            <Label className="text-sm font-medium">نص السؤال <span className="text-red-500">*</span></Label>
  1918→            <Textarea
  1919→              value={text}
  1920→              onChange={(e) => setText(e.target.value)}
  1921→              placeholder="اكتب نص السؤال هنا..."
  1922→              rows={3}
  1923→            />
  1924→          </div>
  1925→
  1926→          {/* Points */}
  1927→          <div className="space-y-2 w-32">
  1928→            <Label className="text-sm font-medium">الدرجة</Label>
  1929→            <Input
  1930→              type="number"
  1931→              min={1}
  1932→              value={points}
  1933→              onChange={(e) => setPoints(Number(e.target.value))}
  1934→            />
  1935→          </div>
  1936→
  1937→          {/* Conditional fields */}
  1938→          {type === 'MCQ' && (
  1939→            <div className="space-y-2">
  1940→              <Label className="text-sm font-medium">الخيارات (2 إلى 8) — حدّد الإجابة الصحيحة</Label>
  1941→              <RadioGroup value={correctAnswer} onValueChange={setCorrectAnswer}>
  1942→                <div className="space-y-2">
  1943→                  {options.map((opt, idx) => (
  1944→                    <div key={idx} className="flex items-center gap-2">
  1945→                      <RadioGroupItem value={String(idx)} id={`opt-${idx}`} />
  1946→                      <Input
  1947→                        value={opt}
  1948→                        onChange={(e) => handleOptionChange(idx, e.target.value)}

  1952→                      <Button

  1954→                        variant="ghost"
  1955→                        size="sm"
  1956→                        onClick={() => handleRemoveOption(idx)}
  1957→                        disabled={options.length <= 2}
  1958→                        className="h-9 w-9 p-0"
  1959→                      >
  1960→                        <X className="w-4 h-4 text-red-500" />
  1961→                      </Button>
  1962→                    </div>
  1963→                  ))}
  1964→                </div>
  1965→              </RadioGroup>
  1966→              {options.length < 8 && (

  1968→                  <Plus className="w-4 h-4 ml-1" />

  1974→
  1975→          {type === 'TRUE_FALSE' && (
