   102→function HomePage() {
   103→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated, setHasHydrated } = useAdminStore()
   104→  const searchParams = useSearchParams()
   105→  const router = useRouter()
   106→  const pathname = usePathname()
   107→  const urlSubdomain = searchParams.get('school')
   108→  const urlAction = searchParams.get('action')
   109→  const hasAppliedUrlSchool = useRef(false)
   110→
   111→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   112→  const [news, setNews] = useState<NewsItem[]>([])
   113→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   114→  const [teachers, setTeachers] = useState<Teacher[]>([])
   115→  const [sliders, setSliders] = useState<SliderItem[]>([])
   116→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   117→  const [loading, setLoading] = useState(true)
   118→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
   119→  const [servicesOpen, setServicesOpen] = useState(false)
   120→  const [showAdminLogin, setShowAdminLogin] = useState(false)
   121→  const [showResultsPage, setShowResultsPage] = useState(false)
   122→  const [showStudentLife, setShowStudentLife] = useState(false)
   123→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   124→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   125→  const [showParentsPage, setShowParentsPage] = useState(false)
   126→  const [showExamsPage, setShowExamsPage] = useState(false)
   127→  const [showScrollTop, setShowScrollTop] = useState(false)
   128→  const [currentSlide, setCurrentSlide] = useState(0)
   129→  const logoClickCount = useRef(0)
   130→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   131→
   132→  const hydrated = _hasHydrated
   133→
   134→  // Fallback: ensure hydration flag is set on the client (Zustand v5
   135→  // onRehydrateStorage may not fire in all environments).
   136→  useEffect(() => {
   137→    if (!_hasHydrated) setHasHydrated(true)
   138→  }, [_hasHydrated, setHasHydrated])
   139→
   140→  // Resolve school from subdomain only (not from store's school list)
   141→  useEffect(() => {
   142→    if (!hydrated) return
   143→    let cancelled = false
   144→    async function resolveSchool() {
   145→      try {
   146→        if (urlSubdomain) {
   147→          // If subdomain is in URL, fetch by subdomain
   148→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   149→          if (res.ok && !cancelled) {
   150→            const data = await res.json()
   151→            if (data.school) {
   152→              setSchoolData(data)
   153→              setSelectedSchoolId(data.school.id)
   154→              return
   155→            }
   156→          }
   157→        }
   158→        // No subdomain in URL — fetch first available school
   159→        const schoolsRes = await fetch('/api/schools')
   160→        if (schoolsRes.ok && !cancelled) {
   161→          const schoolsList = await schoolsRes.json()
   162→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   163→            setSchools(schoolsList)
   164→            const firstSchool = schoolsList[0]
   165→            // Fetch full data for the first school
   166→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   167→            if (res.ok && !cancelled) {
   168→              const data = await res.json()
   169→              if (data.school) {
   170→                setSchoolData(data)
   171→                setSelectedSchoolId(data.school.id)
   172→              }
   173→            }
   174→          }
   175→        }
   176→      } catch {
   177→        // fallback to default data
   178→      }
   179→    }
   180→    resolveSchool()
   181→    return () => { cancelled = true }
   182→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   183→
   184→  // Fetch content for selected school
   185→  useEffect(() => {
   186→    if (!hydrated || !selectedSchoolId) return
   187→    let cancelled = false
   188→    const sid = selectedSchoolId
   189→    async function loadData() {
   190→      setLoading(true)
   191→      try {
   192→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   193→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   194→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   195→          fetch(`/api/teachers?schoolId=${sid}`),
   196→          fetch(`/api/sliders?schoolId=${sid}`),
   197→          fetch(`/api/custom-sections?schoolId=${sid}`),
   198→        ])
   199→        if (cancelled) return
   200→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   201→          const d = await newsRes.value.json()
   202→          if (Array.isArray(d) && !cancelled) setNews(d)
   203→        }
   204→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   205→          const d = await galleryRes.value.json()
   206→          if (Array.isArray(d) && !cancelled) setGallery(d)
   207→        }
   208→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   209→          const d = await teachersRes.value.json()
   210→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   211→        }
   212→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   213→          const d = await slidersRes.value.json()
   214→          if (Array.isArray(d) && !cancelled) setSliders(d)
   215→        }
   216→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   217→          const d = await csRes.value.json()
   218→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   219→        }
   220→      } catch {
   221→        // silently fail
   222→      } finally {
   223→        if (!cancelled) setLoading(false)
   224→      }
   225→    }
   226→    loadData()
   227→    return () => { cancelled = true }
   228→  }, [selectedSchoolId, hydrated])
   229→
   230→  // Auto-rotate slider
   231→  const activeSliders = sliders.filter(s => s.active)
   232→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   233→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   234→  useEffect(() => {
   235→    if (slideItems.length <= 1) return
   236→    const timer = setInterval(() => setCurrentSlide(i => (i + 1) % slideItems.length), 5000)
   237→    return () => clearInterval(timer)
   238→  }, [slideItems.length])
   239→
   240→  // Scroll to top
   241→  useEffect(() => {
   242→    const h = () => setShowScrollTop(window.scrollY > 500)
   243→    window.addEventListener('scroll', h)
   244→    return () => window.removeEventListener('scroll', h)
   245→  }, [])
   246→
   247→  const handleLogoClick = useCallback(() => {
   248→    logoClickCount.current += 1
   249→    if (logoClickTimer.current) clearTimeout(logoClickTimer.current)
   250→    logoClickTimer.current = setTimeout(() => { logoClickCount.current = 0 }, 3000)
   251→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   252→  }, [])
   253→
   254→  // معالجة اختصارات PWA (?action=...) عند فتح التطبيق من الشاشة الرئيسية
   255→  const urlActionProcessed = useRef(false)
   256→  useEffect(() => {
   257→    if (!urlAction || !hydrated || urlActionProcessed.current) return
   258→    urlActionProcessed.current = true
   259→    // تنظيف الـ URL من معامل action بعد تفعيله
   260→    router.replace('/')
   261→    // استخدام microtask لتجنّب تحذير set-state-in-effect
   262→    queueMicrotask(() => {
   263→      if (urlAction === 'exams') { setShowExamsPage(true) }
   264→      else if (urlAction === 'results') setShowResultsPage(true)
   265→      else if (urlAction === 'schedules') setShowSchedulesPage(true)
   266→      else if (urlAction === 'library') setShowLibraryPage(true)
   267→      else if (urlAction === 'parents') setShowParentsPage(true)
   268→    })
   269→  }, [urlAction, hydrated, router])
   270→
   271→  const handleServiceAction = (action: string) => {
   272→    setServicesOpen(false); setMobileMenuOpen(false)
   273→    if (action === 'results') setShowResultsPage(true)
   274→    if (action === 'schedules') setShowSchedulesPage(true)
   275→    if (action === 'library') setShowLibraryPage(true)
   276→    if (action === 'parents') setShowParentsPage(true)
   277→    if (action === 'exams') setShowExamsPage(true)
   278→  }
   279→
   280→  // الانتقال من بوابة أولياء الأمور إلى صفحة الامتحانات الإلكترونية
   281→  const handleOpenExamsFromParents = useCallback(() => {
   282→    setShowParentsPage(false)
   283→    setShowExamsPage(true)
   284→  }, [])
   285→
   286→  const school = schoolData?.school || defaultSchoolData.school
   287→  const settings = schoolData?.settings || defaultSchoolData.settings
   288→  const stats = schoolData?.stats || defaultSchoolData.stats
   289→  const alertNews = news.filter(n => n.category === 'تنبيه')
   290→
   291→  // Build ticker text for seamless scrolling
   292→  const tickerText = news.map(item => item.title).join('     ◆     ')
   293→
   294→  // Sub-page rendering
   295→  if (isAdminMode) return <AdminLayout />
   296→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   297→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   298→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   299→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />
   300→  if (showParentsPage) return <ParentsPortalPage onBack={() => setShowParentsPage(false)} schoolId={selectedSchoolId} onOpenExams={handleOpenExamsFromParents} />
   301→  if (showExamsPage) return <StudentExamsPage onBack={() => setShowExamsPage(false)} schoolId={selectedSchoolId} />
   302→
   303→  if (!hydrated) {
   304→    return (
   305→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   306→        <div className="text-center">
   307→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />
   308→          <p className="text-gray-500 text-sm">جاري التحميل...</p>
   309→        </div>
   310→      </div>
   311→    )
   312→  }
   313→
   314→  return (
   315→    <div className="min-h-screen flex flex-col" dir="rtl">
   316→
   317→      {/* ===== STICKY TOP ===== */}
   318→      <div className="sticky top-0 z-50 shadow-lg">
   319→        {/* Top Bar - Contact info only, NO school switcher for visitors */}
   320→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">
   321→          <div className="max-w-[1280px] mx-auto px-4 py-1.5 flex items-center justify-between">
   322→            <div className="flex items-center gap-4">
   323→              {school.phone && <a href={`tel:${school.phone}`} className="hover:opacity-80 flex items-center gap-1">📞 {school.phone}</a>}
   324→              {school.email && <a href={`mailto:${school.email}`} className="hover:opacity-80 flex items-center gap-1">✉️ {school.email}</a>}
   325→            </div>
   326→            <div className="flex items-center gap-2">
   327→              {school.facebookUrl && (
   328→                <a href={school.facebookUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📘</a>
   329→              )}
   330→              {settings?.youtubeUrl && (
   331→                <a href={settings.youtubeUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📺</a>
   332→              )}
   333→            </div>
   334→          </div>
   335→        </div>
   336→
   337→        {/* Header */}
   338→        <header className="bg-white shadow-sm">
   339→          <div className="max-w-[1280px] mx-auto px-4 py-2 flex items-center gap-3">
   340→            <div onClick={handleLogoClick} className="cursor-pointer shrink-0" title="اضغط 5 مرات للإدارة">
   341→              {school.logoUrl ? (
   342→                <img src={school.logoUrl} alt={school.name} className="w-12 h-12 rounded-full object-cover border-2" style={{ borderColor: school.primaryColor }} />
   343→              ) : (
   344→                <div className="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl" style={{ backgroundColor: school.primaryColor }}>🏫</div>
   345→              )}
   346→            </div>
   347→            <div className="flex-1 min-w-0">
   348→              <h1 className="text-base md:text-xl font-bold truncate" style={{ color: school.primaryColor }}>{school.name}</h1>
   349→              {settings?.heroSubtitle && <p className="text-xs text-gray-500 truncate">{settings.heroSubtitle}</p>}
   350→            </div>
   351→            <button className="lg:hidden p-2 text-2xl text-gray-600" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   352→              {mobileMenuOpen ? '✕' : '☰'}
   353→            </button>
   354→          </div>
   355→        </header>
   356→
   357→        {/* Navigation */}
   358→        <nav style={{ backgroundColor: '#1a2332' }} className="text-white shadow-lg">
   359→          <div className="max-w-[1280px] mx-auto px-4">
   360→            {/* Desktop Nav */}
   361→            <div className="hidden lg:flex items-center h-11">
   362→              {navLinks.map((link) => (
   363→                link.isServices ? (
   364→                  <div key={link.href} className="relative" onMouseEnter={() => setServicesOpen(true)} onMouseLeave={() => setServicesOpen(false)}>
   365→                    <button className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center gap-1 transition-colors">
   366→                      {link.label} ▾
   367→                    </button>
   368→                    {servicesOpen && (
   369→                      <div className="absolute top-full right-0 bg-white rounded-lg shadow-xl border min-w-[220px] z-50 overflow-hidden">
   370→                        {serviceItems.map(item => (
   371→                          <button key={item.action} onClick={() => handleServiceAction(item.action)}
   372→                            className="w-full flex items-center gap-3 px-4 py-3 text-sm text-gray-700 hover:bg-gray-50 transition-colors">
   373→                            <span className="text-lg">{item.icon}</span> {item.label}
   374→                          </button>
   375→                        ))}
   376→                      </div>
   377→                    )}
   378→                  </div>
   379→                ) : (
   380→                  <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   381→                    onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true) } : undefined}
   382→                    className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center transition-colors">
   383→                    {link.label}
   384→                  </a>
   385→                )
   386→              ))}
   387→            </div>
   388→            {/* Mobile Nav */}
   389→            {mobileMenuOpen && (
   390→              <div className="lg:hidden py-2 border-t border-white/10">
   391→                {navLinks.map((link) => (
   392→                  link.isServices ? (
   393→                    <div key={link.href}>
   394→                      <div className="px-4 py-2 text-sm font-medium text-white/70">{link.label}</div>
   395→                      {serviceItems.map(item => (
   396→                        <button key={item.action} onClick={() => handleServiceAction(item.action)}
   397→                          className="w-full flex items-center gap-2 px-8 py-2 text-sm text-white/80 hover:bg-white/10">
   398→                          <span>{item.icon}</span> {item.label}
   399→                        </button>
   400→                      ))}
   401→                    </div>
   402→                  ) : (
   403→                    <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   404→                      onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true); setMobileMenuOpen(false) } : () => setMobileMenuOpen(false)}
   405→                      className="block px-4 py-2 text-sm hover:bg-white/10">{link.label}</a>
   406→                  )
   407→                ))}
   408→              </div>
   409→            )}
   410→          </div>
   411→        </nav>
   412→
   413→        {/* News Ticker - Seamless scrolling */}
   414→        {settings?.showNewsTicker && news.length > 0 && (
   415→          <div className="bg-gray-100 border-b overflow-hidden">
   416→            <div className="flex items-center">
   417→              <span className="bg-red-600 text-white px-4 py-2 text-xs font-bold shrink-0 flex items-center gap-1 z-10">
   418→                🔴 عاجل
   419→              </span>
   420→              <div className="overflow-hidden flex-1 relative">
   421→                <div className="animate-news-ticker-scroll whitespace-nowrap py-2">
   422→                  {/* Duplicate content for seamless loop */}
   423→                  <span className="inline-block text-sm mx-4">
   424→                    {tickerText}
   425→                  </span>
   426→                  <span className="inline-block text-sm mx-4">
   427→                    {tickerText}
   428→                  </span>
   429→                </div>
