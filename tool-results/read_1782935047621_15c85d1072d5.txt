   107→
   108→function HomePage() {
   109→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated, setHasHydrated } = useAdminStore()
   110→  const searchParams = useSearchParams()
   111→  const router = useRouter()
   112→  const pathname = usePathname()
   113→  const urlSubdomain = searchParams.get('school')
   114→  const urlAction = searchParams.get('action')
   115→  const hasAppliedUrlSchool = useRef(false)
   116→
   117→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   118→  const [news, setNews] = useState<NewsItem[]>([])
   119→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   120→  const [teachers, setTeachers] = useState<Teacher[]>([])
   121→  const [sliders, setSliders] = useState<SliderItem[]>([])
   122→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   123→  const [loading, setLoading] = useState(true)
   124→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
   125→  const [servicesOpen, setServicesOpen] = useState(false)
   126→  const [showAdminLogin, setShowAdminLogin] = useState(false)
   127→  const [showResultsPage, setShowResultsPage] = useState(false)
   128→  const [showStudentLife, setShowStudentLife] = useState(false)
   129→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   130→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   131→  const [showParentsPage, setShowParentsPage] = useState(false)
   132→  const [showExamsPage, setShowExamsPage] = useState(false)
   133→  const [showAnalyticsPage, setShowAnalyticsPage] = useState(false)
   134→  const [showTeacherPage, setShowTeacherPage] = useState(false)
   135→  const [showCoordinatorPage, setShowCoordinatorPage] = useState(false)
   136→  const [showScrollTop, setShowScrollTop] = useState(false)
   137→  const [currentSlide, setCurrentSlide] = useState(0)
   138→  const logoClickCount = useRef(0)
   139→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   140→
   141→  const hydrated = _hasHydrated
   142→
   143→  // Fallback: ensure hydration flag is set on the client (Zustand v5
   144→  // onRehydrateStorage may not fire in all environments).
   145→  useEffect(() => {
   146→    if (!_hasHydrated) setHasHydrated(true)
   147→  }, [_hasHydrated, setHasHydrated])
   148→
   149→  // Resolve school from subdomain only (not from store's school list)
   150→  useEffect(() => {
   151→    if (!hydrated) return
   152→    let cancelled = false
   153→    async function resolveSchool() {
   154→      try {
   155→        if (urlSubdomain) {
   156→          // If subdomain is in URL, fetch by subdomain
   157→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   158→          if (res.ok && !cancelled) {
   159→            const data = await res.json()
   160→            if (data.school) {
   161→              setSchoolData(data)
   162→              setSelectedSchoolId(data.school.id)
   163→              return
   164→            }
   165→          }
   166→        }
   167→        // No subdomain in URL — fetch first available school
   168→        const schoolsRes = await fetch('/api/schools')
   169→        if (schoolsRes.ok && !cancelled) {
   170→          const schoolsList = await schoolsRes.json()
   171→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   172→            setSchools(schoolsList)
   173→            const firstSchool = schoolsList[0]
   174→            // Fetch full data for the first school
   175→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   176→            if (res.ok && !cancelled) {
   177→              const data = await res.json()
   178→              if (data.school) {
   179→                setSchoolData(data)
   180→                setSelectedSchoolId(data.school.id)
   181→              }
   182→            }
   183→          }
   184→        }
   185→      } catch {
   186→        // fallback to default data
   187→      }
   188→    }
   189→    resolveSchool()
   190→    return () => { cancelled = true }
   191→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   192→
   193→  // Fetch content for selected school
   194→  useEffect(() => {
   195→    if (!hydrated || !selectedSchoolId) return
   196→    let cancelled = false
   197→    const sid = selectedSchoolId
   198→    async function loadData() {
   199→      setLoading(true)
   200→      try {
   201→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   202→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   203→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   204→          fetch(`/api/teachers?schoolId=${sid}`),
   205→          fetch(`/api/sliders?schoolId=${sid}`),
   206→          fetch(`/api/custom-sections?schoolId=${sid}`),
   207→        ])
   208→        if (cancelled) return
   209→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   210→          const d = await newsRes.value.json()
   211→          if (Array.isArray(d) && !cancelled) setNews(d)
   212→        }
   213→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   214→          const d = await galleryRes.value.json()
   215→          if (Array.isArray(d) && !cancelled) setGallery(d)
   216→        }
   217→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   218→          const d = await teachersRes.value.json()
   219→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   220→        }
   221→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   222→          const d = await slidersRes.value.json()
   223→          if (Array.isArray(d) && !cancelled) setSliders(d)
   224→        }
   225→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   226→          const d = await csRes.value.json()
   227→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   228→        }
   229→      } catch {
   230→        // silently fail
   231→      } finally {
   232→        if (!cancelled) setLoading(false)
   233→      }
   234→    }
   235→    loadData()
   236→    return () => { cancelled = true }
   237→  }, [selectedSchoolId, hydrated])
   238→
   239→  // Auto-rotate slider
   240→  const activeSliders = sliders.filter(s => s.active)
   241→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   242→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   243→  useEffect(() => {
   244→    if (slideItems.length <= 1) return
   245→    const timer = setInterval(() => setCurrentSlide(i => (i + 1) % slideItems.length), 5000)
   246→    return () => clearInterval(timer)
   247→  }, [slideItems.length])
   248→
   249→  // Scroll to top
   250→  useEffect(() => {
   251→    const h = () => setShowScrollTop(window.scrollY > 500)
   252→    window.addEventListener('scroll', h)
   253→    return () => window.removeEventListener('scroll', h)
   254→  }, [])
   255→
   256→  const handleLogoClick = useCallback(() => {
   257→    logoClickCount.current += 1
   258→    if (logoClickTimer.current) clearTimeout(logoClickTimer.current)
   259→    logoClickTimer.current = setTimeout(() => { logoClickCount.current = 0 }, 3000)
   260→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   261→  }, [])
   262→
   263→  // معالجة اختصارات PWA (?action=...) عند فتح التطبيق من الشاشة الرئيسية
   264→  const urlActionProcessed = useRef(false)
   265→  useEffect(() => {
   266→    if (!urlAction || !hydrated || urlActionProcessed.current) return
   267→    urlActionProcessed.current = true
   268→    // تنظيف الـ URL من معامل action بعد تفعيله
   269→    router.replace('/')
   270→    // استخدام microtask لتجنّب تحذير set-state-in-effect
   271→    queueMicrotask(() => {
   272→      if (urlAction === 'exams') { setShowExamsPage(true) }
   273→      else if (urlAction === 'results') setShowResultsPage(true)
   274→      else if (urlAction === 'schedules') setShowSchedulesPage(true)
   275→      else if (urlAction === 'library') setShowLibraryPage(true)
   276→      else if (urlAction === 'parents') setShowParentsPage(true)
   277→      else if (urlAction === 'teacher') setShowTeacherPage(true)
   278→      else if (urlAction === 'coordinator') setShowCoordinatorPage(true)
   279→    })
   280→  }, [urlAction, hydrated, router])
   281→
   282→  const handleServiceAction = (action: string) => {
   283→    setServicesOpen(false); setMobileMenuOpen(false)
   284→    if (action === 'results') setShowResultsPage(true)
   285→    if (action === 'schedules') setShowSchedulesPage(true)
   286→    if (action === 'library') setShowLibraryPage(true)
   287→    if (action === 'parents') setShowParentsPage(true)
   288→    if (action === 'exams') setShowExamsPage(true)
   289→    if (action === 'teacher') setShowTeacherPage(true)
   290→    if (action === 'coordinator') setShowCoordinatorPage(true)
   291→  }
   292→
   293→  // الانتقال من بوابة أولياء الأمور إلى صفحة الامتحانات الإلكترونية
   294→  const handleOpenExamsFromParents = useCallback(() => {
   295→    setShowParentsPage(false)
   296→    setShowExamsPage(true)
   297→  }, [])
   298→
   299→  // الانتقال من بوابة أولياء الأمور إلى صفحة التحليلات المقارنة
   300→  const handleOpenAnalyticsFromParents = useCallback(() => {
   301→    setShowParentsPage(false)
   302→    setShowAnalyticsPage(true)
   303→  }, [])
   304→
   305→  // الانتقال من بوابة أولياء الأمور إلى صفحة الجداول
   306→  const handleOpenSchedulesFromParents = useCallback(() => {
   307→    setShowParentsPage(false)
   308→    setShowSchedulesPage(true)
   309→  }, [])
   310→
   311→  // الانتقال من بوابة أولياء الأمور إلى صفحة المكتبة الرقمية
   312→  const handleOpenLibraryFromParents = useCallback(() => {
   313→    setShowParentsPage(false)
   314→    setShowLibraryPage(true)
   315→  }, [])
   316→
   317→  const school = schoolData?.school || defaultSchoolData.school
   318→  const settings = schoolData?.settings || defaultSchoolData.settings
   319→  const stats = schoolData?.stats || defaultSchoolData.stats
   320→  const alertNews = news.filter(n => n.category === 'تنبيه')
   321→
   322→  // Build ticker text for seamless scrolling
   323→  const tickerText = news.map(item => item.title).join('     ◆     ')
   324→
   325→  // Sub-page rendering
   326→  if (isAdminMode) return <AdminLayout />
   327→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   328→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   329→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   330→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />
   331→  if (showParentsPage) return <ParentsPortalPage onBack={() => setShowParentsPage(false)} schoolId={selectedSchoolId} onOpenExams={handleOpenExamsFromParents} onOpenAnalytics={handleOpenAnalyticsFromParents} onOpenSchedules={handleOpenSchedulesFromParents} onOpenLibrary={handleOpenLibraryFromParents} />
   332→  if (showExamsPage) return <StudentExamsPage onBack={() => setShowExamsPage(false)} schoolId={selectedSchoolId} />
   333→  if (showAnalyticsPage) return <ParentAnalyticsPage onBack={() => setShowAnalyticsPage(false)} schoolId={selectedSchoolId} />
   334→  if (showTeacherPage) return <TeacherExamsPage onBack={() => setShowTeacherPage(false)} schoolId={selectedSchoolId} />
   335→  if (showCoordinatorPage) return <CoordinatorDashboardPage onBack={() => setShowCoordinatorPage(false)} schoolId={selectedSchoolId} />
   336→
   337→  if (!hydrated) {
   338→    return (
   339→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   340→        <div className="text-center">
   341→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />
   342→          <p className="text-gray-500 text-sm">جاري التحميل...</p>
   343→        </div>
   344→      </div>
   345→    )
   346→  }
   347→
   348→  return (
   349→    <div className="min-h-screen flex flex-col" dir="rtl">
   350→
   351→      {/* ===== STICKY TOP ===== */}
   352→      <div className="sticky top-0 z-50 shadow-lg">
   353→        {/* Top Bar - Contact info only, NO school switcher for visitors */}
   354→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">
   355→          <div className="max-w-[1280px] mx-auto px-4 py-1.5 flex items-center justify-between">
   356→            <div className="flex items-center gap-4">
   357→              {school.phone && <a href={`tel:${school.phone}`} className="hover:opacity-80 flex items-center gap-1">📞 {school.phone}</a>}
   358→              {school.email && <a href={`mailto:${school.email}`} className="hover:opacity-80 flex items-center gap-1">✉️ {school.email}</a>}
   359→            </div>
   360→            <div className="flex items-center gap-2">
   361→              {school.facebookUrl && (
   362→                <a href={school.facebookUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📘</a>
   363→              )}
   364→              {settings?.youtubeUrl && (
   365→                <a href={settings.youtubeUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📺</a>
   366→              )}
   367→            </div>
   368→          </div>
   369→        </div>
   370→
   371→        {/* Header */}
   372→        <header className="bg-white shadow-sm">
   373→          <div className="max-w-[1280px] mx-auto px-4 py-2 flex items-center gap-3">
   374→            <div onClick={handleLogoClick} className="cursor-pointer shrink-0" title="اضغط 5 مرات للإدارة">
   375→              {school.logoUrl ? (
   376→                <img src={school.logoUrl} alt={school.name} className="w-12 h-12 rounded-full object-cover border-2" style={{ borderColor: school.primaryColor }} />
   377→              ) : (
   378→                <div className="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl" style={{ backgroundColor: school.primaryColor }}>🏫</div>
   379→              )}
   380→            </div>
   381→            <div className="flex-1 min-w-0">
   382→              <h1 className="text-base md:text-xl font-bold truncate" style={{ color: school.primaryColor }}>{school.name}</h1>
   383→              {settings?.heroSubtitle && <p className="text-xs text-gray-500 truncate">{settings.heroSubtitle}</p>}
   384→            </div>
   385→            {selectedSchoolId && (
   386→              <NotificationBell schoolId={selectedSchoolId} variant="header" />
   387→            )}
   388→            <button className="lg:hidden p-2 text-2xl text-gray-600" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   389→              {mobileMenuOpen ? '✕' : '☰'}
   390→            </button>
   391→          </div>
   392→        </header>
   393→
   394→        {/* Navigation */}
   395→        <nav style={{ backgroundColor: '#1a2332' }} className="text-white shadow-lg">
   396→          <div className="max-w-[1280px] mx-auto px-4">
   397→            {/* Desktop Nav */}
   398→            <div className="hidden lg:flex items-center h-11">
   399→              {navLinks.map((link) => (
   400→                link.isServices ? (
   401→                  <div key={link.href} className="relative" onMouseEnter={() => setServicesOpen(true)} onMouseLeave={() => setServicesOpen(false)}>
   402→                    <button className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center gap-1 transition-colors">
   403→                      {link.label} ▾
   404→                    </button>
   405→                    {servicesOpen && (
   406→                      <div className="absolute top-full right-0 bg-white rounded-lg shadow-xl border min-w-[220px] z-50 overflow-hidden">
   407→                        {serviceItems.map(item => (
   408→                          <button key={item.action} onClick={() => handleServiceAction(item.action)}
   409→                            className="w-full flex items-center gap-3 px-4 py-3 text-sm text-gray-700 hover:bg-gray-50 transition-colors">
   410→                            <span className="text-lg">{item.icon}</span> {item.label}
   411→                          </button>
   412→                        ))}
   413→                      </div>
