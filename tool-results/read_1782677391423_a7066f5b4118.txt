    84→function parseTargetIds(raw: string | string[] | null): string[] {
    85→  if (!raw) return []
    86→  if (Array.isArray(raw)) return raw
    87→  try {
    88→    const parsed: unknown = JSON.parse(raw)
    89→    if (Array.isArray(parsed)) {
    90→      return parsed.filter((x): x is string => typeof x === 'string')
    91→    }
    92→    return []
    93→  } catch {
    94→    return []
    95→  }
    96→}
    97→
    98→function extractErrorMessage(data: unknown, fallback: string): string {
    99→  if (data && typeof data === 'object' && 'error' in data) {
   100→    const err = (data as { error?: unknown }).error
   101→    if (typeof err === 'string') return err
   102→  }
   103→  return fallback
   104→}
