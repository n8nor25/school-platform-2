    84→// ===== Helper: Parse stops JSON =====
    85→function parseStops(stopsJson: string): Stop[] {
    86→  try {
    87→    const parsed = JSON.parse(stopsJson)
    88→    if (Array.isArray(parsed)) {
    89→      return parsed.map((s: Record<string, string>) => ({
    90→        name: s.name || '',
    91→        time: s.time || '',
    92→      }))
    93→    }
    94→    return []
    95→  } catch {
    96→    return []
    97→  }
    98→}
    99→
   100→// ===== Helper: Stringify stops =====
   101→function stringifyStops(stops: Stop[]): string {
   102→  return JSON.stringify(stops)
   103→}

   105→// ===== Stops Timeline Component =====
   106→function StopsTimeline({ stops }: { stops: Stop[] }) {
   107→  if (stops.length === 0) {
   108→    return (
   109→      <p className="text-sm text-gray-400 italic">لا توجد محطات</p>
   110→    )
   111→  }
   112→
   113→  return (
   114→    <div className="flex flex-row-reverse items-start gap-0">
   115→      {stops.map((stop, index) => (
   116→        <div key={index} className="flex flex-col items-center flex-1 min-w-0">
   117→          {/* Stop circle */}
   118→          <div className="relative flex items-center w-full">
   119→            {/* Line before (right side in RTL) */}
   120→            {index < stops.length - 1 && (
   121→              <div className="flex-1 h-0.5 bg-teal-300" />
   122→            )}
   123→            {/* Circle */}
   124→            <div
   125→              className={`w-6 h-6 rounded-full flex items-center justify-center shrink-0 border-2 ${
   126→                index === 0
   127→                  ? 'bg-teal-600 border-teal-600 text-white'
   128→                  : index === stops.length - 1
   129→                    ? 'bg-red-500 border-red-500 text-white'
   130→                    : 'bg-white border-teal-400 text-teal-600'
   131→              }`}
   132→            >
   133→              <span className="text-[10px] font-bold">
   134→                {index === 0 ? '●' : index === stops.length - 1 ? '■' : index}
   135→              </span>
   136→            </div>
   137→            {/* Line after (left side in RTL) */}
   138→            {index > 0 && (
   139→              <div className="flex-1 h-0.5 bg-teal-300" />
   140→            )}
   141→          </div>
   142→          {/* Stop info */}
   143→          <div className="mt-1.5 text-center px-0.5">
   144→            <p className="text-xs font-medium text-gray-700 truncate max-w-[80px]">
   145→              {stop.name}
   146→            </p>
   147→            {stop.time && (
   148→              <p className="text-[10px] text-teal-600 font-medium">{stop.time}</p>
   149→            )}
   150→          </div>
   151→        </div>
   152→      ))}
   153→    </div>
   154→  )
   155→}

   157→// ===== Stops Editor Component =====
   158→function StopsEditor({
   159→  stops,
   160→  onChange,
   161→}: {
   162→  stops: Stop[]
   163→  onChange: (stops: Stop[]) => void
   164→}) {
   165→  const addStop = () => {
   166→    onChange([...stops, { name: '', time: '' }])
   167→  }
   168→
   169→  const removeStop = (index: number) => {
   170→    onChange(stops.filter((_, i) => i !== index))
   171→  }
   172→
   173→  const updateStop = (index: number, field: keyof Stop, value: string) => {
   174→    const updated = [...stops]
   175→    updated[index] = { ...updated[index], [field]: value }
   176→    onChange(updated)
   177→  }
   178→
   179→  return (
   180→    <div className="space-y-3">
   181→      <div className="flex items-center justify-between">
   182→        <Label className="text-sm font-semibold text-gray-700">المحطات</Label>
   183→        <Button
   184→          type="button"
   185→          variant="outline"
   186→          size="sm"
   187→          onClick={addStop}
   188→          className="min-h-[44px] text-teal-700 border-teal-300 hover:bg-teal-50"
   189→        >
   190→          <Plus className="w-4 h-4 ml-1" />
   191→          إضافة محطة
   192→        </Button>
   193→      </div>
   194→
   195→      {stops.length === 0 && (
   196→        <p className="text-sm text-gray-400 italic text-center py-3">
   197→          لم يتم إضافة محطات بعد. اضغط &quot;إضافة محطة&quot; للبدء.
   198→        </p>
   199→      )}
   200→
   201→      <div className="space-y-2">
   202→        {stops.map((stop, index) => (
   203→          <div
   204→            key={index}
   205→            className="flex items-center gap-2 bg-gray-50 p-2 rounded-lg border border-gray-100"
   206→          >
   207→            <div className="flex items-center justify-center w-6 h-6 rounded-full bg-teal-100 text-teal-700 text-xs font-bold shrink-0">
   208→              {index + 1}
   209→            </div>
   210→            <Input
   211→              placeholder="اسم المحطة"
   212→              value={stop.name}
   213→              onChange={(e) => updateStop(index, 'name', e.target.value)}
   214→              className="flex-1 min-h-[44px] text-sm"
   215→            />
   216→            <Input
   217→              placeholder="الوقت"
   218→              value={stop.time}
   219→              onChange={(e) => updateStop(index, 'time', e.target.value)}
   220→              className="w-24 min-h-[44px] text-sm"
   221→              type="time"
   222→            />
   223→            <Button
   224→              type="button"
   225→              variant="ghost"
   226→              size="icon"
   227→              onClick={() => removeStop(index)}
   228→              className="text-red-500 hover:text-red-700 hover:bg-red-50 shrink-0 min-h-[44px] min-w-[44px]"
   229→            >
   230→              <X className="w-4 h-4" />
   231→            </Button>
   232→          </div>
   233→        ))}
   234→      </div>
   235→    </div>
   236→  )
   237→}

   240→export function BusRoutesManagement() {
   241→  const { selectedSchoolId } = useAdminStore()
   242→
   243→  // Data state
   244→  const [routes, setRoutes] = useState<BusRoute[]>([])
   245→  const [buses, setBuses] = useState<BusData[]>([])
   246→  const [loading, setLoading] = useState(true)
   247→  const [saving, setSaving] = useState(false)
   248→
   249→  // Tab state
   250→  const [activeTab, setActiveTab] = useState<'list' | 'add'>('list')
   251→
   252→  // Filter state
   253→  const [searchQuery, setSearchQuery] = useState('')
   254→  const [filterBusId, setFilterBusId] = useState('all')
   255→  const [filterArea, setFilterArea] = useState('')
   256→
   257→  // Edit dialog state
   258→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   259→  const [editingRoute, setEditingRoute] = useState<BusRoute | null>(null)
   260→  const [editForm, setEditForm] = useState<RouteForm>(defaultForm)
   261→
   262→  // Delete dialog state
   263→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   264→  const [deletingRoute, setDeletingRoute] = useState<BusRoute | null>(null)
   265→
   266→  // Add form state
   267→  const [addForm, setAddForm] = useState<RouteForm>(defaultForm)
   268→
   269→  // Expanded stops state
   270→  const [expandedStops, setExpandedStops] = useState<Record<string, boolean>>({})
   271→
   272→  // ===== Load routes =====
   273→  useEffect(() => {
   274→    let cancelled = false
   275→    async function loadRoutes() {
   276→      if (!selectedSchoolId) return
   277→      setLoading(true)
   278→      try {
   279→        const params = new URLSearchParams({ schoolId: selectedSchoolId })
   280→        if (filterBusId && filterBusId !== 'all') params.set('busId', filterBusId)
   281→        if (filterArea) params.set('area', filterArea)
   282→
   283→        const res = await fetch(`/api/bus-routes?${params.toString()}`)
   284→        if (res.ok) {
   285→          const data = await res.json()
   286→          if (!cancelled) setRoutes(data)
   287→        }
   288→      } catch (error) {
   289→        console.error('Error loading routes:', error)
   290→        if (!cancelled) toast.error('خطأ في تحميل خطوط السير')
   291→      } finally {
   292→        if (!cancelled) setLoading(false)
   293→      }
   294→    }
   295→    loadRoutes()
   296→    return () => { cancelled = true }
   297→  }, [selectedSchoolId, filterBusId, filterArea])
   298→
   299→  // ===== Load buses =====
   300→  useEffect(() => {
   301→    let cancelled = false
   302→    async function loadBuses() {
   303→      if (!selectedSchoolId) return
   304→      try {
   305→        const res = await fetch(`/api/buses?schoolId=${selectedSchoolId}&active=true`)
   306→        if (res.ok) {
   307→          const data = await res.json()
   308→          if (!cancelled) setBuses(data)
   309→        }
   310→      } catch (error) {
   311→        console.error('Error loading buses:', error)
   312→      }
   313→    }
   314→    loadBuses()
   315→    return () => { cancelled = true }
   316→  }, [selectedSchoolId])
   317→
   318→  // ===== Filter routes by search =====
   319→  const filteredRoutes = routes.filter((route) => {
   320→    if (!searchQuery) return true
   321→    const q = searchQuery.toLowerCase()
   322→    return (
   323→      route.name.toLowerCase().includes(q) ||
   324→      route.area.toLowerCase().includes(q) ||
   325→      (route.bus?.plateNumber?.toLowerCase().includes(q) ?? false) ||
   326→      (route.bus?.driverName?.toLowerCase().includes(q) ?? false)
   327→    )
   328→  })
   329→
   330→  // ===== Get unique areas for display =====
   331→  const uniqueAreas = Array.from(new Set(routes.map((r) => r.area))).sort()
   332→
   333→  // ===== Toggle stops expansion =====
   334→  const toggleStops = (routeId: string) => {
   335→    setExpandedStops((prev) => ({ ...prev, [routeId]: !prev[routeId] }))
   336→  }
   337→
   338→  // ===== Open edit dialog =====
   339→  const openEdit = (route: BusRoute) => {
   340→    setEditingRoute(route)
   341→    setEditForm({
   342→      name: route.name,
   343→      area: route.area,
   344→      busId: route.busId,
   345→      morningTime: route.morningTime || '',
   346→      afternoonTime: route.afternoonTime || '',
   347→      monthlyFee: String(route.monthlyFee),
   348→      stops: parseStops(route.stops),
   349→      active: route.active,
   350→    })
   351→    setEditDialogOpen(true)
   352→  }
   353→
   354→  // ===== Save edit =====
   355→  const saveEdit = async () => {
   356→    if (!editingRoute) return
   357→    if (!editForm.name.trim() || !editForm.area.trim() || !editForm.busId) {
   358→      toast.error('يرجى ملء جميع الحقول المطلوبة')
   359→      return
   360→    }
   361→    setSaving(true)
   362→    try {
   363→      const res = await fetch(`/api/bus-routes/${editingRoute.id}`, {
   364→        method: 'PUT',
   365→        headers: { 'Content-Type': 'application/json' },
   366→        body: JSON.stringify({
   367→          schoolId: selectedSchoolId,
   368→          ...editForm,
   369→          monthlyFee: parseFloat(editForm.monthlyFee) || 0,
   370→          stops: stringifyStops(editForm.stops),
   371→        }),
   372→      })
   373→      if (res.ok) {
   374→        toast.success('تم تحديث خط السير بنجاح')
   375→        setEditDialogOpen(false)
   376→        setEditingRoute(null)
   377→        // Refresh list
   378→        const params = new URLSearchParams({ schoolId: selectedSchoolId })
   379→        if (filterBusId && filterBusId !== 'all') params.set('busId', filterBusId)
   380→        if (filterArea) params.set('area', filterArea)
   381→        const refreshRes = await fetch(`/api/bus-routes?${params.toString()}`)
   382→        if (refreshRes.ok) {
   383→          const data = await refreshRes.json()
   384→          setRoutes(data)
   385→        }
   386→      } else {
   387→        const data = await res.json()
   388→        toast.error(data.error || 'خطأ في تحديث خط السير')
   389→      }
   390→    } catch {
   391→      toast.error('خطأ في الاتصال بالخادم')
   392→    } finally {
   393→      setSaving(false)
   394→    }
   395→  }
   396→
   397→  // ===== Delete route =====
   398→  const confirmDelete = async () => {
   399→    if (!deletingRoute) return
   400→    setSaving(true)
   401→    try {
   402→      const res = await fetch(
   403→        `/api/bus-routes/${deletingRoute.id}?schoolId=${selectedSchoolId}`,
   404→        { method: 'DELETE' }
   405→      )
   406→      if (res.ok) {
   407→        toast.success('تم حذف خط السير بنجاح')
   408→        setDeleteDialogOpen(false)
   409→        setDeletingRoute(null)
   410→        setRoutes((prev) => prev.filter((r) => r.id !== deletingRoute.id))
   411→      } else {
   412→        const data = await res.json()
   413→        toast.error(data.error || 'خطأ في حذف خط السير')
   414→      }
   415→    } catch {
   416→      toast.error('خطأ في الاتصال بالخادم')
   417→    } finally {
   418→      setSaving(false)
   419→    }
   420→  }
   421→
   422→  // ===== Add new route =====
   423→  const addRoute = async () => {
   424→    if (!addForm.name.trim() || !addForm.area.trim() || !addForm.busId) {
   425→      toast.error('يرجى ملء جميع الحقول المطلوبة')
   426→      return
   427→    }
   428→    setSaving(true)
   429→    try {
   430→      const res = await fetch('/api/bus-routes', {
   431→        method: 'POST',
   432→        headers: { 'Content-Type': 'application/json' },
   433→        body: JSON.stringify({
   434→          schoolId: selectedSchoolId,
   435→          ...addForm,
   436→          monthlyFee: parseFloat(addForm.monthlyFee) || 0,
   437→          stops: stringifyStops(addForm.stops),
   438→        }),
   439→      })
   440→      if (res.ok) {
   441→        toast.success('تم إضافة خط السير بنجاح')
   442→        setAddForm(defaultForm)
   443→        setActiveTab('list')
   444→        // Refresh list
   445→        const params = new URLSearchParams({ schoolId: selectedSchoolId })
   446→        const refreshRes = await fetch(`/api/bus-routes?${params.toString()}`)
   447→        if (refreshRes.ok) {
   448→          const data = await refreshRes.json()
   449→          setRoutes(data)
   450→        }
   451→      } else {
   452→        const data = await res.json()
   453→        toast.error(data.error || 'خطأ في إضافة خط السير')
   454→      }
   455→    } catch {
   456→      toast.error('خطأ في الاتصال بالخادم')
   457→    } finally {
   458→      setSaving(false)
   459→    }
   460→  }
   461→

   466→        method: 'PUT',
   467→        headers: { 'Content-Type': 'application/json' },

   470→          active: !route.active,

   476→          prev.map((r) => (r.id === route.id ? { ...r, active: !r.active } : r))

   478→      } else {
   479→        toast.error('خطأ في تحديث حالة خط السير')
   480→      }
   481→    } catch {
   482→      toast.error('خطأ في الاتصال بالخادم')
   483→    }
   484→  }
   485→
   486→  // ===== Render loading skeleton =====
   487→  if (loading && routes.length === 0) {
   488→    return (
   489→      <div className="space-y-4" dir="rtl">
   490→        <div className="flex items-center gap-3">
   491→          <Skeleton className="h-10 w-48" />
   492→          <Skeleton className="h-10 w-32" />
   493→        </div>
   494→        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   495→          {[1, 2, 3, 4].map((i) => (
   496→            <Skeleton key={i} className="h-64 rounded-xl" />
   497→          ))}
   498→        </div>
   499→      </div>
   500→    )
   501→  }
   502→
   503→  return (
   504→    <div className="space-y-4" dir="rtl">
   505→      {/* ===== Tabs ===== */}
   506→      <div className="flex items-center gap-2 bg-gray-100 p-1 rounded-lg w-fit">
   507→        <button
   508→          onClick={() => setActiveTab('list')}
   509→          className={`flex items-center gap-2 px-4 py-2.5 rounded-md text-sm font-medium transition-all min-h-[44px] ${
   510→            activeTab === 'list'
   511→              ? 'bg-white text-[#610000] shadow-sm'
   512→              : 'text-gray-500 hover:text-gray-700'
   513→          }`}
   514→        >
   515→          <Route className="w-4 h-4" />
   516→          خطوط السير

   519→          onClick={() => setActiveTab('add')}

   521→            activeTab === 'add'
   522→              ? 'bg-white text-[#610000] shadow-sm'
   523→              : 'text-gray-500 hover:text-gray-700'
   524→          }`}
   525→        >
   526→          <Plus className="w-4 h-4" />
   527→          إضافة خط
   528→        </button>
   529→      </div>
   530→
   531→      {/* ===== Tab 1: Routes List ===== */}
   532→      {activeTab === 'list' && (
   533→        <div className="space-y-4">
   534→          {/* Search & Filters */}
   535→          <Card className="border-0 shadow-sm">
   536→            <CardContent className="p-4">
   537→              <div className="flex flex-col sm:flex-row gap-3">
   538→                {/* Search */}
   539→                <div className="relative flex-1">
   540→                  <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   541→                  <Input
   542→                    placeholder="بحث بالاسم أو المنطقة..."
   543→                    value={searchQuery}
   544→                    onChange={(e) => setSearchQuery(e.target.value)}
   545→                    className="pr-9 min-h-[44px]"
   546→                  />
   547→                </div>
   548→
   549→                {/* Filter by bus */}
   550→                <Select
   551→                  value={filterBusId}
   552→                  onValueChange={setFilterBusId}
   553→                >
   554→                  <SelectTrigger className="w-full sm:w-[180px] min-h-[44px]">
   555→                    <Bus className="w-4 h-4 ml-1.5 text-gray-400" />
   556→                    <SelectValue placeholder="كل الباصات" />
   557→                  </SelectTrigger>
   558→                  <SelectContent>
   559→                    <SelectItem value="all">كل الباصات</SelectItem>

   561→                      <SelectItem key={bus.id} value={bus.id}>
   562→                        {bus.plateNumber} - {bus.driverName}
   563→                      </SelectItem>
   564→                    ))}
   565→                  </SelectContent>
   566→                </Select>
   567→
   568→                {/* Filter by area */}
   569→                <Select
   570→                  value={filterArea}
   571→                  onValueChange={setFilterArea}
   572→                >
   573→                  <SelectTrigger className="w-full sm:w-[160px] min-h-[44px]">
   574→                    <MapPin className="w-4 h-4 ml-1.5 text-gray-400" />
   575→                    <SelectValue placeholder="كل المناطق" />
   576→                  </SelectTrigger>
   577→                  <SelectContent>
   578→                    <SelectItem value="">كل المناطق</SelectItem>
   579→                    {uniqueAreas.map((area) => (
