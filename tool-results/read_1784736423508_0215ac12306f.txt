    61→/** خريطة نوع الملف → أيقونة + لون */
    62→function getFileTypeDisplay(fileName: string): { Icon: FileTypeIcon; color: string } {
    63→  switch (getFileTypeIcon(fileName)) {
    64→    case 'pdf':
    65→      return { Icon: FileText, color: '#dc2626' } // أحمر
    66→    case 'word':
    67→      return { Icon: FileText, color: '#2563eb' } // أزرق (لون Word)
    68→    case 'excel':
    69→      return { Icon: Sheet, color: '#16a34a' } // أخضر
    70→    case 'powerpoint':
    71→      return { Icon: Presentation, color: '#ea580c' } // برتقالي
    72→    case 'image':
    73→      return { Icon: ImageIcon, color: '#9333ea' } // بنفسجي
    74→    case 'archive':
    75→      return { Icon: Archive, color: '#475569' } // رمادي
    76→    case 'text':
    77→      return { Icon: FileText, color: '#475569' } // رمادي
    78→    default:
    79→      return { Icon: File, color: '#475569' } // رمادي
    80→  }
    81→}

    83→/** تحويل hex إلى rgba مع شفافية */
    84→function withAlpha(hex: string, alpha: number): string {
    85→  const h = hex.replace('#', '')
    86→  const r = parseInt(h.slice(0, 2), 16)
    87→  const g = parseInt(h.slice(2, 4), 16)
    88→  const b = parseInt(h.slice(4, 6), 16)
    89→  return `rgba(${r}, ${g}, ${b}, ${alpha})`
    90→}

    92→/** تنسيق التاريخ بالعربية */
    93→function formatDate(iso: string): string {
    94→  try {
    95→    return new Date(iso).toLocaleDateString('ar-EG', {
    96→      year: 'numeric',
    97→      month: 'long',
    98→      day: 'numeric',
    99→    })
   100→  } catch {
   101→    return iso.slice(0, 10)
   102→  }
   103→}
