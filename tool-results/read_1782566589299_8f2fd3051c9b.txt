    85→    85→function parseStops(stopsJson: string): Stop[] {
    86→    86→  try {
    87→    87→    const parsed = JSON.parse(stopsJson)
    88→    88→    if (Array.isArray(parsed)) {
    89→    89→      return parsed.map((s: Record<string, string>) => ({
    90→    90→        name: s.name || '',
    91→    91→        time: s.time || '',
    92→    92→      }))
    93→    93→    }
    94→    94→    return []
    95→    95→  } catch {
    96→    96→    return []
    97→    97→  }
    98→    98→}

   106→   106→function StopsTimeline({ stops }: { stops: Stop[] }) {

   114→   114→    <div className="flex flex-row-reverse items-start gap-0">
   115→   115→      {stops.map((stop, index) => (
   116→   116→        <div key={index} className="flex flex-col items-center flex-1 min-w-0">
   117→   117→          {/* Stop circle */}
   118→   118→          <div className="relative flex items-center w-full">
   119→   119→            {/* Line before (right side in RTL) */}
   120→   120→            {index < stops.length - 1 && (
   121→   121→              <div className="flex-1 h-0.5 bg-teal-300" />
   122→   122→            )}
   123→   123→            {/* Circle */}
   124→   124→            <div
   125→   125→              className={`w-6 h-6 rounded-full flex items-center justify-center shrink-0 border-2 ${
   126→   126→                index === 0
   127→   127→                  ? 'bg-teal-600 border-teal-600 text-white'
   128→   128→                  : index === stops.length - 1
   129→   129→                    ? 'bg-red-500 border-red-500 text-white'

   138→   138→            {index > 0 && (
   139→   139→              <div className="flex-1 h-0.5 bg-teal-300" />
   140→   140→            )}
   141→   141→          </div>

   157→   157→// ===== Stops Editor Component =====
   158→   158→function StopsEditor({
   159→   159→  stops,
   160→   160→  onChange,
   161→   161→}: {
   162→   162→  stops: Stop[]
   163→   163→  onChange: (stops: Stop[]) => void
   164→   164→}) {
   165→   165→  const addStop = () => {
   166→   166→    onChange([...stops, { name: '', time: '' }])
   167→   167→  }
   168→   168→
   169→   169→  const removeStop = (index: number) => {
   170→   170→    onChange(stops.filter((_, i) => i !== index))
   171→   171→  }
   172→   172→
   173→   173→  const updateStop = (index: number, field: keyof Stop, value: string) => {
   174→   174→    const updated = [...stops]
   175→   175→    updated[index] = { ...updated[index], [field]: value }
   176→   176→    onChange(updated)
   177→   177→  }
   178→   178→
   179→   179→  return (
   180→   180→    <div className="space-y-3">
   181→   181→      <div className="flex items-center justify-between">
   182→   182→        <Label className="text-sm font-semibold text-gray-700">المحطات</Label>
   183→   183→        <Button
   184→   184→          type="button"
   185→   185→          variant="outline"
   186→   186→          size="sm"
   187→   187→          onClick={addStop}
   188→   188→          className="min-h-[44px] text-teal-700 border-teal-300 hover:bg-teal-50"
   189→   189→        >
   190→   190→          <Plus className="w-4 h-4 ml-1" />
   191→   191→          إضافة محطة
   192→   192→        </Button>
   193→   193→      </div>
   194→   194→
   195→   195→      {stops.length === 0 && (
   196→   196→        <p className="text-sm text-gray-400 italic text-center py-3">
   197→   197→          لم يتم إضافة محطات بعد. اضغط &quot;إضافة محطة&quot; للبدء.
   198→   198→        </p>
   199→   199→      )}
   200→   200→
   201→   201→      <div className="space-y-2">
   202→   202→        {stops.map((stop, index) => (
   203→   203→          <div
   204→   204→            key={index}
   205→   205→            className="flex items-center gap-2 bg-gray-50 p-2 rounded-lg border border-gray-100"
   206→   206→          >
   207→   207→            <div className="flex items-center justify-center w-6 h-6 rounded-full bg-teal-100 text-teal-700 text-xs font-bold shrink-0">
   208→   208→              {index + 1}
   209→   209→            </div>
   210→   210→            <Input
   211→   211→              placeholder="اسم المحطة"
   212→   212→              value={stop.name}
   213→   213→              onChange={(e) => updateStop(index, 'name', e.target.value)}
   214→   214→              className="flex-1 min-h-[44px] text-sm"
   215→   215→            />
   216→   216→            <Input
   217→   217→              placeholder="الوقت"
   218→   218→              value={stop.time}
   219→   219→              onChange={(e) => updateStop(index, 'time', e.target.value)}
   220→   220→              className="w-24 min-h-[44px] text-sm"
   221→   221→              type="time"
   222→   222→            />
   223→   223→            <Button
   224→   224→              type="button"
   225→   225→              variant="ghost"
   226→   226→              size="icon"
   227→   227→              onClick={() => removeStop(index)}
   228→   228→              className="text-red-500 hover:text-red-700 hover:bg-red-50 shrink-0 min-h-[44px] min-w-[44px]"
   229→   229→            >
   230→   230→              <X className="w-4 h-4" />
   231→   231→            </Button>
   232→   232→          </div>
   233→   233→        ))}
   234→   234→      </div>
   235→   235→    </div>
   236→   236→  )
   237→   237→}
