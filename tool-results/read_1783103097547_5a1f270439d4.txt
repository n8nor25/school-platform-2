   222→function categoryBadgeClass(category: string): string {
   223→  const c = (category || '').trim();
   224→  const map: Record<string, string> = {
   225→    'تنبيه': 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300',
   226→    'أخبار': 'bg-violet-100 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300',
   227→    'إعلان': 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-300',
   228→    'فعالية': 'bg-fuchsia-100 text-fuchsia-700 dark:bg-fuchsia-900/30 dark:text-fuchsia-300',
   229→  };
   230→  return map[c] || 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300';
   231→}

   233→function formatDate(iso: string): string {
   234→  try {
   235→    const d = new Date(iso);
   236→    return d.toLocaleDateString('ar-EG', {
   237→      year: 'numeric',
   238→      month: 'long',
   239→      day: 'numeric',
   240→    });
   241→  } catch {
   242→    return '';
   243→  }

   246→export default function TeacherPortalPage({ onBack, schoolId }: TeacherPortalPageProps) {

   255→  const [submitting, setSubmitting] = useState(false);

   257→  // Session state
   258→  const [session, setSession] = useState<LoginResponse | null>(null);
   259→  const [dashboard, setDashboard] = useState<DashboardData | null>(null);
   260→  const [dashboardLoading, setDashboardLoading] = useState(false);
   261→
   262→  // Sub-page navigation
   263→  const [showExamsPage, setShowExamsPage] = useState(false);
   264→  const [showGradingPage, setShowGradingPage] = useState(false);
   265→
   266→  // ===== Restore session on mount =====
   267→  useEffect(() => {
   268→    let restored: LoginResponse | null = null;
   269→    try {
   270→      const raw = sessionStorage.getItem(SESSION_KEY);
   271→      if (raw) restored = JSON.parse(raw) as LoginResponse;
   272→    } catch {
   273→      restored = null;
   274→    }
   275→    // set-state-in-effect pattern
   276→    const t = setTimeout(() => {
   277→      if (restored) setSession(restored);
   278→      setFadeIn(true);
   279→    }, 0);
   280→    return () => clearTimeout(t);
   281→  }, []);
   282→
   283→  // ===== Fetch dashboard when session changes =====
   284→  useEffect(() => {
   285→    if (!session) {
   286→      const t = setTimeout(() => setDashboard(null), 0);
   287→      return () => clearTimeout(t);
   288→    }

   290→    const t1 = setTimeout(() => setDashboardLoading(true), 0);
   291→    async function run() {
   292→      try {
   293→        const res = await fetch(
   294→          `/api/teacher/dashboard?schoolId=${encodeURIComponent(
   295→            schoolId
   296→          )}&teacherId=${encodeURIComponent(session.teacherId)}`
   297→        );
   298→        const data = await res.json();
   299→        if (!res.ok) {
   300→          throw new Error(data?.error || 'فشل تحميل البيانات');
   301→        }
   302→        if (cancelled) return;
   303→        const t = setTimeout(() => setDashboard(data as DashboardData), 0);
   304→        void t;
   305→      } catch (err) {
   306→        const msg = err instanceof Error ? err.message : 'فشل تحميل البيانات';
   307→        toast.error(msg);
   308→      } finally {
   309→        if (cancelled) return;
   310→        const t = setTimeout(() => setDashboardLoading(false), 0);
   311→        void t;
   312→      }
   313→    }
   314→    run();
   315→    return () => {
   316→      cancelled = true;
   317→      clearTimeout(t1);
   318→    };
   319→  }, [session, schoolId]);
   320→
   321→  // Staggered card animation for quick actions
   322→  useEffect(() => {
   323→    if (!session) return;
   324→    const timeouts: ReturnType<typeof setTimeout>[] = [];
   325→    quickActions.forEach((_, index) => {
   326→      timeouts.push(
   327→        setTimeout(() => {
   328→          setVisibleCards((prev) =>
   329→            prev.includes(index) ? prev : [...prev, index]
   330→          );
   331→        }, 300 + index * 120)
   332→      );
   333→    });
   334→    return () => timeouts.forEach(clearTimeout);
   335→  }, [session]);
   336→
   337→  // ===== Handlers =====
   338→  const handleLogin = async (e: React.FormEvent) => {
   339→    e.preventDefault();
   340→    setLoginError(null);
   341→    if (!teacherCode.trim() || !phone.trim()) {
   342→      const t = setTimeout(() => setLoginError('يرجى إدخال كود المعلم ورقم الهاتف'), 0);
   343→      return () => clearTimeout(t);
   344→    }
   345→    setSubmitting(true);
   346→    try {
   347→      const res = await fetch('/api/teacher/login', {
   348→        method: 'POST',
   349→        headers: { 'Content-Type': 'application/json' },
   350→        body: JSON.stringify({ schoolId, teacherCode: teacherCode.trim(), phone: phone.trim() }),
   351→      });
   352→      const data = await res.json();
   353→      if (!res.ok) {
   354→        throw new Error(data?.error || 'فشل تسجيل الدخول');
   355→      }
   356→      const loginData = data as LoginResponse;
   357→      try {
   358→        sessionStorage.setItem(SESSION_KEY, JSON.stringify(loginData));
   359→      } catch {
   360→        // ignore quota errors
   361→      }
   362→      const t = setTimeout(() => {
   363→        setSession(loginData);
   364→        setVisibleCards([]);
   365→      }, 0);
   366→      void t;
   367→      if (loginData.testMode) {
   368→        toast.success('تم تسجيل الدخول في وضع التجربة');
   369→      } else {
   370→        toast.success(`مرحباً ${loginData.name}`);
   371→      }
   372→      if (loginData.warning) {
   373→        toast.info(loginData.warning);
   374→      }
   375→    } catch (err) {
   376→      const msg = err instanceof Error ? err.message : 'فشل تسجيل الدخول';
   377→      const t = setTimeout(() => setLoginError(msg), 0);
   378→      void t;
   379→    } finally {
   380→      const t = setTimeout(() => setSubmitting(false), 0);
   381→      void t;
   382→    }
   383→  };
   384→
   385→  const handleLogout = () => {
   386→    try {
   387→      sessionStorage.removeItem(SESSION_KEY);
   388→    } catch {
   389→      // ignore
   390→    }

   392→      setSession(null);
   393→      setDashboard(null);
   394→      setTeacherCode('');
   395→      setPhone('');
   396→      setVisibleCards([]);
   397→    }, 0);

   411→      return;
   412→    }
   413→    toast.info(`${title} — قريباً`, {
   414→      description: 'هذه الخدمة قيد التطوير وستتوفر في الإصدار القادم.',
   415→    });
   416→  };
   417→
   418→  // ===== Render: Login Form =====
   419→  if (!session) {
   420→    return (
   421→      <div
   422→        className="min-h-screen flex flex-col bg-gradient-to-br from-slate-50 via-violet-50/30 to-slate-100 dark:from-gray-900 dark:via-gray-800 dark:to-gray-900"
   423→        dir="rtl"
   424→      >
   425→        {/* Header */}
   426→        <header className="bg-gradient-to-l from-[#2A374E] to-[#3d4f6e] text-white shadow-xl sticky top-0 z-50">
   427→          <div className="container mx-auto px-4 py-4">
   428→            <div className="flex items-center justify-between">
   429→              <button
   430→                onClick={onBack}
   431→                className="flex items-center gap-2 text-white hover:text-violet-300 transition-colors"
   432→              >
   433→                <ArrowRight className="w-5 h-5" />
   434→                <span className="font-medium">العودة للرئيسية</span>
   435→              </button>
   436→              <div className="flex items-center gap-3">

   451→          <div

   454→            }`}
   455→          >
   456→            <Card className="border-0 shadow-2xl overflow-hidden">
   457→              <CardContent className="p-0">
   458→                {/* Gradient header */}

   517→                  <Button
