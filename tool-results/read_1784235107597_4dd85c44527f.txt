     1→'use client';
     2→
     3→import { useState, useEffect } from 'react';
     4→import {

     6→  XCircle, Trophy, TrendingUp, BarChart3, Star, RotateCcw,
     7→  ChevronDown, Loader2, Award, Users, Percent, Heart, Sparkles
     8→} from 'lucide-react';
     9→import { Button } from '@/components/ui/button';
    10→import { Card, CardContent } from '@/components/ui/card';
    11→import { Input } from '@/components/ui/input';
    12→import { Badge } from '@/components/ui/badge';
    13→import { Progress } from '@/components/ui/progress';
    14→import {

    16→  SelectContent,
    17→  SelectItem,
    18→  SelectTrigger,
    19→  SelectValue,
    20→} from '@/components/ui/select';
    21→
    22→interface ResultsPageProps {
    23→  onBack: () => void;
    24→  schoolId: string;
    25→}
    26→
    27→interface GradeOption {
    28→  id: string;
    29→  gradeName: string;
    30→  studentCount: number;
    31→  updatedAt: string;
    32→  term: string;
    33→  archived: boolean;
    34→}
    35→
    36→interface SubjectDetail {
    37→  subject: string;
    38→  subjectLabel?: string;
    39→  score: number;
    40→  max: number;
    41→  minPass: number;
    42→  passed: boolean;
    43→}
    44→
    45→interface PassStatus {
    46→  passed: boolean;
    47→  totalPass: boolean;
    48→  addedSubjectPass: boolean;
    49→  notAddedSubjectPass: boolean;
    50→  minPassingPercent: number;
    51→  studentAddedTotal: number;
    52→  maxAddedTotal: number;
    53→  subjectDetails: {
    54→    added: SubjectDetail[];
    55→    notAdded: SubjectDetail[];

    57→}
    58→
    59→interface StudentResult {
    60→  seatNumber: string;
    61→  studentName: string;
    62→  arabic: number;
    63→  english: number;
    64→  social: number;
    65→  math: number;
    66→  science: number;
    67→  total: number;
    68→  religion: number;
    69→  art: number;
    70→  computer: number;
    71→}
    72→
    73→interface MaxScores {
    74→  arabic: number;
    75→  english: number;
    76→  social: number;
    77→  math: number;
    78→  science: number;
    79→  religion: number;
    80→  art: number;
    81→  computer: number;
    82→  total: number;
    83→}
    84→
    85→interface QueryResult {
    86→  student: StudentResult;
    87→  maxScores: MaxScores;
    88→  passStatus: PassStatus;
    89→  gradeName: string;
    90→  term: string;
    91→}
    92→
    93→const subjectLabels: Record<string, string> = {
    94→  arabic: 'عربي',
    95→  english: 'انجليزي',
    96→  social: 'دراسات',
    97→  math: 'رياضيات',
    98→  science: 'علوم',
    99→  religion: 'دين',
   100→  art: 'فنية',
   101→  computer: 'كمبيوتر',
   102→};
   103→
   104→function getScoreColor(percentage: number): string {
   105→  if (percentage >= 85) return 'text-emerald-600 dark:text-emerald-400';
   106→  if (percentage >= 70) return 'text-blue-600 dark:text-blue-400';
   107→  if (percentage >= 50) return 'text-yellow-600 dark:text-yellow-400';
   108→  return 'text-red-600 dark:text-red-400';
   109→}
   110→
   111→function getScoreBg(percentage: number): string {
   112→  if (percentage >= 85) return 'bg-emerald-50 dark:bg-emerald-900/20 border-emerald-200 dark:border-emerald-800';
   113→  if (percentage >= 70) return 'bg-blue-50 dark:bg-blue-900/20 border-blue-200 dark:border-blue-800';
   114→  if (percentage >= 50) return 'bg-yellow-50 dark:bg-yellow-900/20 border-yellow-200 dark:border-yellow-800';
   115→  return 'bg-red-50 dark:bg-red-900/20 border-red-200 dark:border-red-800';
   116→}
   117→
   118→function getProgressBarColor(percentage: number): string {
   119→  if (percentage >= 85) return 'bg-emerald-500';
   120→  if (percentage >= 70) return 'bg-blue-500';
   121→  if (percentage >= 50) return 'bg-yellow-500';
   122→  return 'bg-red-500';
   123→}
   124→
   125→export default function ResultsPage({ onBack, schoolId }: ResultsPageProps) {
   126→  const [grades, setGrades] = useState<GradeOption[]>([]);
   127→  const [selectedGradeId, setSelectedGradeId] = useState<string>('');
   128→  const [selectedGrade, setSelectedGrade] = useState<string>('');
   129→  const [seatNumber, setSeatNumber] = useState<string>('');
   130→  const [loading, setLoading] = useState<boolean>(false);
   131→  const [gradesLoading, setGradesLoading] = useState<boolean>(true);
   132→  const [result, setResult] = useState<QueryResult | null>(null);
   133→  const [error, setError] = useState<string | null>(null);
   134→  const [searched, setSearched] = useState<boolean>(false);
   135→  const [fadeIn, setFadeIn] = useState<boolean>(false);
   136→
   137→  // Fetch available grades on mount
   138→  useEffect(() => {
   139→    const fetchGrades = async () => {
   140→      try {
   141→        setGradesLoading(true);
   142→        const res = await fetch(`/api/exam-results?archived=false&schoolId=${encodeURIComponent(schoolId)}`);
   143→        if (!res.ok) throw new Error('فشل في تحميل الصفوف');
   144→        const data = await res.json();
   145→        setGrades(data);
   146→      } catch {
   147→        setError('فشل في تحميل الصفوف الدراسية. حاول مرة أخرى.');
   148→      } finally {
   149→        setGradesLoading(false);
   150→      }
   151→    };
   152→    fetchGrades();
   153→  }, []);
   154→
   155→  // Fade-in animation on mount
   156→  useEffect(() => {
   157→    const timer = setTimeout(() => setFadeIn(true), 100);
   158→    return () => clearTimeout(timer);
   159→  }, []);
   160→
   161→  const handleSearch = async () => {
   162→    if (!selectedGradeId) {
   163→      setError('يرجى اختيار الصف الدراسي والترم');
   164→      return;
   165→    }
   166→    if (!seatNumber.trim()) {
   167→      setError('يرجى إدخال رقم الجلوس');
   168→      return;
   169→    }
   170→
   171→    setLoading(true);
   172→    setError(null);
   173→    setResult(null);
   174→    setSearched(true);
   175→
   176→    try {
   177→      const res = await fetch(
   178→        `/api/exam-results/query?gradeId=${encodeURIComponent(selectedGradeId)}&seat=${encodeURIComponent(seatNumber.trim())}`
   179→      );
   180→      const data = await res.json();
   181→
   182→      if (!res.ok) {
   183→        setError(data.error || 'فشل في البحث عن النتيجة');
   184→        return;
   185→      }
   186→
   187→      setResult(data);
   188→    } catch {
   189→      setError('فشل في الاتصال بالخادم. تأكد من اتصالك بالإنترنت.');
   190→    } finally {
   191→      setLoading(false);
   192→    }
   193→  };
   194→
   195→  const handleReset = () => {
   196→    setSelectedGradeId('');
   197→    setSelectedGrade('');
   198→    setSeatNumber('');
   199→    setResult(null);
   200→    setError(null);
   201→    setSearched(false);
   202→  };
   203→
   204→  const handleKeyPress = (e: React.KeyboardEvent) => {
   205→    if (e.key === 'Enter') {
   206→      handleSearch();
   207→    }
   208→  };
   209→
   210→  // Compute statistics from result
   211→  const computeStats = () => {
   212→    if (!result) return null;
   213→    const { student, maxScores, passStatus } = result;
   214→    const totalMax = maxScores.total || 1;
   215→    const totalPercent = Math.round((student.total / totalMax) * 100);
   216→
   217→    const addedSubjects = passStatus.subjectDetails.added;
   218→    const avgAdded = addedSubjects.length > 0
   219→      ? Math.round(addedSubjects.reduce((sum, s) => sum + (s.max > 0 ? (s.score / s.max) * 100 : 0), 0) / addedSubjects.length)
   220→      : 0;
   221→
   222→    const highestSubject = [...addedSubjects].sort((a, b) => {
   223→      const pA = a.max > 0 ? (a.score / a.max) * 100 : 0;
   224→      const pB = b.max > 0 ? (b.score / b.max) * 100 : 0;
   225→      return pB - pA;
   226→    })[0];
   227→
   228→    const addedTotalPercent = passStatus.maxAddedTotal > 0
   229→      ? Math.round((passStatus.studentAddedTotal / passStatus.maxAddedTotal) * 100)
   230→      : 0;
   231→
   232→    return {
   233→      totalPercent,
   234→      avgAdded,
   235→      addedTotalPercent,
   236→      highestSubject: highestSubject ? {
   237→        name: subjectLabels[highestSubject.subject] || highestSubject.subject,
   238→        percent: Math.round(highestSubject.max > 0 ? (highestSubject.score / highestSubject.max) * 100 : 0),
   239→      } : null,
   240→      passedSubjects: addedSubjects.filter(s => s.passed).length,
   241→      totalAddedSubjects: addedSubjects.length,
   242→    };
   243→  };
   244→
   245→  const stats = computeStats();
   246→
   247→  return (
   248→    <div className="min-h-screen flex flex-col bg-gradient-to-br from-slate-50 via-blue-50/30 to-slate-100 dark:from-gray-900 dark:via-gray-800 dark:to-gray-900" dir="rtl">
   249→      {/* Header */}
   250→      <header className="bg-[#2A374E] text-white shadow-xl sticky top-0 z-50">
   251→        <div className="container mx-auto px-4 py-4">
   252→          <div className="flex items-center justify-between">
   253→            <button
   254→              onClick={onBack}
   255→              className="flex items-center gap-2 text-white hover:text-blue-300 transition-colors"
   256→            >
   257→              <ArrowRight className="w-5 h-5" />
   258→              <span className="font-medium">العودة للرئيسية</span>
   259→            </button>
   260→            <div className="flex items-center gap-3">
   261→              <div className="w-10 h-10 bg-gradient-to-r from-red-500 to-red-600 rounded-full flex items-center justify-center shadow-lg">
   262→                <GraduationCap className="w-6 h-6 text-white" />
   263→              </div>
   264→              <div>
   265→                <h1 className="text-xl font-bold">نتائج الطلاب</h1>
   266→                <p className="text-blue-200 text-xs">استعلام عن نتائج الامتحانات</p>
   267→              </div>
   268→            </div>
   269→            <div className="w-28" />
   270→          </div>
   271→        </div>
   272→      </header>
   273→
   274→      <main className="flex-1 container mx-auto px-4 py-6 max-w-4xl">
   275→        {/* Search Section */}
   276→        <div
   277→          className={`mb-8 transition-all duration-700 ${fadeIn ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'}`}
   278→        >
   279→          <div className="relative">
   280→            {/* Gradient border wrapper */}
   281→            <div className="absolute -inset-[2px] bg-gradient-to-r from-[#2A374E] via-red-500 to-[#2A374E] rounded-2xl blur-[1px]" />
   282→            <Card className="relative border-0 shadow-2xl rounded-2xl overflow-hidden">
   283→              <CardContent className="p-0">
   284→                {/* Top accent bar */}
   285→                <div className="h-1.5 bg-gradient-to-r from-[#2A374E] via-red-500 to-[#2A374E]" />
   286→                
   287→                <div className="p-6 md:p-8">
   288→                  {/* Logo & Title */}
   289→                  <div className="text-center mb-8">
   290→                    <div className="w-20 h-20 mx-auto bg-gradient-to-br from-[#2A374E] to-[#3d4f6e] rounded-full flex items-center justify-center mb-4 shadow-lg ring-4 ring-white dark:ring-gray-700">
   291→                      <GraduationCap className="w-10 h-10 text-white" />
   292→                    </div>
   293→                    <h2 className="text-3xl font-bold text-[#2A374E] dark:text-white mb-2">
   294→                      نتائج الطلاب
   295→                    </h2>
   296→                    <p className="text-gray-500 dark:text-gray-400 text-lg">
   297→                      منصة لعرض نتائج الطلاب وتتبعها
   298→                    </p>
   299→                  </div>
   300→
   301→                  {/* Search Form */}
   302→                  <div className="space-y-5 max-w-lg mx-auto">
   303→                    {/* Grade Select */}
   304→                    <div>
   305→                      <label className="block text-sm font-semibold text-gray-700 dark:text-gray-300 mb-2">
   306→                        الصف الدراسي
   307→                      </label>
   308→                      <Select
   309→                        value={selectedGradeId}
   310→                        onValueChange={(val) => {
   311→                          setSelectedGradeId(val);
   312→                          const g = grades.find(gr => gr.id === val);
   313→                          setSelectedGrade(g?.gradeName || '');
   314→                          setError(null);
   315→                        }}
   316→                        dir="rtl"
   317→                      >
   318→                        <SelectTrigger className="w-full h-12 text-right bg-white dark:bg-gray-800 border-gray-200 dark:border-gray-600 text-base">
   319→                          <SelectValue placeholder={gradesLoading ? 'جارٍ التحميل...' : 'اختر الصف والترم'} />
   320→                        </SelectTrigger>
   321→                        <SelectContent>
   322→                          {grades.map((grade) => (
   323→                            <SelectItem key={grade.id} value={grade.id}>
   324→                              <div className="flex items-center gap-2">
   325→                                <GraduationCap className="w-4 h-4 text-[#2A374E] dark:text-blue-300" />
   326→                                <span className="font-medium">{grade.gradeName}</span>
   327→                                {grade.term && (
   328→                                  <Badge variant="outline" className="text-xs border-emerald-300 text-emerald-600 dark:border-emerald-500 dark:text-emerald-400">
   329→                                    {grade.term}
   330→                                  </Badge>
   331→                                )}
   332→                                <span className="text-xs text-gray-400">
   333→                                  ({grade.studentCount} طالب)
   334→                                </span>
   335→                              </div>
   336→                            </SelectItem>
   337→                          ))}
   338→                        </SelectContent>
   339→                      </Select>
   340→                    </div>
   341→
   342→                    {/* Seat Number Input */}
   343→                    <div>
   344→                      <label className="block text-sm font-semibold text-gray-700 dark:text-gray-300 mb-2">
   345→                        رقم الجلوس
   346→                      </label>
   347→                      <div className="relative">
   348→                        <Input
   349→                          type="text"
   350→                          inputMode="numeric"
   351→                          value={seatNumber}
   352→                          onChange={(e) => {
   353→                            const arabicMap: Record<string, string> = {
   354→                              '٠': '0', '١': '1', '٢': '2', '٣': '3', '٤': '4',
   355→                              '٥': '5', '٦': '6', '٧': '7', '٨': '8', '٩': '9',
   356→                            }
   357→                            const val = e.target.value.replace(/[٠-٩]/g, (c) => arabicMap[c] || c).replace(/[^0-9]/g, '')
   358→                            setSeatNumber(val);
   359→                            setError(null);
   360→                          }}
   361→                          onKeyDown={handleKeyPress}
   362→                          placeholder="أدخل رقم الجلوس بالأرقام"
   363→                          className="h-12 text-right text-lg pr-4 pl-12 bg-white dark:bg-gray-800 border-gray-200 dark:border-gray-600"
   364→                          disabled={loading}
   365→                        />
   366→                        <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-gray-400" />
   367→                      </div>
   368→                    </div>
   369→
   370→                    {/* Error Message */}
   371→                    {error && (
   372→                      <div className="bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 rounded-xl p-4 flex items-start gap-3">
   373→                        <AlertCircle className="w-5 h-5 text-red-500 shrink-0 mt-0.5" />
   374→                        <div className="flex-1">
   375→                          <p className="text-red-700 dark:text-red-400 text-sm font-medium">{error}</p>
   376→                        </div>
   377→                        <button
   378→                          onClick={() => setError(null)}
   379→                          className="text-red-400 hover:text-red-600 transition-colors"
   380→                        >
   381→                          <XCircle className="w-4 h-4" />
   382→                        </button>
   383→                      </div>
   384→                    )}
   385→
   386→                    {/* Search Button */}
   387→                    <div className="flex gap-3">
   388→                      <Button
   389→                        onClick={handleSearch}
   390→                        disabled={loading || !selectedGrade || !seatNumber.trim()}
   391→                        className="flex-1 h-12 text-lg font-bold bg-gradient-to-r from-[#2A374E] to-[#3d4f6e] hover:from-[#1e2a3e] hover:to-[#2A374E] text-white shadow-lg transition-all duration-300 hover:shadow-xl disabled:opacity-50"
   392→                      >
   393→                        {loading ? (
   394→                          <div className="flex items-center gap-2">
   395→                            <Loader2 className="w-5 h-5 animate-spin" />
   396→                            <span>جارٍ البحث...</span>
   397→                          </div>
   398→                        ) : (
   399→                          <div className="flex items-center gap-2">
   400→                            <Search className="w-5 h-5" />
   401→                            <span>بحث</span>
   402→                          </div>
   403→                        )}
   404→                      </Button>
   405→                      {(searched || result) && (
   406→                        <Button
   407→                          onClick={handleReset}
   408→                          variant="outline"
   409→                          className="h-12 px-4 border-gray-300 dark:border-gray-600 hover:bg-gray-100 dark:hover:bg-gray-700 transition-all"
   410→                        >
   411→                          <RotateCcw className="w-5 h-5" />
   412→                        </Button>
   413→                      )}
   414→                    </div>
   415→                  </div>
   416→                </div>
   417→              </CardContent>
   418→            </Card>
   419→          </div>
   420→        </div>
   421→
   422→        {/* Loading State */}
   423→        {loading && (
   424→          <div className="text-center py-16">
   425→            <div className="w-20 h-20 mx-auto mb-6 relative">
   426→              <div className="absolute inset-0 border-4 border-gray-200 dark:border-gray-700 rounded-full" />
   427→              <div className="absolute inset-0 border-4 border-[#2A374E] border-t-transparent rounded-full animate-spin" />
   428→            </div>
   429→            <p className="text-gray-500 dark:text-gray-400 text-lg">جارٍ البحث عن النتيجة...</p>
   430→          </div>
   431→        )}
   432→
   433→        {/* No Results State */}
   434→        {searched && !loading && !result && !error && (
   435→          <div className="text-center py-16">
   436→            <div className="w-24 h-24 mx-auto bg-gray-100 dark:bg-gray-800 rounded-full flex items-center justify-center mb-6">
   437→              <Search className="w-12 h-12 text-gray-400" />
   438→            </div>
   439→            <h3 className="text-xl font-bold text-gray-700 dark:text-gray-200 mb-2">
   440→              لم يتم العثور على نتيجة
   441→            </h3>
   442→            <p className="text-gray-500 dark:text-gray-400 mb-6">
   443→              تأكد من رقم الجلوس والصف الدراسي وحاول مرة أخرى
   444→            </p>
   445→            <Button
   446→              onClick={handleReset}
   447→              variant="outline"
   448→              className="border-[#2A374E] text-[#2A374E] dark:border-gray-500 dark:text-gray-300"
   449→            >
   450→              <RotateCcw className="w-4 h-4 ml-2" />
   451→              محاولة جديدة
   452→            </Button>
   453→          </div>
   454→        )}
   455→
   456→        {/* Results Display */}
   457→        {result && !loading && (
   458→          <div className="space-y-6">
   459→            {/* Student Info Card */}
   460→            <Card className="border-0 shadow-xl overflow-hidden">
   461→              <CardContent className="p-0">
   462→                {/* Card header with gradient */}
   463→                <div className="bg-gradient-to-r from-[#2A374E] to-[#3d4f6e] p-6 text-white">
   464→                  <div className="flex items-center justify-between flex-wrap gap-4">
   465→                    <div>
   466→                      <h3 className="text-2xl md:text-3xl font-bold mb-1">
   467→                        {result.student.studentName}
   468→                      </h3>
   469→                      <div className="flex flex-wrap items-center gap-3 text-blue-200">
   470→                        <span className="flex items-center gap-1">
   471→                          <GraduationCap className="w-4 h-4" />
   472→                          {result.gradeName}
   473→                        </span>
   474→                        {result.term && (
   475→                          <>
   476→                            <span className="text-blue-300">|</span>
   477→                            <span className="flex items-center gap-1 text-amber-200">
   478→                              {result.term}
   479→                            </span>
   480→                          </>
   481→                        )}
   482→                        <span className="text-blue-300">|</span>
   483→                        <span>رقم الجلوس: {result.student.seatNumber}</span>
   484→                      </div>
   485→                    </div>
   486→                    {/* Pass/Fail Badge */}
   487→                    <div
   488→                      className={`px-6 py-3 rounded-xl font-bold text-lg shadow-lg transition-all duration-500 ${
   489→                        result.passStatus.passed
   490→                          ? 'bg-emerald-500 text-white'
   491→                          : 'bg-red-500 text-white'
   492→                      }`}
   493→                    >
   494→                      <div className="flex items-center gap-2">
   495→                        {result.passStatus.passed ? (
   496→                          <CheckCircle2 className="w-6 h-6" />
   497→                        ) : (
   498→                          <XCircle className="w-6 h-6" />
   499→                        )}
   500→                        <span>{result.passStatus.passed ? 'ناجح' : 'راسب'}</span>
   501→                      </div>
   502→                    </div>
   503→                  </div>
   504→                </div>
   505→
   506→                {/* Total Score Section */}
   507→                <div className="p-6 bg-white dark:bg-gray-800">
   508→                  <div className="flex items-center justify-between mb-3">
   509→                    <span className="text-lg font-bold text-[#2A374E] dark:text-white">
   510→                      المجموع الكلي
   511→                    </span>
   512→                    <div className="text-left">
   513→                      <span className="text-3xl font-bold text-[#2A374E] dark:text-white">
   514→                        {result.student.total}
   515→                      </span>
   516→                      <span className="text-gray-400 text-lg mr-1">
   517→                        / {result.maxScores.total}
   518→                      </span>
   519→                    </div>
   520→                  </div>
   521→                  <div className="relative">
   522→                    <div className="w-full h-4 bg-gray-100 dark:bg-gray-700 rounded-full overflow-hidden">
