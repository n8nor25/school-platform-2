    78→// ===== Helper: Format current date in Arabic =====
    79→function formatArabicDate(): string {
    80→  const now = new Date()
    81→  return now.toLocaleDateString('ar-SA', {
    82→    year: 'numeric',
    83→    month: 'long',
    84→    day: 'numeric',
    85→  })
    86→}
    87→
    88→// ===== Helper: Direction badge color =====
    89→function getDirectionBadge(direction: string) {
    90→  switch (direction) {
    91→    case 'ذهاب':
    92→      return 'bg-emerald-100 text-emerald-800 border-emerald-200'
    93→    case 'عودة':
    94→      return 'bg-amber-100 text-amber-800 border-amber-200'
    95→    case 'ذهاب وعودة':
    96→      return 'bg-sky-100 text-sky-800 border-sky-200'
    97→    default:
    98→      return 'bg-gray-100 text-gray-800 border-gray-200'
    99→  }
   100→}
   101→
   102→// ===== Helper: Occupancy bar color =====
   103→function getOccupancyColor(rate: number): string {
   104→  if (rate >= 90) return 'bg-red-500'
   105→  if (rate >= 70) return 'bg-amber-500'
   106→  if (rate >= 40) return 'bg-emerald-500'
   107→  return 'bg-sky-500'
   108→}
