    97→    97→  // Tab state

   123→   123→          setBuses([])

   191→   191→        driverPhone: editForm.driverPhone.trim() || null,

   213→   213→      } else {
   214→   214→        toast.error('فشل في تحديث البيانات')
   215→   215→      }
   216→   216→    } catch {
   217→   217→      toast.error('فشل في تحديث البيانات')
   218→   218→    } finally {
   219→   219→      setSaving(false)
   220→   220→    }
   221→   221→  }
   222→   222→
   223→   223→  // Handle add bus
   224→   224→  const handleAddBus = async () => {

   226→   226→      toast.error('يرجى إدخال رقم اللوحة واسم السائق')

   232→   232→    }
   233→   233→    setSaving(true)
   234→   234→    try {
   235→   235→      const body = {
   236→   236→        schoolId: selectedSchoolId,
   237→   237→        plateNumber: addForm.plateNumber.trim(),
   238→   238→        driverName: addForm.driverName.trim(),
   239→   239→        driverPhone: addForm.driverPhone.trim() || null,

   248→   248→      }

   251→   251→        method: 'POST',

   264→   264→      }

   266→   266→      toast.error('فشل في إضافة الباص')

   280→   280→        toast.error('فشل في تعطيل الباص')

   285→   285→      setDeleteDialogOpen(false)

   294→   294→      const params = new URLSearchParams({ schoolId: selectedSchoolId })
   295→   295→      if (activeFilter === 'active') params.set('active', 'true')
   296→   296→      else if (activeFilter === 'inactive') params.set('active', 'false')
   297→   297→
   298→   298→      const res = await fetch(`/api/buses?${params.toString()}`)
   299→   299→      if (res.ok) {
   300→   300→        const data = await res.json()
   301→   301→        setBuses(Array.isArray(data) ? data : [])
   302→   302→      }

   304→   304→      toast.error('فشل في تحديث البيانات')
   305→   305→    }
   306→   306→  }
   307→   307→

   313→   313→
   314→   314→  return (
   315→   315→    <div className="space-y-6" dir="rtl">
   316→   316→      {/* Header */}

   319→   319→          <Bus className="w-5 h-5 text-[#610000]" />
   320→   320→          إدارة أسطول الباصات

   324→   324→      {/* Tabs */}

   329→   329→            الأسطول

   376→   376→                  onClick={() => setActiveTab('add')}
   377→   377→                >
   378→   378→                  <Plus className="w-4 h-4 ml-1" />
   379→   379→                  إضافة باص جديد
   380→   380→                </Button>
   381→   381→              </CardContent>
   382→   382→            </Card>

   384→   384→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   385→   385→              {filteredBuses.map((bus) => (

   389→   389→                  onEdit={() => openEdit(bus)}
   390→   390→                  onDelete={() => {

   395→   395→                />

   404→   404→            {/* Bus Info Section */}
