   282→   282→    try {
   283→   283→      const raw = sessionStorage.getItem(SESSION_KEY);

   285→   285→    } catch {
   286→   286→      restored = null;
   287→   287→    }

   290→   290→      if (restored) setSession(restored);

   298→   298→    if (!session) {
   299→   299→      const t = setTimeout(() => setDashboard(null), 0);
   300→   300→      return () => clearTimeout(t);
   301→   301→    }

   306→   306→        const res = await fetch(
   307→   307→          `/api/teacher/dashboard?schoolId=${encodeURIComponent(
   308→   308→            schoolId

   313→   313→          throw new Error(data?.error || 'فشل تحميل البيانات');
   314→   314→        }
   315→   315→        if (cancelled) return;
   316→   316→        const t = setTimeout(() => setDashboard(data as DashboardData), 0);
   317→   317→        void t;
   318→   318→      } catch (err) {
   319→   319→        const msg = err instanceof Error ? err.message : 'فشل تحميل البيانات';
   320→   320→        toast.error(msg);
   321→   321→      } finally {
   322→   322→        if (cancelled) return;
   323→   323→        const t = setTimeout(() => setDashboardLoading(false), 0);
   324→   324→        void t;
   325→   325→      }
   326→   326→    }

   328→   328→    return () => {
   329→   329→      cancelled = true;
   330→   330→      clearTimeout(t1);

   375→   375→      const t = setTimeout(() => {

   387→   387→      }
   388→   388→    } catch (err) {
   389→   389→      const msg = err instanceof Error ? err.message : 'فشل تسجيل الدخول';
   390→   390→      const t = setTimeout(() => setLoginError(msg), 0);
   391→   391→      void t;
   392→   392→    } finally {
   393→   393→      const t = setTimeout(() => setSubmitting(false), 0);
   394→   394→      void t;
   395→   395→    }
   396→   396→  };

   400→   400→      sessionStorage.removeItem(SESSION_KEY);
   401→   401→    } catch {
   402→   402→      // ignore
   403→   403→    }
   404→   404→    const t = setTimeout(() => {
   405→   405→      setSession(null);
   406→   406→      setDashboard(null);
   407→   407→      setTeacherCode('');
   408→   408→      setPhone('');
   409→   409→      setVisibleCards([]);
   410→   410→    }, 0);
   411→   411→    void t;
   412→   412→    toast.info('تم تسجيل الخروج من بوابة المعلم');

   435→   435→
   436→   436→  // ===== Render: Login Form =====
   437→   437→  if (!session) {

   441→   441→        dir="rtl"
   442→   442→      >
   443→   443→        {/* Header */}
   444→   444→        <header className="bg-gradient-to-l from-[#2A374E] to-[#3d4f6e] text-white shadow-xl sticky top-0 z-50">
   445→   445→          <div className="container mx-auto px-4 py-4">
