   200→   200→function buildUrl(base: string, student: StudentInfo | null, extra?: Record<string, string>): string {
   201→   201→  const params = new URLSearchParams();
   202→   202→  if (student) {
   203→   203→    params.set('schoolId', student.schoolId);
   204→   204→    params.set('studentId', student.studentId);
   205→   205→  }
   206→   206→  if (extra) {
   207→   207→    for (const [k, v] of Object.entries(extra)) params.set(k, v);
   208→   208→  }
   209→   209→  const qs = params.toString();
   210→   210→  return qs ? `${base}?${qs}` : base;
   211→   211→}

   213→   213→function formatTime(seconds: number): string {
   214→   214→  if (seconds < 0) seconds = 0;
   215→   215→  const h = Math.floor(seconds / 3600);
   216→   216→  const m = Math.floor((seconds % 3600) / 60);
   217→   217→  const s = seconds % 60;
   218→   218→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   219→   219→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   220→   220→}
   221→   221→
   222→   222→function formatDate(iso: string): string {
   223→   223→  try {
   224→   224→    const d = new Date(iso);
   225→   225→    return d.toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   226→   226→  } catch {
   227→   227→    return iso;
   228→   228→  }

   399→   399→function StudentLogin({
   400→   400→  onLogin,
   401→   401→  schoolId,
   402→   402→}: {
   403→   403→  onLogin: (info: StudentInfo) => void;
   404→   404→  schoolId?: string;
   405→   405→}) {
   406→   406→  const [studentId, setStudentId] = useState('');
   407→   407→  const [studentName, setStudentName] = useState('');
   408→   408→  const [schoolIdInput, setSchoolIdInput] = useState(schoolId || '');
   409→   409→  const [loading, setLoading] = useState(false);
   410→   410→  const [error, setError] = useState<string | null>(null);
   411→   411→
   412→   412→  // مزامنة schoolIdInput مع الـ prop عندما يتوفر (selectedSchoolId يُحمّل من store
   413→   413→  // بعد أول render). نُلّف setState بـ setTimeout لتجنّب تحذير set-state-in-effect.
   414→   414→  useEffect(() => {
   415→   415→    if (schoolId && schoolId !== schoolIdInput) {
   416→   416→      const id = schoolId;
   417→   417→      const t = setTimeout(() => setSchoolIdInput(id), 0);
   418→   418→      return () => clearTimeout(t);
   419→   419→    }
   420→   420→  }, [schoolId, schoolIdInput]);
   421→   421→
   422→   422→  const handleSubmit = async (e: React.FormEvent) => {
   423→   423→    e.preventDefault();
   424→   424→    setError(null);
   425→   425→    if (!studentId.trim()) {
   426→   426→      setError('معرّف الطالب مطلوب');
   427→   427→      return;
   428→   428→    }
   429→   429→    if (!studentName.trim()) {
   430→   430→      setError('اسم الطالب مطلوب');
   431→   431→      return;
   432→   432→    }
   433→   433→    if (!schoolIdInput.trim()) {
   434→   434→      setError('معرّف المدرسة مطلوب');
   435→   435→      return;
   436→   436→    }
   437→   437→    setLoading(true);
   438→   438→    // تحقق سريع من توفر امتحانات (لا يلزم — فقط للتأكد من المدرسة)
   439→   439→    try {
   440→   440→      const res = await fetch(
   441→   441→        `/api/exams/available?schoolId=${encodeURIComponent(schoolIdInput)}&studentId=${encodeURIComponent(studentId)}`,
   442→   442→        { headers: { 'x-student-id': studentId } }
   443→   443→      );
   444→   444→      const data = await res.json();
   445→   445→      if (!res.ok) {
   446→   446→        setError(data.error || 'فشل التحقق من البيانات');
   447→   447→        setLoading(false);
   448→   448→        return;
   449→   449→      }
   450→   450→      onLogin({
   451→   451→        studentId: studentId.trim(),
   452→   452→        studentName: studentName.trim(),
   453→   453→        schoolId: schoolIdInput.trim(),
   454→   454→      });
   455→   455→    } catch (e) {
   456→   456→      // نسمح بالدخول حتى لو فشل الفحص (قد لا توجد امتحانات متاحة)
   457→   457→      onLogin({
   458→   458→        studentId: studentId.trim(),
   459→   459→        studentName: studentName.trim(),
   460→   460→        schoolId: schoolIdInput.trim(),
   461→   461→      });
   462→   462→    } finally {
   463→   463→      setLoading(false);
   464→   464→    }
   465→   465→  };
   466→   466→
   467→   467→  return (
   468→   468→    <div className="max-w-md mx-auto pt-8">
