   103→function HomePage() {
   104→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated, setHasHydrated } = useAdminStore()
   105→  const searchParams = useSearchParams()
   106→  const router = useRouter()
   107→  const pathname = usePathname()
   108→  const urlSubdomain = searchParams.get('school')
   109→  const urlAction = searchParams.get('action')
   110→  const hasAppliedUrlSchool = useRef(false)
   111→
   112→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   113→  const [news, setNews] = useState<NewsItem[]>([])
   114→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   115→  const [teachers, setTeachers] = useState<Teacher[]>([])
   116→  const [sliders, setSliders] = useState<SliderItem[]>([])
   117→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   118→  const [loading, setLoading] = useState(true)
   119→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
   120→  const [servicesOpen, setServicesOpen] = useState(false)
   121→  const [showAdminLogin, setShowAdminLogin] = useState(false)
   122→  const [showResultsPage, setShowResultsPage] = useState(false)
   123→  const [showStudentLife, setShowStudentLife] = useState(false)
   124→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   125→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   126→  const [showParentsPage, setShowParentsPage] = useState(false)
   127→  const [showExamsPage, setShowExamsPage] = useState(false)
   128→  const [showAnalyticsPage, setShowAnalyticsPage] = useState(false)
   129→  const [showScrollTop, setShowScrollTop] = useState(false)
   130→  const [currentSlide, setCurrentSlide] = useState(0)
   131→  const logoClickCount = useRef(0)
   132→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   133→
   134→  const hydrated = _hasHydrated
   135→
   136→  // Fallback: ensure hydration flag is set on the client (Zustand v5
   137→  // onRehydrateStorage may not fire in all environments).
   138→  useEffect(() => {
   139→    if (!_hasHydrated) setHasHydrated(true)
   140→  }, [_hasHydrated, setHasHydrated])
   141→
   142→  // Resolve school from subdomain only (not from store's school list)
   143→  useEffect(() => {
   144→    if (!hydrated) return
   145→    let cancelled = false
   146→    async function resolveSchool() {
   147→      try {
   148→        if (urlSubdomain) {
   149→          // If subdomain is in URL, fetch by subdomain
   150→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   151→          if (res.ok && !cancelled) {
   152→            const data = await res.json()
   153→            if (data.school) {
   154→              setSchoolData(data)
   155→              setSelectedSchoolId(data.school.id)
   156→              return
   157→            }
   158→          }
   159→        }
   160→        // No subdomain in URL — fetch first available school
   161→        const schoolsRes = await fetch('/api/schools')
   162→        if (schoolsRes.ok && !cancelled) {
   163→          const schoolsList = await schoolsRes.json()
   164→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   165→            setSchools(schoolsList)
   166→            const firstSchool = schoolsList[0]
   167→            // Fetch full data for the first school
   168→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   169→            if (res.ok && !cancelled) {
   170→              const data = await res.json()
   171→              if (data.school) {
   172→                setSchoolData(data)
   173→                setSelectedSchoolId(data.school.id)
   174→              }
   175→            }
   176→          }
   177→        }
   178→      } catch {
   179→        // fallback to default data
   180→      }
   181→    }
   182→    resolveSchool()
   183→    return () => { cancelled = true }
   184→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   185→
   186→  // Fetch content for selected school
   187→  useEffect(() => {
   188→    if (!hydrated || !selectedSchoolId) return
   189→    let cancelled = false
   190→    const sid = selectedSchoolId
   191→    async function loadData() {
   192→      setLoading(true)
   193→      try {
   194→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   195→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   196→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   197→          fetch(`/api/teachers?schoolId=${sid}`),
   198→          fetch(`/api/sliders?schoolId=${sid}`),
   199→          fetch(`/api/custom-sections?schoolId=${sid}`),
   200→        ])
   201→        if (cancelled) return
   202→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   203→          const d = await newsRes.value.json()
   204→          if (Array.isArray(d) && !cancelled) setNews(d)
   205→        }
   206→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   207→          const d = await galleryRes.value.json()
   208→          if (Array.isArray(d) && !cancelled) setGallery(d)
   209→        }
   210→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   211→          const d = await teachersRes.value.json()
   212→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   213→        }
   214→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   215→          const d = await slidersRes.value.json()
   216→          if (Array.isArray(d) && !cancelled) setSliders(d)
   217→        }
   218→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   219→          const d = await csRes.value.json()
   220→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   221→        }
   222→      } catch {
   223→        // silently fail
   224→      } finally {
   225→        if (!cancelled) setLoading(false)
   226→      }
   227→    }
   228→    loadData()
   229→    return () => { cancelled = true }
   230→  }, [selectedSchoolId, hydrated])
   231→
   232→  // Auto-rotate slider
   233→  const activeSliders = sliders.filter(s => s.active)
   234→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   235→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   236→  useEffect(() => {
   237→    if (slideItems.length <= 1) return
   238→    const timer = setInterval(() => setCurrentSlide(i => (i + 1) % slideItems.length), 5000)
   239→    return () => clearInterval(timer)
   240→  }, [slideItems.length])
   241→
   242→  // Scroll to top
   243→  useEffect(() => {
   244→    const h = () => setShowScrollTop(window.scrollY > 500)
   245→    window.addEventListener('scroll', h)
   246→    return () => window.removeEventListener('scroll', h)
   247→  }, [])
   248→
   249→  const handleLogoClick = useCallback(() => {
   250→    logoClickCount.current += 1
   251→    if (logoClickTimer.current) clearTimeout(logoClickTimer.current)
   252→    logoClickTimer.current = setTimeout(() => { logoClickCount.current = 0 }, 3000)
   253→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   254→  }, [])
   255→
   256→  // معالجة اختصارات PWA (?action=...) عند فتح التطبيق من الشاشة الرئيسية
   257→  const urlActionProcessed = useRef(false)
   258→  useEffect(() => {
   259→    if (!urlAction || !hydrated || urlActionProcessed.current) return
   260→    urlActionProcessed.current = true
   261→    // تنظيف الـ URL من معامل action بعد تفعيله
   262→    router.replace('/')
   263→    // استخدام microtask لتجنّب تحذير set-state-in-effect
   264→    queueMicrotask(() => {
   265→      if (urlAction === 'exams') { setShowExamsPage(true) }
   266→      else if (urlAction === 'results') setShowResultsPage(true)
   267→      else if (urlAction === 'schedules') setShowSchedulesPage(true)
   268→      else if (urlAction === 'library') setShowLibraryPage(true)
   269→      else if (urlAction === 'parents') setShowParentsPage(true)
   270→    })
   271→  }, [urlAction, hydrated, router])
   272→
   273→  const handleServiceAction = (action: string) => {
   274→    setServicesOpen(false); setMobileMenuOpen(false)
   275→    if (action === 'results') setShowResultsPage(true)
   276→    if (action === 'schedules') setShowSchedulesPage(true)
   277→    if (action === 'library') setShowLibraryPage(true)
   278→    if (action === 'parents') setShowParentsPage(true)
   279→    if (action === 'exams') setShowExamsPage(true)
   280→  }
   281→
   282→  // الانتقال من بوابة أولياء الأمور إلى صفحة الامتحانات الإلكترونية
   283→  const handleOpenExamsFromParents = useCallback(() => {
   284→    setShowParentsPage(false)
   285→    setShowExamsPage(true)
   286→  }, [])
   287→
   288→  // الانتقال من بوابة أولياء الأمور إلى صفحة التحليلات المقارنة
   289→  const handleOpenAnalyticsFromParents = useCallback(() => {
   290→    setShowParentsPage(false)
   291→    setShowAnalyticsPage(true)
   292→  }, [])
   293→
   294→  const school = schoolData?.school || defaultSchoolData.school
   295→  const settings = schoolData?.settings || defaultSchoolData.settings
   296→  const stats = schoolData?.stats || defaultSchoolData.stats
   297→  const alertNews = news.filter(n => n.category === 'تنبيه')
   298→
   299→  // Build ticker text for seamless scrolling
   300→  const tickerText = news.map(item => item.title).join('     ◆     ')
   301→
   302→  // Sub-page rendering
   303→  if (isAdminMode) return <AdminLayout />
   304→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   305→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   306→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   307→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />
   308→  if (showParentsPage) return <ParentsPortalPage onBack={() => setShowParentsPage(false)} schoolId={selectedSchoolId} onOpenExams={handleOpenExamsFromParents} onOpenAnalytics={handleOpenAnalyticsFromParents} />
   309→  if (showExamsPage) return <StudentExamsPage onBack={() => setShowExamsPage(false)} schoolId={selectedSchoolId} />
   310→  if (showAnalyticsPage) return <ParentAnalyticsPage onBack={() => setShowAnalyticsPage(false)} schoolId={selectedSchoolId} />
   311→
   312→  if (!hydrated) {
   313→    return (
   314→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   315→        <div className="text-center">
   316→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />
   317→          <p className="text-gray-500 text-sm">جاري التحميل...</p>
   318→        </div>
   319→      </div>
   320→    )
   321→  }
   322→
   323→  return (
   324→    <div className="min-h-screen flex flex-col" dir="rtl">
   325→
   326→      {/* ===== STICKY TOP ===== */}
   327→      <div className="sticky top-0 z-50 shadow-lg">
   328→        {/* Top Bar - Contact info only, NO school switcher for visitors */}
   329→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">
   330→          <div className="max-w-[1280px] mx-auto px-4 py-1.5 flex items-center justify-between">
   331→            <div className="flex items-center gap-4">
   332→              {school.phone && <a href={`tel:${school.phone}`} className="hover:opacity-80 flex items-center gap-1">📞 {school.phone}</a>}
   333→              {school.email && <a href={`mailto:${school.email}`} className="hover:opacity-80 flex items-center gap-1">✉️ {school.email}</a>}
   334→            </div>
   335→            <div className="flex items-center gap-2">
   336→              {school.facebookUrl && (
   337→                <a href={school.facebookUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📘</a>
   338→              )}
   339→              {settings?.youtubeUrl && (
   340→                <a href={settings.youtubeUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📺</a>
   341→              )}
   342→            </div>
   343→          </div>
   344→        </div>
   345→
   346→        {/* Header */}
   347→        <header className="bg-white shadow-sm">
   348→          <div className="max-w-[1280px] mx-auto px-4 py-2 flex items-center gap-3">
   349→            <div onClick={handleLogoClick} className="cursor-pointer shrink-0" title="اضغط 5 مرات للإدارة">
   350→              {school.logoUrl ? (
   351→                <img src={school.logoUrl} alt={school.name} className="w-12 h-12 rounded-full object-cover border-2" style={{ borderColor: school.primaryColor }} />
   352→              ) : (
   353→                <div className="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl" style={{ backgroundColor: school.primaryColor }}>🏫</div>
   354→              )}
   355→            </div>
   356→            <div className="flex-1 min-w-0">
   357→              <h1 className="text-base md:text-xl font-bold truncate" style={{ color: school.primaryColor }}>{school.name}</h1>
   358→              {settings?.heroSubtitle && <p className="text-xs text-gray-500 truncate">{settings.heroSubtitle}</p>}
   359→            </div>
   360→            <button className="lg:hidden p-2 text-2xl text-gray-600" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   361→              {mobileMenuOpen ? '✕' : '☰'}
   362→            </button>
   363→          </div>
   364→        </header>
   365→
   366→        {/* Navigation */}
   367→        <nav style={{ backgroundColor: '#1a2332' }} className="text-white shadow-lg">
   368→          <div className="max-w-[1280px] mx-auto px-4">
   369→            {/* Desktop Nav */}
   370→            <div className="hidden lg:flex items-center h-11">
   371→              {navLinks.map((link) => (
   372→                link.isServices ? (
   373→                  <div key={link.href} className="relative" onMouseEnter={() => setServicesOpen(true)} onMouseLeave={() => setServicesOpen(false)}>
   374→                    <button className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center gap-1 transition-colors">
   375→                      {link.label} ▾
   376→                    </button>
   377→                    {servicesOpen && (
   378→                      <div className="absolute top-full right-0 bg-white rounded-lg shadow-xl border min-w-[220px] z-50 overflow-hidden">
   379→                        {serviceItems.map(item => (
   380→                          <button key={item.action} onClick={() => handleServiceAction(item.action)}
   381→                            className="w-full flex items-center gap-3 px-4 py-3 text-sm text-gray-700 hover:bg-gray-50 transition-colors">
   382→                            <span className="text-lg">{item.icon}</span> {item.label}
   383→                          </button>
   384→                        ))}
   385→                      </div>
   386→                    )}
   387→                  </div>
   388→                ) : (
   389→                  <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   390→                    onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true) } : undefined}
   391→                    className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center transition-colors">
   392→                    {link.label}
   393→                  </a>
   394→                )
   395→              ))}
   396→            </div>
   397→            {/* Mobile Nav */}
   398→            {mobileMenuOpen && (
   399→              <div className="lg:hidden py-2 border-t border-white/10">
   400→                {navLinks.map((link) => (
   401→                  link.isServices ? (
   402→                    <div key={link.href}>
   403→                      <div className="px-4 py-2 text-sm font-medium text-white/70">{link.label}</div>
   404→                      {serviceItems.map(item => (
   405→                        <button key={item.action} onClick={() => handleServiceAction(item.action)}
   406→                          className="w-full flex items-center gap-2 px-8 py-2 text-sm text-white/80 hover:bg-white/10">
   407→                          <span>{item.icon}</span> {item.label}
   408→                        </button>
   409→                      ))}
   410→                    </div>
   411→                  ) : (
   412→                    <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   413→                      onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true); setMobileMenuOpen(false) } : () => setMobileMenuOpen(false)}
   414→                      className="block px-4 py-2 text-sm hover:bg-white/10">{link.label}</a>
   415→                  )
   416→                ))}
   417→              </div>
   418→            )}
   419→          </div>
   420→        </nav>
   421→
   422→        {/* News Ticker - Seamless scrolling */}
   423→        {settings?.showNewsTicker && news.length > 0 && (
