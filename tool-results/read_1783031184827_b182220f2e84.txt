   198→   198→  const c = (category || '').trim();

   200→   200→    'تنبيه': 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300',
   201→   201→    'أخبار': 'bg-violet-100 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300',

   203→   203→    'فعالية': 'bg-fuchsia-100 text-fuchsia-700 dark:bg-fuchsia-900/30 dark:text-fuchsia-300',
   204→   204→  };
   205→   205→  return map[c] || 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300';
   206→   206→}

   212→   212→      year: 'numeric',

   214→   214→      day: 'numeric',
   215→   215→    });

   218→   218→  }

   278→   278→        toast.error(msg);
   279→   279→      } finally {
   280→   280→        if (cancelled) return;

   282→   282→        void t;
   283→   283→      }
