   234→function categoryBadgeClass(category: string): string {
   235→  const c = (category || '').trim();
   236→  const map: Record<string, string> = {
   237→    'تنبيه': 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300',
   238→    'أخبار': 'bg-violet-100 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300',
   239→    'إعلان': 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-300',
   240→    'فعالية': 'bg-fuchsia-100 text-fuchsia-700 dark:bg-fuchsia-900/30 dark:text-fuchsia-300',
   241→  };
   242→  return map[c] || 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300';
   243→}

   245→function formatDate(iso: string): string {
   246→  try {
   247→    const d = new Date(iso);
   248→    return d.toLocaleDateString('ar-EG', {
   249→      year: 'numeric',
   250→      month: 'long',
   251→      day: 'numeric',
   252→    });
   253→  } catch {
   254→    return '';
   255→  }

   258→export default function TeacherPortalPage({ onBack, schoolId }: TeacherPortalPageProps) {
   259→  // ===== State =====
   260→  const [fadeIn, setFadeIn] = useState(false);
   261→  const [visibleCards, setVisibleCards] = useState<number[]>([]);
   262→
   263→  // Login form state
   264→  const [teacherCode, setTeacherCode] = useState('');
   265→  const [phone, setPhone] = useState('');
   266→  const [loginError, setLoginError] = useState<string | null>(null);
   267→  const [submitting, setSubmitting] = useState(false);
   268→
   269→  // Session state
   270→  const [session, setSession] = useState<LoginResponse | null>(null);
   271→  const [dashboard, setDashboard] = useState<DashboardData | null>(null);
   272→  const [dashboardLoading, setDashboardLoading] = useState(false);
   273→
   274→  // Sub-page navigation
   275→  const [showExamsPage, setShowExamsPage] = useState(false);
   276→  const [showGradingPage, setShowGradingPage] = useState(false);
   277→  const [showAttendancePage, setShowAttendancePage] = useState(false);
   278→
   279→  // ===== Restore session on mount =====
   280→  useEffect(() => {
   281→    let restored: LoginResponse | null = null;
   282→    try {
   283→      const raw = sessionStorage.getItem(SESSION_KEY);
   284→      if (raw) restored = JSON.parse(raw) as LoginResponse;
   285→    } catch {
   286→      restored = null;
   287→    }
   288→    // set-state-in-effect pattern
   289→    const t = setTimeout(() => {
   290→      if (restored) setSession(restored);
   291→      setFadeIn(true);
   292→    }, 0);
   293→    return () => clearTimeout(t);
   294→  }, []);
   295→
   296→  // ===== Fetch dashboard when session changes =====
   297→  useEffect(() => {
   298→    if (!session) {
   299→      const t = setTimeout(() => setDashboard(null), 0);
   300→      return () => clearTimeout(t);
   301→    }
   302→    let cancelled = false;
   303→    const t1 = setTimeout(() => setDashboardLoading(true), 0);
   304→    async function run() {
   305→      try {
   306→        const res = await fetch(
   307→          `/api/teacher/dashboard?schoolId=${encodeURIComponent(
   308→            schoolId
   309→          )}&teacherId=${encodeURIComponent(session.teacherId)}`
   310→        );
   311→        const data = await res.json();
   312→        if (!res.ok) {
   313→          throw new Error(data?.error || 'فشل تحميل البيانات');
   314→        }
   315→        if (cancelled) return;
   316→        const t = setTimeout(() => setDashboard(data as DashboardData), 0);
   317→        void t;
   318→      } catch (err) {
   319→        const msg = err instanceof Error ? err.message : 'فشل تحميل البيانات';
   320→        toast.error(msg);
   321→      } finally {
   322→        if (cancelled) return;
   323→        const t = setTimeout(() => setDashboardLoading(false), 0);
   324→        void t;
   325→      }
   326→    }
   327→    run();
   328→    return () => {
   329→      cancelled = true;
   330→      clearTimeout(t1);
   331→    };
   332→  }, [session, schoolId]);
   333→
   334→  // Staggered card animation for quick actions
   335→  useEffect(() => {
   336→    if (!session) return;
   337→    const timeouts: ReturnType<typeof setTimeout>[] = [];
   338→    quickActions.forEach((_, index) => {
   339→      timeouts.push(
   340→        setTimeout(() => {
   341→          setVisibleCards((prev) =>
   342→            prev.includes(index) ? prev : [...prev, index]
   343→          );
   344→        }, 300 + index * 120)
   345→      );
   346→    });
   347→    return () => timeouts.forEach(clearTimeout);
   348→  }, [session]);
   349→
   350→  // ===== Handlers =====
   351→  const handleLogin = async (e: React.FormEvent) => {
   352→    e.preventDefault();
   353→    setLoginError(null);
   354→    if (!teacherCode.trim() || !phone.trim()) {
   355→      const t = setTimeout(() => setLoginError('يرجى إدخال كود المعلم ورقم الهاتف'), 0);
   356→      return () => clearTimeout(t);
   357→    }
   358→    setSubmitting(true);
   359→    try {
   360→      const res = await fetch('/api/teacher/login', {
   361→        method: 'POST',
   362→        headers: { 'Content-Type': 'application/json' },
   363→        body: JSON.stringify({ schoolId, teacherCode: teacherCode.trim(), phone: phone.trim() }),
   364→      });
   365→      const data = await res.json();
   366→      if (!res.ok) {
   367→        throw new Error(data?.error || 'فشل تسجيل الدخول');
   368→      }
   369→      const loginData = data as LoginResponse;
   370→      try {
   371→        sessionStorage.setItem(SESSION_KEY, JSON.stringify(loginData));
   372→      } catch {
   373→        // ignore quota errors
   374→      }
   375→      const t = setTimeout(() => {
   376→        setSession(loginData);
   377→        setVisibleCards([]);
   378→      }, 0);
   379→      void t;
   380→      if (loginData.testMode) {
   381→        toast.success('تم تسجيل الدخول في وضع التجربة');
   382→      } else {
   383→        toast.success(`مرحباً ${loginData.name}`);
   384→      }
   385→      if (loginData.warning) {
   386→        toast.info(loginData.warning);
   387→      }
   388→    } catch (err) {
   389→      const msg = err instanceof Error ? err.message : 'فشل تسجيل الدخول';
   390→      const t = setTimeout(() => setLoginError(msg), 0);
   391→      void t;
   392→    } finally {
   393→      const t = setTimeout(() => setSubmitting(false), 0);
   394→      void t;
   395→    }
   396→  };
   397→
   398→  const handleLogout = () => {
   399→    try {
   400→      sessionStorage.removeItem(SESSION_KEY);
   401→    } catch {
   402→      // ignore
   403→    }
   404→    const t = setTimeout(() => {
   405→      setSession(null);
   406→      setDashboard(null);
   407→      setTeacherCode('');
   408→      setPhone('');
   409→      setVisibleCards([]);
   410→    }, 0);
   411→    void t;
   412→    toast.info('تم تسجيل الخروج من بوابة المعلم');
   413→  };
   414→
   415→  const handleQuickAction = (title: string) => {
   416→    if (title === 'إدارة الامتحانات') {
   417→      const t = setTimeout(() => setShowExamsPage(true), 0);
   418→      void t;
   419→      return;
   420→    }
   421→    if (title === 'تصحيح الدرجات') {
   422→      const t = setTimeout(() => setShowGradingPage(true), 0);
   423→      void t;
   424→      return;
   425→    }
   426→    if (title === 'سجل الحضور') {
   427→      const t = setTimeout(() => setShowAttendancePage(true), 0);
   428→      void t;
   429→      return;
   430→    }
   431→    toast.info(`${title} — قريباً`, {
   432→      description: 'هذه الخدمة قيد التطوير وستتوفر في الإصدار القادم.',
   433→    });
   434→  };
   435→
   436→  // ===== Render: Login Form =====
   437→  if (!session) {
   438→    return (
   439→      <div
   440→        className="min-h-screen flex flex-col bg-gradient-to-br from-slate-50 via-violet-50/30 to-slate-100 dark:from-gray-900 dark:via-gray-800 dark:to-gray-900"
   441→        dir="rtl"
   442→      >
   443→        {/* Header */}
   444→        <header className="bg-gradient-to-l from-[#2A374E] to-[#3d4f6e] text-white shadow-xl sticky top-0 z-50">
   445→          <div className="container mx-auto px-4 py-4">
   446→            <div className="flex items-center justify-between">
   447→              <button
   448→                onClick={onBack}
   449→                className="flex items-center gap-2 text-white hover:text-violet-300 transition-colors"
   450→              >
   451→                <ArrowRight className="w-5 h-5" />
   452→                <span className="font-medium">العودة للرئيسية</span>
   453→              </button>
   454→              <div className="flex items-center gap-3">
   455→                <div className="w-10 h-10 bg-gradient-to-r from-violet-500 to-purple-600 rounded-full flex items-center justify-center shadow-lg">
   456→                  <GraduationCap className="w-6 h-6 text-white" />
   457→                </div>
   458→                <div>
   459→                  <h1 className="text-xl font-bold">بوابة المعلم</h1>
   460→                  <p className="text-violet-200 text-xs">منصة المعلمين الإلكترونية</p>
   461→                </div>
   462→              </div>
   463→              <div className="w-28" />
   464→            </div>
   465→          </div>
   466→        </header>
   467→
   468→        <main className="flex-1 container mx-auto px-4 py-8 max-w-md flex items-center justify-center">
   469→          <div
   470→            className={`w-full transition-all duration-700 ${
   471→              fadeIn ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'
   472→            }`}
   473→          >
   474→            <Card className="border-0 shadow-2xl overflow-hidden">
   475→              <CardContent className="p-0">
   476→                {/* Gradient header */}
   477→                <div className="bg-gradient-to-l from-violet-600 via-purple-600 to-fuchsia-600 p-8 text-white text-center relative overflow-hidden">
   478→                  <div className="absolute top-0 left-0 w-48 h-48 bg-white/10 rounded-full -translate-x-1/2 -translate-y-1/2" />
   479→                  <div className="absolute bottom-0 right-0 w-40 h-40 bg-white/5 rounded-full translate-x-1/3 translate-y-1/3" />
   480→                  <div className="relative z-10">
   481→                    <div className="w-20 h-20 bg-white/20 backdrop-blur rounded-2xl flex items-center justify-center mx-auto mb-4 ring-4 ring-white/10">
   482→                      <GraduationCap className="w-10 h-10 text-white" />
   483→                    </div>
   484→                    <h2 className="text-2xl font-bold mb-1">تسجيل دخول المعلم</h2>
   485→                    <p className="text-violet-100 text-sm">
   486→                      أدخل بياناتك للوصول إلى لوحة التحكم
   487→                    </p>
   488→                  </div>
   489→                </div>
   490→
   491→                {/* Form */}
   492→                <form onSubmit={handleLogin} className="p-6 space-y-5">
   493→                  <div className="space-y-2">
   494→                    <Label htmlFor="teacherCode" className="text-sm font-medium flex items-center gap-1.5">

   498→                    <Input
   499→                      id="teacherCode"
   500→                      type="text"
   501→                      value={teacherCode}
   502→                      onChange={(e) => setTeacherCode(e.target.value)}
   503→                      placeholder="example@school.edu"
   504→                      autoComplete="username"
   505→                      disabled={submitting}
   506→                      className="h-11"
   507→                    />
   508→                  </div>
   509→
   510→                  <div className="space-y-2">
   511→                    <Label htmlFor="phone" className="text-sm font-medium flex items-center gap-1.5">
   512→                      <Phone className="w-4 h-4 text-violet-600" />
   513→                      رقم الهاتف
   514→                    </Label>
   515→                    <Input
   516→                      id="phone"
   517→                      type="tel"
   518→                      value={phone}
   519→                      onChange={(e) => setPhone(e.target.value)}
   520→                      placeholder="01XXXXXXXXX"
   521→                      autoComplete="tel"
   522→                      disabled={submitting}
   523→                      className="h-11"
   524→                      dir="ltr"
   525→                    />
   526→                  </div>
