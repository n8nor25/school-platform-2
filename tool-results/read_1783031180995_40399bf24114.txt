   197→function categoryBadgeClass(category: string): string {
   198→  const c = (category || '').trim();
   199→  const map: Record<string, string> = {
   200→    'تنبيه': 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300',
   201→    'أخبار': 'bg-violet-100 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300',
   202→    'إعلان': 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-300',
   203→    'فعالية': 'bg-fuchsia-100 text-fuchsia-700 dark:bg-fuchsia-900/30 dark:text-fuchsia-300',
   204→  };
   205→  return map[c] || 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300';
   206→}

   208→function formatDate(iso: string): string {
   209→  try {
   210→    const d = new Date(iso);
   211→    return d.toLocaleDateString('ar-EG', {
   212→      year: 'numeric',
   213→      month: 'long',
   214→      day: 'numeric',
   215→    });
   216→  } catch {
   217→    return '';
   218→  }
   219→}
