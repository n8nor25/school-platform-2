     6→  Phone, User, Hash, Calendar, Settings, BarChart3

     8→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     9→import { Button } from '@/components/ui/button'
    10→import { Input } from '@/components/ui/input'
    11→import { Label } from '@/components/ui/label'
    12→import { Badge } from '@/components/ui/badge'
    13→import { Textarea } from '@/components/ui/textarea'
    14→import { Switch } from '@/components/ui/switch'
    15→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from '@/components/ui/dialog'

    17→import { Skeleton } from '@/components/ui/skeleton'
    18→import { toast } from 'sonner'
    19→import { useAdminStore } from '@/lib/admin-store'
    20→

    22→  id: string

    39→  driverName: string

    42→  supervisorName: string
    43→  supervisorPhone: string
    44→  capacity: string
    45→  model: string
    46→  color: string
    47→  notes: string
    48→  active: boolean

    51→const defaultForm: BusForm = {
    52→  plateNumber: '',
    53→  driverName: '',
    54→  driverPhone: '',
    55→  driverLicense: '',
    56→  supervisorName: '',
    57→  supervisorPhone: '',
    58→  capacity: '30',
    59→  model: '',
    60→  color: '',
    61→  notes: '',
    62→  active: true,
    63→}
    64→
    65→export function BusFleetManagement() {
    66→  const [buses, setBuses] = useState<Bus[]>([])
    67→  const [loading, setLoading] = useState(true)
    68→  const [form, setForm] = useState<BusForm>(defaultForm)
    69→  const [editId, setEditId] = useState<string | null>(null)
    70→  const [dialogOpen, setDialogOpen] = useState(false)
    71→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
    72→  const [deleteTarget, setDeleteTarget] = useState<Bus | null>(null)
    73→  const [saving, setSaving] = useState(false)
    74→  const [search, setSearch] = useState('')
    75→  const [activeFilter, setActiveFilter] = useState<string>('all')
    76→  const { selectedSchoolId } = useAdminStore()
    77→
    78→  const fetchBuses = useCallback(async () => {
    79→    if (!selectedSchoolId) return
    80→    try {
    81→      setLoading(true)
    82→      const params = new URLSearchParams({ schoolId: selectedSchoolId })
    83→      if (search.trim()) params.set('search', search.trim())
    84→      if (activeFilter === 'true') params.set('active', 'true')
    85→      if (activeFilter === 'false') params.set('active', 'false')
    86→      const res = await fetch(`/api/buses?${params.toString()}`)
    87→      if (res.ok) {
    88→        const data = await res.json()
    89→        setBuses(Array.isArray(data.buses) ? data.buses : [])
    90→      } else {
    91→        toast.error('فشل في تحميل الباصات')
    92→      }
    93→    } catch {
    94→      toast.error('فشل في تحميل الباصات')
    95→    } finally {
    96→      setLoading(false)
    97→    }
    98→  }, [selectedSchoolId, search, activeFilter])
    99→
   100→  useEffect(() => {
   101→    let cancelled = false
   102→    const load = async () => {
   103→      if (!selectedSchoolId) return
   104→      try {
   105→        setLoading(true)
   106→        const params = new URLSearchParams({ schoolId: selectedSchoolId })
   107→        if (search.trim()) params.set('search', search.trim())
   108→        if (activeFilter === 'true') params.set('active', 'true')
   109→        if (activeFilter === 'false') params.set('active', 'false')
   110→        const res = await fetch(`/api/buses?${params.toString()}`)
   111→        if (res.ok) {
   112→          const data = await res.json()
   113→          if (!cancelled) setBuses(Array.isArray(data.buses) ? data.buses : [])
   114→        } else if (!cancelled) {
   115→          toast.error('فشل في تحميل الباصات')
   116→        }
   117→      } catch {
   118→        if (!cancelled) toast.error('فشل في تحميل الباصات')
   119→      } finally {
   120→        if (!cancelled) setLoading(false)
   121→      }
   122→    }
   123→    load()
   124→    return () => { cancelled = true }
   125→  }, [selectedSchoolId, search, activeFilter])
   126→
   127→  const openAdd = () => {
   128→    setEditId(null)
   129→    setForm(defaultForm)
   130→    setDialogOpen(true)
   131→  }
   132→
   133→  const openEdit = (bus: Bus) => {
   134→    setEditId(bus.id)
   135→    setForm({
   136→      plateNumber: bus.plateNumber,
   137→      driverName: bus.driverName,
   138→      driverPhone: bus.driverPhone || '',
   139→      driverLicense: bus.driverLicense || '',
   140→      supervisorName: bus.supervisorName || '',
   141→      supervisorPhone: bus.supervisorPhone || '',
   142→      capacity: String(bus.capacity),
   143→      model: bus.model || '',
   144→      color: bus.color || '',
   145→      notes: bus.notes || '',
   146→      active: bus.active,
   147→    })
   148→    setDialogOpen(true)
   149→  }
   150→
   151→  const handleSave = async () => {
   152→    if (!form.plateNumber.trim() || !form.driverName.trim()) {
   153→      toast.error('رقم اللوحة واسم السائق مطلوبان')
   154→      return
   155→    }
   156→    setSaving(true)
   157→    try {
   158→      const body = {
   159→        plateNumber: form.plateNumber.trim(),
   160→        driverName: form.driverName.trim(),
   161→        driverPhone: form.driverPhone.trim() || null,
   162→        driverLicense: form.driverLicense.trim() || null,
   163→        supervisorName: form.supervisorName.trim() || null,
   164→        supervisorPhone: form.supervisorPhone.trim() || null,
   165→        capacity: Number(form.capacity) || 30,
   166→        model: form.model.trim() || null,
   167→        color: form.color.trim() || null,
   168→        notes: form.notes.trim() || null,
   169→        active: form.active,
   170→      }

   172→        ? `/api/buses/${editId}?schoolId=${selectedSchoolId}`
   173→        : `/api/buses?schoolId=${selectedSchoolId}`
   174→      const res = await fetch(url, {
   175→        method: editId ? 'PUT' : 'POST',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify(body),
   178→      })
   179→      const data = await res.json()
   180→      if (res.ok) {
   181→        toast.success(editId ? 'تم تحديث الباص' : 'تم إضافة الباص بنجاح')
   182→        setDialogOpen(false)
   183→        fetchBuses()
   184→      } else if (res.status === 409) {
   185→        toast.error(data.error || 'رقم اللوحة مُستخدم بالفعل')
   186→      } else {
   187→        toast.error(data.error || 'فشل في حفظ البيانات')
   188→      }
   189→    } catch {
   190→      toast.error('فشل في حفظ البيانات')
   191→    } finally {
   192→      setSaving(false)
   193→    }
   194→  }
   195→
   196→  const handleDelete = async () => {
   197→    if (!deleteTarget) return
   198→    try {
   199→      const res = await fetch(`/api/buses/${deleteTarget.id}?schoolId=${selectedSchoolId}`, {
   200→        method: 'DELETE',
   201→      })
   202→      const data = await res.json()
   203→      if (res.ok) {
   204→        toast.success('تم حذف الباص بنجاح')
   205→        fetchBuses()
   206→      } else if (res.status === 409) {
   207→        toast.error(data.error || 'لا يمكن حذف الباص')
   208→      } else {
   209→        toast.error(data.error || 'فشل في حذف الباص')
   210→      }
   211→    } catch {
   212→      toast.error('فشل في حذف الباص')
   213→    } finally {
   214→      setDeleteDialogOpen(false)
   215→      setDeleteTarget(null)
   216→    }
   217→  }
   218→
   219→  // Stats
   220→  const totalBuses = buses.length
   221→  const activeBuses = buses.filter((b) => b.active).length
   222→  const totalCapacity = buses.reduce((sum, b) => sum + b.capacity, 0)
   223→  const modelCounts: Record<string, number> = {}
   224→  buses.forEach((b) => {
   225→    const m = b.model || 'غير محدد'
   226→    modelCounts[m] = (modelCounts[m] || 0) + 1
   227→  })
   228→
   229→  return (
   230→    <div className="space-y-6">
   231→      <div className="flex justify-between items-center flex-wrap gap-3">
   232→        <h2 className="text-lg font-bold text-[#1a1a2e] flex items-center gap-2">
   233→          <BusIcon className="w-5 h-5 text-[#610000]" />
   234→          إدارة أسطول الباصات
   235→        </h2>
   236→      </div>
   237→
   238→      <Tabs defaultValue="list">
   239→        <TabsList className="bg-gray-100">
   240→          <TabsTrigger value="list" className="data-[state=active]:bg-white">
   241→            <BusIcon className="w-4 h-4 ml-1" /> القائمة
   242→          </TabsTrigger>
   243→          <TabsTrigger value="add" className="data-[state=active]:bg-white">
   244→            <Plus className="w-4 h-4 ml-1" /> إضافة باص
   245→          </TabsTrigger>
   246→          <TabsTrigger value="stats" className="data-[state=active]:bg-white">
   247→            <BarChart3 className="w-4 h-4 ml-1" /> إحصائيات
   248→          </TabsTrigger>
   249→        </TabsList>
   250→
   251→        {/* List Tab */}
   252→        <TabsContent value="list" className="space-y-4">
   253→          <Card>
   254→            <CardContent className="p-4">
   255→              <div className="flex flex-wrap gap-3 items-center">
   256→                <div className="relative flex-1 min-w-[200px]">
   257→                  <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   258→                  <Input
   259→                    placeholder="بحث برقم اللوحة أو اسم السائق..."
   260→                    value={search}
   261→                    onChange={(e) => setSearch(e.target.value)}
   262→                    className="pr-10 h-11"
   263→                  />
   264→                </div>
   265→                <div className="flex gap-1 bg-gray-100 rounded-lg p-1">
   266→                  {[
   267→                    { value: 'all', label: 'الكل' },
   268→                    { value: 'true', label: 'نشط' },
   269→                    { value: 'false', label: 'متوقف' },
   270→                  ].map((opt) => (
   271→                    <button
   272→                      key={opt.value}
   273→                      onClick={() => setActiveFilter(opt.value)}
   274→                      className={`px-3 py-1.5 rounded-md text-sm font-medium transition-colors min-h-[36px] ${
   275→                        activeFilter === opt.value
   276→                          ? 'bg-white text-[#610000] shadow-sm'
   277→                          : 'text-gray-500 hover:text-gray-700'
   278→                      }`}
   279→                    >
   280→                      {opt.label}
   281→                    </button>
   282→                  ))}
   283→                </div>
   284→                <Button
   285→                  onClick={openAdd}
   286→                  className="bg-[#610000] hover:bg-[#8B0000] text-white min-h-[44px]"
   287→                >
   288→                  <Plus className="w-4 h-4 ml-1" />
   289→                  إضافة باص
   290→                </Button>
   291→              </div>
   292→            </CardContent>
   293→          </Card>
   294→
   295→          {loading ? (
   296→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   297→              {Array.from({ length: 6 }).map((_, i) => (
   298→                <Skeleton key={i} className="h-48 rounded-xl" />
   299→              ))}
   300→            </div>
   301→          ) : buses.length === 0 ? (
   302→            <Card>
   303→              <CardContent className="py-12 text-center">
   304→                <AlertCircle className="w-12 h-12 mx-auto text-gray-300 mb-3" />
   305→                <p className="text-gray-400">لا توجد باصات مسجلة</p>
   306→              </CardContent>
   307→            </Card>
   308→          ) : (
   309→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   310→              {buses.map((bus) => (
   311→                <Card key={bus.id} className="hover:shadow-lg transition-shadow">
   312→                  <CardContent className="p-4">
   313→                    <div className="flex items-start justify-between gap-2 mb-3">
   314→                      <div className="flex items-center gap-2">
   315→                        <div className="w-11 h-11 rounded-lg bg-[#610000]/10 flex items-center justify-center">
   316→                          <BusIcon className="w-5 h-5 text-[#610000]" />
   317→                        </div>
   318→                        <div>
   319→                          <h3 className="font-bold text-[#1a1a2e]">{bus.plateNumber}</h3>
   320→                          <p className="text-xs text-gray-500">{bus.model || '—'}</p>
   321→                        </div>
   322→                      </div>
   323→                      <Badge
   324→                        className={
   325→                          bus.active
   326→                            ? 'bg-green-100 text-green-700 hover:bg-green-100'
   327→                            : 'bg-gray-200 text-gray-600 hover:bg-gray-200'
   328→                        }
   329→                      >
   330→                        {bus.active ? 'نشط' : 'متوقف'}
   331→                      </Badge>
   332→                    </div>
   333→
   334→                    <div className="space-y-2 text-sm">
   335→                      <div className="flex items-center gap-2 text-gray-600">
   336→                        <User className="w-4 h-4 text-[#610000]" />
   337→                        <span className="font-medium">{bus.driverName}</span>
   338→                      </div>
   339→                      {bus.driverPhone && (
   340→                        <div className="flex items-center gap-2 text-gray-600" dir="ltr">
   341→                          <Phone className="w-4 h-4 text-[#610000]" />
   342→                          <span>{bus.driverPhone}</span>
   343→                        </div>
   344→                      )}
   345→                      {bus.supervisorName && (
   346→                        <div className="flex items-center gap-2 text-gray-600">
   347→                          <User className="w-4 h-4 text-gray-400" />
   348→                          <span className="text-xs">مشرف: {bus.supervisorName}</span>
   349→                        </div>
   350→                      )}
   351→                      <div className="flex items-center gap-2 text-gray-600">
   352→                        <Hash className="w-4 h-4 text-[#610000]" />
   353→                        <span className="text-xs">السعة: {bus.capacity} راكب</span>
   354→                      </div>
   355→                      <div className="flex items-center gap-2 text-gray-600">
   356→                        <Calendar className="w-4 h-4 text-[#610000]" />
   357→                        <span className="text-xs">الخطوط: {bus._count?.routes || 0}</span>
   358→                      </div>
   359→                    </div>
   360→
   361→                    <div className="flex gap-2 mt-3 justify-end">
   362→                      <Button
   363→                        variant="outline"
   364→                        size="sm"
   365→                        className="min-h-[40px]"
   366→                        onClick={() => openEdit(bus)}
   367→                      >
   368→                        <Edit className="w-3.5 h-3.5 ml-1" />
   369→                        تعديل
   370→                      </Button>
   371→                      <Button
   372→                        variant="outline"
   373→                        size="sm"
   374→                        className="text-red-500 hover:text-red-700 hover:bg-red-50 min-h-[40px]"
   375→                        onClick={() => { setDeleteTarget(bus); setDeleteDialogOpen(true) }}
   376→                      >
   377→                        <Trash2 className="w-3.5 h-3.5 ml-1" />
   378→                        حذف
   379→                      </Button>
   380→                    </div>
   381→                  </CardContent>
   382→                </Card>
   383→              ))}
   384→            </div>
   385→          )}
   386→        </TabsContent>
   387→
   388→        {/* Add Tab */}
   389→        <TabsContent value="add">
   390→          <Card>
   391→            <CardHeader>
   392→              <CardTitle className="text-base flex items-center gap-2">
   393→                <Plus className="w-4 h-4 text-[#610000]" />
   394→                إضافة باص جديد
   395→              </CardTitle>
   396→            </CardHeader>
   397→            <CardContent>
   398→              <BusFormComponent
   399→                form={defaultForm}
   400→                onChange={() => {}}
   401→                onSubmit={async (body) => {
   402→                  setSaving(true)
   403→                  try {
   404→                    const res = await fetch(`/api/buses?schoolId=${selectedSchoolId}`, {
   405→                      method: 'POST',
   406→                      headers: { 'Content-Type': 'application/json' },
   407→                      body: JSON.stringify(body),
   408→                    })
   409→                    const data = await res.json()
   410→                    if (res.ok) {
   411→                      toast.success('تم إضافة الباص بنجاح')
   412→                      fetchBuses()
   413→                    } else if (res.status === 409) {
   414→                      toast.error(data.error || 'رقم اللوحة مُستخدم')
   415→                    } else {
   416→                      toast.error(data.error || 'فشل')
   417→                    }
   418→                  } finally {
   419→                    setSaving(false)
   420→                  }
   421→                }}
   422→                saving={saving}
   423→                submitLabel="إضافة الباص"
   424→              />
   425→            </CardContent>
   426→          </Card>
   427→        </TabsContent>
   428→
   429→        {/* Stats Tab */}
   430→        <TabsContent value="stats">
   431→          <div className="grid sm:grid-cols-2 lg:grid-cols-4 gap-4">
   432→            <Card>
   433→              <CardContent className="p-4">
   434→                <div className="flex items-center gap-3">
   435→                  <div className="w-11 h-11 rounded-lg bg-[#610000]/10 flex items-center justify-center">
   436→                    <BusIcon className="w-5 h-5 text-[#610000]" />
   437→                  </div>
   438→                  <div>
   439→                    <p className="text-xs text-gray-500">إجمالي الباصات</p>
   440→                    <p className="text-xl font-bold text-[#1a1a2e]">{totalBuses}</p>
   441→                  </div>
   442→                </div>
   443→              </CardContent>
   444→            </Card>
   445→            <Card>
   446→              <CardContent className="p-4">
   447→                <div className="flex items-center gap-3">
   448→                  <div className="w-11 h-11 rounded-lg bg-green-100 flex items-center justify-center">
   449→                    <Settings className="w-5 h-5 text-green-600" />
   450→                  </div>
   451→                  <div>
   452→                    <p className="text-xs text-gray-500">باصات نشطة</p>
   453→                    <p className="text-xl font-bold text-[#1a1a2e]">{activeBuses}</p>
   454→                  </div>
   455→                </div>
   456→              </CardContent>
   457→            </Card>
   458→            <Card>
   459→              <CardContent className="p-4">
   460→                <div className="flex items-center gap-3">
   461→                  <div className="w-11 h-11 rounded-lg bg-amber-100 flex items-center justify-center">
   462→                    <Hash className="w-5 h-5 text-amber-600" />
   463→                  </div>
   464→                  <div>
   465→                    <p className="text-xs text-gray-500">إجمالي السعة</p>
   466→                    <p className="text-xl font-bold text-[#1a1a2e]">{totalCapacity}</p>
   467→                  </div>
   468→                </div>
   469→              </CardContent>
   470→            </Card>
   471→            <Card>
   472→              <CardContent className="p-4">
   473→                <div className="flex items-center gap-3">
   474→                  <div className="w-11 h-11 rounded-lg bg-gray-100 flex items-center justify-center">
   475→                    <BusIcon className="w-5 h-5 text-gray-600" />
   476→                  </div>
   477→                  <div>
   478→                    <p className="text-xs text-gray-500">باصات متوقفة</p>
   479→                    <p className="text-xl font-bold text-[#1a1a2e]">{totalBuses - activeBuses}</p>
   480→                  </div>
   481→                </div>
   482→              </CardContent>
   483→            </Card>
   484→          </div>
   485→
   486→          <Card className="mt-4">
   487→            <CardHeader>
   488→              <CardTitle className="text-base">توزيع الباصات حسب الموديل</CardTitle>
   489→            </CardHeader>
   490→            <CardContent>
   491→              {Object.keys(modelCounts).length === 0 ? (
   492→                <p className="text-gray-400 text-center py-6">لا توجد بيانات</p>
   493→              ) : (
   494→                <div className="space-y-3">
   495→                  {Object.entries(modelCounts).map(([model, count]) => (
   496→                    <div key={model} className="flex items-center gap-3">
   497→                      <span className="text-sm font-medium w-32 truncate">{model}</span>
   498→                      <div className="flex-1 bg-gray-100 rounded-full h-6 overflow-hidden">
   499→                        <div
   500→                          className="bg-[#610000] h-full rounded-full flex items-center justify-end px-2 text-white text-xs font-bold"
   501→                          style={{ width: `${(count / totalBuses) * 100}%` }}
   502→                        >
   503→                          {count}
   504→                        </div>
   505→                      </div>
   506→                    </div>
   507→                  ))}
   508→                </div>
   509→              )}
   510→            </CardContent>
   511→          </Card>
   512→        </TabsContent>
   513→      </Tabs>
   514→
   515→      {/* Edit Dialog */}
   516→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   517→        <DialogContent dir="rtl" className="max-w-2xl max-h-[90vh] overflow-y-auto">
   518→          <DialogHeader>
   519→            <DialogTitle>{editId ? 'تعديل بيانات الباص' : 'إضافة باص جديد'}</DialogTitle>
   520→            <DialogDescription>
   521→              {editId ? 'تعديل بيانات الباص' : 'أدخل بيانات الباص الجديد'}
   522→            </DialogDescription>
   523→          </DialogHeader>
   524→          <div className="space-y-4 mt-2">
   525→            <div className="grid sm:grid-cols-2 gap-3">
   526→              <div>
   527→                <Label>رقم اللوحة *</Label>
   528→                <Input
   529→                  value={form.plateNumber}
   530→                  onChange={(e) => setForm({ ...form, plateNumber: e.target.value })}
   531→                  className="h-11 mt-1.5"
   532→                  placeholder="رقم اللوحة"
   533→                />
