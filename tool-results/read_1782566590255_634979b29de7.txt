   167→   167→  if (!dateStr) return '—'

   169→   169→  return d.toLocaleDateString('ar-EG', { year: 'numeric', month: 'short', day: 'numeric' })

   172→   172→// Calculate how many months a subscription should have payments for
   173→   173→function getExpectedMonths(startDate: string, endDate: string | null): number {
   174→   174→  const start = new Date(startDate)
   175→   175→  const end = endDate ? new Date(endDate) : new Date()
   176→   176→  const months = (end.getFullYear() - start.getFullYear()) * 12 + (end.getMonth() - start.getMonth()) + 1
   177→   177→  return Math.max(0, months)
   178→   178→}
