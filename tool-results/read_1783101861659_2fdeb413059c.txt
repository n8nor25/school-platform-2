   148→// ===== Helpers =====
   149→function formatArabicDate(iso: string): string {
   150→  try {
   151→    const d = new Date(iso);
   152→    if (isNaN(d.getTime())) return '—';
   153→    return d.toLocaleDateString('ar-EG', {
   154→      year: 'numeric',
   155→      month: 'long',
   156→      day: 'numeric',
   157→      hour: '2-digit',
   158→      minute: '2-digit',
   159→    });
   160→  } catch {
   161→    return '—';
   162→  }
   163→}
   164→
   165→function formatArabicDateShort(iso: string): string {
   166→  try {
   167→    const d = new Date(iso);
   168→    if (isNaN(d.getTime())) return '—';
   169→    return d.toLocaleDateString('ar-EG', {
   170→      year: 'numeric',
   171→      month: 'short',
   172→      day: 'numeric',
   173→    });
   174→  } catch {
   175→    return '—';
   176→  }
   177→}

   273→export default function TeacherExamsPage({
   274→  onBack,
   275→  schoolId,
   276→  teacherId,
   277→  teacherName,
   278→}: TeacherExamsPageProps) {
   279→  // ===== State =====
   280→  const [fadeIn, setFadeIn] = useState(false);
   281→  const [exams, setExams] = useState<ExamListItem[]>([]);
   282→  const [loading, setLoading] = useState(false);
   283→  const [error, setError] = useState<string | null>(null);
   284→  const [search, setSearch] = useState('');
   285→  const [debouncedSearch, setDebouncedSearch] = useState('');
   286→  const [statusFilter, setStatusFilter] = useState<'all' | ExamStatus>('all');
   287→
   288→  // Detail modal state
   289→  const [detailOpen, setDetailOpen] = useState(false);
   290→  const [detailLoading, setDetailLoading] = useState(false);
   291→  const [detailError, setDetailError] = useState<string | null>(null);
   292→  const [detailExam, setDetailExam] = useState<ExamDetail | null>(null);
   293→  const [submissionsStats, setSubmissionsStats] = useState<SubmissionsStats | null>(null);
   294→
   295→  // Mounted ref (prevent setState after unmount)
   296→  const mountedRef = useRef(true);
   297→
   298→  // ===== Restore fadeIn =====
   299→  useEffect(() => {
   300→    const t = setTimeout(() => setFadeIn(true), 0);
   301→    return () => clearTimeout(t);
   302→  }, []);
   303→
   304→  // ===== mountedRef lifecycle =====
   305→  useEffect(() => {
   306→    mountedRef.current = true;
   307→    return () => {
   308→      mountedRef.current = false;
   309→    };
   310→  }, []);
   311→
   312→  // ===== Debounced search (300ms) =====
   313→  useEffect(() => {
   314→    const t = setTimeout(() => {
   315→      if (mountedRef.current) setDebouncedSearch(search);
   316→    }, 300);
   317→    return () => clearTimeout(t);
   318→  }, [search]);
   319→
   320→  // ===== Fetch exams =====
   321→  const fetchExams = useCallback(async (silent: boolean = false) => {
   322→    if (!mountedRef.current) return;
   323→    if (!silent) {
   324→      const t = setTimeout(() => setLoading(true), 0);
   325→      void t;
   326→    }
   327→    if (!silent) {
   328→      const t = setTimeout(() => setError(null), 0);
   329→      void t;
   330→    }
   331→    try {
   332→      const params = new URLSearchParams();
   333→      params.set('schoolId', schoolId);
   334→      params.set('teacherId', teacherId);
   335→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim());
   336→      if (statusFilter !== 'all') params.set('status', statusFilter);
   337→      params.set('limit', '100');
   338→
   339→      const res = await fetch(`/api/exams/teacher?${params.toString()}`);
   340→      const data = await res.json();
   341→      if (!res.ok) {
   342→        throw new Error(data?.error || 'فشل جلب الامتحانات');
   343→      }
   344→      if (!mountedRef.current) return;
   345→      const list = (data as ExamsListResponse).exams || [];
   346→      const t = setTimeout(() => setExams(list), 0);
   347→      void t;
   348→    } catch (err) {
   349→      if (!mountedRef.current) return;
   350→      const msg = err instanceof Error ? err.message : 'فشل جلب الامتحانات';
   351→      const t = setTimeout(() => setError(msg), 0);
   352→      void t;
   353→      if (!silent) toast.error(msg);
   354→    } finally {
   355→      if (!mountedRef.current) return;
   356→      const t = setTimeout(() => setLoading(false), 0);
   357→      void t;
   358→    }
   359→  }, [schoolId, teacherId, debouncedSearch, statusFilter]);
   360→
   361→  // ===== Initial fetch + auto-refresh + focus =====
   362→  useEffect(() => {
   363→    fetchExams(false);
   364→    const interval = setInterval(() => {
   365→      fetchExams(true);
   366→    }, 60_000);
   367→    const onFocus = () => fetchExams(true);
   368→    window.addEventListener('focus', onFocus);
   369→    return () => {
   370→      clearInterval(interval);
   371→      window.removeEventListener('focus', onFocus);
   372→    };
   373→  }, [debouncedSearch, statusFilter, schoolId, teacherId]);
   374→
   375→  // ===== Fetch exam detail =====
   376→  const openExamDetail = useCallback(async (examId: string) => {
   377→    if (!mountedRef.current) return;
   378→    const t1 = setTimeout(() => {
   379→      setDetailOpen(true);
   380→      setDetailLoading(true);
   381→      setDetailError(null);
   382→      setDetailExam(null);
   383→      setSubmissionsStats(null);
   384→    }, 0);
   385→    void t1;
   386→    try {
   387→      const params = new URLSearchParams();
   388→      params.set('schoolId', schoolId);
   389→      params.set('teacherId', teacherId);
   390→
   391→      const [examRes, subsRes] = await Promise.allSettled([
   392→        fetch(`/api/exams/teacher/${examId}?${params.toString()}`),
   393→        fetch(`/api/exams/teacher/${examId}/submissions?${params.toString()}&limit=1`),
   394→      ]);
   395→
   396→      if (!mountedRef.current) return;
   397→
   398→      if (examRes.status === 'fulfilled') {
   399→        const examData = await examRes.value.json();
   400→        if (!examRes.value.ok) {
   401→          throw new Error(examData?.error || 'فشل جلب تفاصيل الامتحان');
   402→        }
   403→        const t = setTimeout(() => setDetailExam((examData as ExamDetailResponse).exam), 0);
   404→        void t;
   405→      } else {
   406→        throw new Error('فشل الاتصال بالخادم');
   407→      }
   408→
   409→      if (subsRes.status === 'fulfilled') {
   410→        const subsData = await subsRes.value.json();
   411→        if (subsRes.value.ok && subsData?.stats) {
   412→          const t = setTimeout(() => setSubmissionsStats(subsData.stats as SubmissionsStats), 0);
   413→          void t;
   414→        }
   415→      }
   416→    } catch (err) {
   417→      if (!mountedRef.current) return;
   418→      const msg = err instanceof Error ? err.message : 'فشل جلب تفاصيل الامتحان';
   419→      const t = setTimeout(() => setDetailError(msg), 0);
   420→      void t;
   421→      toast.error(msg);
   422→    } finally {
   423→      if (!mountedRef.current) return;
   424→      const t = setTimeout(() => setDetailLoading(false), 0);
   425→      void t;
   426→    }
   427→  }, [schoolId, teacherId]);
   428→
   429→  // ===== Compute KPIs =====
   430→  const kpis = {
   431→    total: exams.length,
   432→    published: exams.filter((e) => e.status === 'PUBLISHED').length,
   433→    upcoming: exams.filter((e) => e.timeStatus === 'UPCOMING').length,
   434→    submissions: exams.reduce((sum, e) => sum + (e.submissionsCount || 0), 0),
   435→  };
   436→
   437→  // ===== Render =====
   438→  return (
   439→    <div
   440→      className="min-h-screen flex flex-col bg-gradient-to-br from-slate-50 via-violet-50/20 to-slate-100 dark:from-gray-900 dark:via-gray-800 dark:to-gray-900"
   441→      dir="rtl"
   442→    >
   443→      {/* Header */}
   444→      <header className="bg-gradient-to-l from-[#2A374E] to-[#3d4f6e] text-white shadow-xl sticky top-0 z-50">
   445→        <div className="container mx-auto px-4 py-4">
   446→          <div className="flex items-center justify-between gap-3">
   447→            <button
   448→              onClick={onBack}
   449→              className="flex items-center gap-2 text-white hover:text-violet-300 transition-colors shrink-0"
   450→            >
   451→              <ArrowRight className="w-5 h-5" />
   452→              <span className="font-medium hidden sm:inline">العودة للوحة التحكم</span>
   453→            </button>
   454→            <div className="flex items-center gap-3 min-w-0">
   455→              <div className="w-10 h-10 bg-gradient-to-r from-violet-500 to-purple-600 rounded-full flex items-center justify-center shadow-lg shrink-0">
   456→                <FileText className="w-6 h-6 text-white" />
   457→              </div>
   458→              <div className="min-w-0">
   459→                <h1 className="text-base md:text-xl font-bold truncate">إدارة الامتحانات</h1>
   460→                <p className="text-violet-200 text-xs truncate">{teacherName}</p>
   461→              </div>
   462→            </div>
   463→            <Button
   464→              onClick={() => fetchExams(false)}
   465→              variant="outline"
   466→              size="sm"
   467→              className="bg-white/10 border-white/20 text-white hover:bg-white/20 hover:text-white shrink-0"
   468→              disabled={loading}
   469→            >
   470→              <RefreshCw className={`w-4 h-4 ${loading ? 'animate-spin' : ''}`} />
   471→              <span className="hidden sm:inline">تحديث</span>
   472→            </Button>
   473→          </div>
   474→        </div>
   475→      </header>
   476→
   477→      <main className="flex-1 container mx-auto px-4 py-6 max-w-6xl">
   478→        {/* Page intro banner */}
   479→        <div
   480→          className={`mb-6 transition-all duration-700 ${
   481→            fadeIn ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'
   482→          }`}
   483→        >
   484→          <div className="relative overflow-hidden rounded-2xl">
   485→            <div className="absolute -inset-[2px] bg-gradient-to-r from-violet-500 via-purple-500 to-fuchsia-500 rounded-2xl blur-[1px] opacity-60" />
   486→            <Card className="relative border-0 shadow-2xl rounded-2xl overflow-hidden">
   487→              <CardContent className="p-0">
   488→                <div className="bg-gradient-to-l from-violet-700 via-purple-700 to-fuchsia-700 p-6 md:p-8 text-white relative overflow-hidden">
   489→                  <div className="absolute top-0 left-0 w-64 h-64 bg-white/10 rounded-full -translate-x-1/2 -translate-y-1/2" />
   490→                  <div className="absolute bottom-0 right-0 w-48 h-48 bg-white/5 rounded-full translate-x-1/4 translate-y-1/4" />
   491→                  <div className="relative z-10 flex items-center gap-4">
   492→                    <div className="w-14 h-14 bg-white/20 backdrop-blur rounded-xl flex items-center justify-center ring-4 ring-white/20 shrink-0">
   493→                      <Sparkles className="w-7 h-7 text-white" />
   494→                    </div>
   495→                    <div>
   496→                      <h2 className="text-xl md:text-2xl font-bold mb-1">إدارة الامتحانات الإلكترونية</h2>
   497→                      <p className="text-violet-100/90 text-sm">
   498→                        تابع امتحاناتك، التسليمات، والإحصائيات في مكان واحد
   499→                      </p>
   500→                    </div>
   501→                  </div>
   502→                </div>
   503→              </CardContent>
   504→            </Card>
   505→          </div>
   506→        </div>
   507→
   508→        {/* KPI Cards */}
   509→        <div className="mb-6 grid grid-cols-2 lg:grid-cols-4 gap-4">
   510→          {kpiConfig.map((k) => {
   511→            const Icon = k.icon;
   512→            const value = kpis[k.key];
   513→            return (
   514→              <Card
   515→                key={k.key}
   516→                className="border-0 shadow-md hover:shadow-lg transition-shadow overflow-hidden"
   517→              >
   518→                <CardContent className="p-5">
   519→                  <div className={`w-12 h-12 rounded-xl bg-gradient-to-br ${k.gradient} flex items-center justify-center shadow-md mb-3`}>
   520→                    <Icon className="w-6 h-6 text-white" />
   521→                  </div>
   522→                  <div className="text-3xl font-bold text-[#2A374E] dark:text-white mb-0.5">
   523→                    {loading && exams.length === 0 ? '—' : value}
   524→                  </div>
   525→                  <div className="text-xs text-gray-500 dark:text-gray-400">
   526→                    {k.label}
   527→                  </div>
