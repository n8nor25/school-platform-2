   245→function buildUrl(base: string, student: StudentInfo | null, extra?: Record<string, string>): string {
   246→  const params = new URLSearchParams();
   247→  if (student) {
   248→    params.set('schoolId', student.schoolId);
   249→    params.set('studentId', student.studentId);
   250→  }
   251→  if (extra) {
   252→    for (const [k, v] of Object.entries(extra)) params.set(k, v);
   253→  }
   254→  const qs = params.toString();
   255→  return qs ? `${base}?${qs}` : base;
   256→}

   258→function formatTime(seconds: number): string {
   259→  if (seconds < 0) seconds = 0;
   260→  const h = Math.floor(seconds / 3600);
   261→  const m = Math.floor((seconds % 3600) / 60);
   262→  const s = seconds % 60;
   263→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   264→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
   265→}

   267→function formatDate(iso: string): string {
   268→  try {
   269→    const d = new Date(iso);
   270→    return d.toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric', hour: '2-digit', minute: '2-digit' });
   271→  } catch {
   272→    return iso;
   273→  }

   504→function StudentLogin({
   505→  onLogin,
   506→  schoolId,
   507→}: {
   508→  onLogin: (info: StudentInfo) => void;
   509→  schoolId?: string;
   510→}) {
   511→  const [studentId, setStudentId] = useState('');
   512→  const [studentName, setStudentName] = useState('');
   513→  const [schoolIdInput, setSchoolIdInput] = useState(schoolId || '');
   514→  const [loading, setLoading] = useState(false);
   515→  const [error, setError] = useState<string | null>(null);
   516→
   517→  // مزامنة schoolIdInput مع الـ prop عندما يتوفر (selectedSchoolId يُحمّل من store
   518→  // بعد أول render). نُلّف setState بـ setTimeout لتجنّب تحذير set-state-in-effect.
   519→  useEffect(() => {
   520→    if (schoolId && schoolId !== schoolIdInput) {
   521→      const id = schoolId;
   522→      const t = setTimeout(() => setSchoolIdInput(id), 0);
   523→      return () => clearTimeout(t);
   524→    }
   525→  }, [schoolId, schoolIdInput]);
   526→
   527→  const handleSubmit = async (e: React.FormEvent) => {
   528→    e.preventDefault();
   529→    setError(null);
   530→    if (!studentId.trim()) {
   531→      setError('معرّف الطالب مطلوب');
   532→      return;
   533→    }
   534→    if (!studentName.trim()) {
   535→      setError('اسم الطالب مطلوب');
   536→      return;
   537→    }
   538→    if (!schoolIdInput.trim()) {
   539→      setError('معرّف المدرسة مطلوب');
   540→      return;
   541→    }
   542→    setLoading(true);
   543→    // تحقق سريع من توفر امتحانات (لا يلزم — فقط للتأكد من المدرسة)
   544→    try {
   545→      const res = await fetch(
   546→        `/api/exams/available?schoolId=${encodeURIComponent(schoolIdInput)}&studentId=${encodeURIComponent(studentId)}`,
   547→        { headers: { 'x-student-id': studentId } }
   548→      );
   549→      const data = await res.json();
   550→      if (!res.ok) {

   670→function ExamsList({
   671→  student,
   672→  onSelectExam,
   673→}: {
   674→  student: StudentInfo;
   675→  onSelectExam: (exam: AvailableExam) => void;
   676→}) {
   677→  const [exams, setExams] = useState<AvailableExam[]>([]);
   678→  const [loading, setLoading] = useState(true);
   679→  const [error, setError] = useState<string | null>(null);
   680→
   681→  // ===== إحصائيات الطالب =====
   682→  const [stats, setStats] = useState<StudentStats | null>(null);
   683→  const [statsLoading, setStatsLoading] = useState(true);
   684→
   685→  // ===== الفلترة والتبويب =====
   686→  const [activeTab, setActiveTab] = useState<'available' | 'history'>('available');
   687→  // التبويب الرئيسي: رسمي / تدريبي (يفتلر قائمة الامتحانات حسب category)
   688→  const [categoryTab, setCategoryTab] = useState<ExamCategory>('OFFICIAL');
   689→  const [searchTerm, setSearchTerm] = useState('');
   690→  const [subjectFilter, setSubjectFilter] = useState('all');
   691→  const [statusFilter, setStatusFilter] = useState('all');
   692→
   693→  const loadExams = useCallback(async () => {
   694→    setLoading(true);
   695→    setError(null);
   696→    try {
   697→      // الـ API المحدّث: /api/exams/student?schoolId&studentId&studentName
   698→      // يُرجع استجابة مقسّمة: available, inProgress, upcoming, endedOrExhausted
   699→      const res = await fetch(
   700→        buildUrl('/api/exams/student', student, { studentName: student.studentName }),
   701→        { headers: { 'x-student-id': student.studentId } },
   702→      );
   703→      const data: ExamsStudentResponse = await res.json();
   704→      if (!res.ok) {
   705→        setError((data as unknown as { error?: string }).error || 'فشل جلب الامتحانات');
   706→        setExams([]);
   707→        return;
   708→      }
   709→      // جمّع كل الأقسام في قائمة واحدة (مع دعم الاستجابة القديمة data.exams)
   710→      const list: AvailableExam[] = [];
   711→      if (Array.isArray(data.available)) list.push(...data.available);
   712→      if (Array.isArray(data.inProgress)) list.push(...data.inProgress);
   713→      if (Array.isArray(data.upcoming)) list.push(...data.upcoming);
   714→      if (Array.isArray(data.endedOrExhausted)) list.push(...data.endedOrExhausted);
   715→      if (list.length === 0 && Array.isArray(data.exams)) list.push(...data.exams);
   716→      // ابحث: إزالة التكرار بناءً على id
   717→      const seen = new Set<string>();
   718→      const deduped = list.filter((e) => {
   719→        if (seen.has(e.id)) return false;
   720→        seen.add(e.id);
   721→        return true;
   722→      });
   723→      setExams(deduped);
   724→    } catch (e) {
   725→      setError('تعذّر الاتصال بالخادم');
   726→      setExams([]);
   727→    } finally {
   728→      setLoading(false);
   729→    }
   730→  }, [student]);
   731→
   732→  const loadStats = useCallback(async () => {
   733→    setStatsLoading(true);
   734→    try {
   735→      const res = await fetch(buildUrl('/api/exams/student-stats', student), {
   736→        headers: { 'x-student-id': student.studentId },
   737→      });
   738→      const data = await res.json();
   739→      if (res.ok && data.success) {
   740→        setStats(data as StudentStats);
   741→      } else {
   742→        setStats(null);
   743→      }
   744→    } catch (e) {
   745→      setStats(null);
   746→    } finally {
   747→      setStatsLoading(false);
   748→    }
   749→  }, [student]);
   750→
   751→  useEffect(() => {
   752→    let cancelled = false;
   753→    (async () => {
   754→      await Promise.all([loadExams(), loadStats()]);
   755→      if (cancelled) return;
   756→    })();
   757→    return () => { cancelled = true; };
   758→  }, []);
   759→
   760→  // قائمة المواد المتاحة للفلترة (ضمن التبويب الحالي)
   761→  const examsInCategory = useMemo(
   762→    () => exams.filter((e) => (isTrainingExam(e) ? 'TRAINING' : 'OFFICIAL') === categoryTab),
   763→    [exams, categoryTab],
   764→  );
   765→
   766→  const availableSubjects = useMemo(() => {
   767→    const set = new Set<string>();
   768→    examsInCategory.forEach((e) => { if (e.subject) set.add(e.subject); });
   769→    stats?.subjectBreakdown.forEach((s) => set.add(s.subject));
   770→    return Array.from(set).sort();
   771→  }, [examsInCategory, stats]);
   772→
   773→  // أعداد كل تبويب رئيسي
   774→  const officialCount = useMemo(
   775→    () => exams.filter((e) => !isTrainingExam(e)).length,
   776→    [exams],
   777→  );
   778→  const trainingCount = useMemo(
   779→    () => exams.filter((e) => isTrainingExam(e)).length,
   780→    [exams],
   781→  );
   782→
   783→  // تطبيق الفلترة على امتحانات التبويب الحالي
   784→  const filteredExams = useMemo(() => {
   785→    return examsInCategory.filter((exam) => {
   786→      if (searchTerm.trim()) {
   787→        const q = searchTerm.trim().toLowerCase();
   788→        const matches =
   789→          exam.title.toLowerCase().includes(q) ||
   790→          exam.subject.toLowerCase().includes(q) ||
   791→          (exam.teacherName || '').toLowerCase().includes(q);
   792→        if (!matches) return false;
   793→      }
   794→      if (subjectFilter !== 'all' && exam.subject !== subjectFilter) return false;
   795→      if (statusFilter !== 'all' && exam.timeStatus !== statusFilter) return false;
   796→      return true;
   797→    });
   798→  }, [examsInCategory, searchTerm, subjectFilter, statusFilter]);
   799→
   800→  // ===== شاشة التحميل =====
   801→  if (loading) {
   802→    return (
   803→      <div className="flex flex-col items-center justify-center py-20">
   804→        <Loader2 className="w-10 h-10 text-[#610000] animate-spin mb-3" />
   805→        <p className="text-gray-500 text-sm">جاري تحميل الامتحانات...</p>
   806→      </div>
   807→    );
   808→  }
   809→
   810→  // ===== شاشة الخطأ =====
   811→  if (error) {
   812→    return (
   813→      <div className="max-w-md mx-auto py-12">
   814→        <Alert variant="destructive">
   815→          <AlertCircle className="w-4 h-4" />
   816→          <AlertTitle>خطأ</AlertTitle>
   817→          <AlertDescription>{error}</AlertDescription>
   818→        </Alert>
   819→        <Button onClick={loadExams} variant="outline" className="w-full mt-4">
   820→          <RefreshCw className="w-4 h-4 ml-2" /> إعادة المحاولة
   821→        </Button>
   822→      </div>
   823→    );
   824→  }
   825→
   826→  return (
   827→    <div className="space-y-5">
   828→      {/* ====== رأس إحصائيات الطالب ====== */}
   829→      <StudentStatsHeader stats={stats} loading={statsLoading} studentName={student.studentName} />
   830→
   831→      {/* ====== منحنى أدائي (placeholder — الـ API يُبنى في Task ID 6) ====== */}
   832→      <PerformanceCurve student={student} />
   833→
   834→      {/* ====== تبويب: متاحة / منجزة ====== */}
   835→      <div className="flex items-center gap-1 p-1 bg-gray-100 rounded-lg w-fit">
   836→        <button
   837→          onClick={() => setActiveTab('available')}
   838→          className={`flex items-center gap-1.5 px-4 py-2 rounded-md text-sm font-medium transition-all ${
   839→            activeTab === 'available'
   840→              ? 'bg-white text-[#610000] shadow-sm'
   841→              : 'text-gray-600 hover:text-gray-900'
   842→          }`}
   843→        >
   844→          <FileText className="w-4 h-4" />
   845→          المتاحة
   846→          {exams.length > 0 && (
   847→            <Badge variant="secondary" className="text-xs px-1.5 py-0">{exams.length}</Badge>
   848→          )}
   849→        </button>
   850→        <button
   851→          onClick={() => setActiveTab('history')}
   852→          className={`flex items-center gap-1.5 px-4 py-2 rounded-md text-sm font-medium transition-all ${
   853→            activeTab === 'history'
   854→              ? 'bg-white text-[#610000] shadow-sm'
   855→              : 'text-gray-600 hover:text-gray-900'
   856→          }`}
   857→        >
   858→          <History className="w-4 h-4" />
   859→          النتائج السابقة
   860→          {stats && !stats.isEmpty && stats.kpis.totalExams > 0 && (
   861→            <Badge variant="secondary" className="text-xs px-1.5 py-0">{stats.kpis.totalExams}</Badge>
   862→          )}
   863→        </button>
   864→      </div>
   865→
   866→      {activeTab === 'available' ? (
   867→        <>
   868→          {/* ====== التبويب الرئيسي: رسمي / تدريبي ====== */}
   869→          <Tabs value={categoryTab} onValueChange={(v) => setCategoryTab(v as ExamCategory)} className="w-full">
   870→            <TabsList className="h-10">
   871→              <TabsTrigger value="OFFICIAL" className="px-4 gap-1.5">
   872→                <ShieldCheck className="w-4 h-4 text-violet-600" />
   873→                <span>الامتحانات الرسمية</span>

   875→                  <Badge variant="secondary" className="text-xs px-1.5 py-0 ml-1">{officialCount}</Badge>

   981→        <HistoryTab stats={stats} loading={statsLoading} onRetry={loadStats} />

  1276→// يدعم شكل Task 6 (timeline) وشكل مبسّط (points).
  1277→function normalizePerformanceData(raw: PerformanceData | undefined | null): PerformanceData {
  1278→  if (!raw || typeof raw !== 'object') return { isEmpty: true, points: [] };
  1279→
  1280→  // شكل مبسّط: points + avgScore
  1281→  if (Array.isArray((raw as PerformanceData).points)) {
  1282→    return raw as PerformanceData;
  1283→  }
  1284→
  1285→  // شكل Task 6: timeline + stats
  1286→  const r = raw as PerformanceData & {
  1287→    timeline?: Array<{
  1288→      date?: string;
  1289→      examTitle?: string;
  1290→      subject?: string;
  1291→      percentage?: number;
  1292→      attemptNumber?: number;
  1293→    }>;
  1294→    stats?: {
  1295→      avgTraining?: number | null;
  1296→      avgOfficial?: number | null;
  1297→      bestScore?: number | null;
  1298→      totalAttempts?: number;
  1299→    };
  1300→  };
  1301→
  1302→  const timeline = Array.isArray(r.timeline) ? r.timeline : [];
  1303→  if (timeline.length === 0) return { isEmpty: true, points: [] };
  1304→
  1305→  const points: PerformancePoint[] = timeline.map((t, i) => ({
  1306→    label: `#${i + 1}`,
  1307→    score: typeof t.percentage === 'number' ? t.percentage : 0,
  1308→    date: t.date,
  1309→    examTitle: t.examTitle,
  1310→    subject: t.subject,
  1311→  }));
  1312→
  1313→  // متوسط = متوسط كل النسب
  1314→  const validScores = timeline
  1315→    .map((t) => t.percentage)
  1316→    .filter((p): p is number => typeof p === 'number');
  1317→  const avgScore = validScores.length > 0
  1318→    ? Math.round((validScores.reduce((a, b) => a + b, 0) / validScores.length) * 10) / 10
  1319→    : undefined;
  1320→
  1321→  return {
  1322→    isEmpty: false,
  1323→    points,
  1324→    avgScore,
  1325→    bestScore: r.stats?.bestScore ?? undefined,
  1326→    totalExams: r.stats?.totalAttempts,
  1327→  };
  1328→}
