   227→function formatArabicDate(iso: string): string {
   228→  try {
   229→    const d = new Date(iso);
   230→    if (isNaN(d.getTime())) return '—';
   231→    return d.toLocaleDateString('ar-EG', {
   232→      year: 'numeric',
   233→      month: 'long',
   234→      day: 'numeric',
   235→      hour: '2-digit',
   236→      minute: '2-digit',
   237→    });
   238→  } catch {
   239→    return '—';
   240→  }
   241→}

   249→      month: 'short',

   351→export default function TeacherExamsPage({
   352→  onBack,
   353→  schoolId,
   354→  teacherId,
   355→  teacherName,
   356→}: TeacherExamsPageProps) {
   357→  // ===== State =====
   358→  const [fadeIn, setFadeIn] = useState(false);
   359→  const [exams, setExams] = useState<ExamListItem[]>([]);
   360→  const [loading, setLoading] = useState(false);
   361→  const [error, setError] = useState<string | null>(null);
   362→  const [search, setSearch] = useState('');
   363→  const [debouncedSearch, setDebouncedSearch] = useState('');
   364→  const [statusFilter, setStatusFilter] = useState<'all' | ExamStatus>('all');
   365→
   366→  // Detail modal state
   367→  const [detailOpen, setDetailOpen] = useState(false);
   368→  const [detailLoading, setDetailLoading] = useState(false);
   369→  const [detailError, setDetailError] = useState<string | null>(null);
   370→  const [detailExam, setDetailExam] = useState<ExamDetail | null>(null);
   371→  const [submissionsStats, setSubmissionsStats] = useState<SubmissionsStats | null>(null);
   372→
   373→  // Create dialog state
   374→  const [showCreateDialog, setShowCreateDialog] = useState(false);
   375→  const [createSubmitting, setCreateSubmitting] = useState(false);
   376→  const [createForm, setCreateForm] = useState<CreateExamForm>(makeEmptyCreateForm());
   377→  const [createErrors, setCreateErrors] = useState<Record<string, string>>({});
   378→
   379→  // Mounted ref (prevent setState after unmount)
   380→  const mountedRef = useRef(true);
   381→
   382→  // ===== Restore fadeIn =====
   383→  useEffect(() => {
   384→    const t = setTimeout(() => setFadeIn(true), 0);
   385→    return () => clearTimeout(t);
   386→  }, []);
   387→
   388→  // ===== mountedRef lifecycle =====
   389→  useEffect(() => {
   390→    mountedRef.current = true;
   391→    return () => {
   392→      mountedRef.current = false;
   393→    };
   394→  }, []);
   395→
   396→  // ===== Debounced search (300ms) =====
   397→  useEffect(() => {
   398→    const t = setTimeout(() => {
   399→      if (mountedRef.current) setDebouncedSearch(search);
   400→    }, 300);
   401→    return () => clearTimeout(t);
   402→  }, [search]);
   403→
   404→  // ===== Fetch exams =====
   405→  const fetchExams = useCallback(async (silent: boolean = false) => {
   406→    if (!mountedRef.current) return;
   407→    if (!silent) {
   408→      const t = setTimeout(() => setLoading(true), 0);
   409→      void t;
   410→    }
   411→    if (!silent) {
   412→      const t = setTimeout(() => setError(null), 0);
   413→      void t;
   414→    }
   415→    try {
   416→      const params = new URLSearchParams();
   417→      params.set('schoolId', schoolId);
   418→      params.set('teacherId', teacherId);
   419→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim());
   420→      if (statusFilter !== 'all') params.set('status', statusFilter);
   421→      params.set('limit', '100');
   422→
   423→      const res = await fetch(`/api/exams/teacher?${params.toString()}`);
   424→      const data = await res.json();
   425→      if (!res.ok) {
   426→        throw new Error(data?.error || 'فشل جلب الامتحانات');
   427→      }
   428→      if (!mountedRef.current) return;
   429→      const list = (data as ExamsListResponse).exams || [];
   430→      const t = setTimeout(() => setExams(list), 0);
   431→      void t;
   432→    } catch (err) {
   433→      if (!mountedRef.current) return;
   434→      const msg = err instanceof Error ? err.message : 'فشل جلب الامتحانات';
   435→      const t = setTimeout(() => setError(msg), 0);
   436→      void t;
   437→      if (!silent) toast.error(msg);
   438→    } finally {
   439→      if (!mountedRef.current) return;
   440→      const t = setTimeout(() => setLoading(false), 0);
   441→      void t;
   442→    }
   443→  }, [schoolId, teacherId, debouncedSearch, statusFilter]);
   444→
   445→  // ===== Initial fetch + auto-refresh + focus =====
   446→  useEffect(() => {
   447→    fetchExams(false);
   448→    const interval = setInterval(() => {
   449→      fetchExams(true);
   450→    }, 60_000);
   451→    const onFocus = () => fetchExams(true);
   452→    window.addEventListener('focus', onFocus);
   453→    return () => {
   454→      clearInterval(interval);
   455→      window.removeEventListener('focus', onFocus);
   456→    };
   457→  }, [debouncedSearch, statusFilter, schoolId, teacherId]);
   458→
   459→  // ===== Fetch exam detail =====
   460→  const openExamDetail = useCallback(async (examId: string) => {
   461→    if (!mountedRef.current) return;
   462→    const t1 = setTimeout(() => {
   463→      setDetailOpen(true);
   464→      setDetailLoading(true);
   465→      setDetailError(null);
   466→      setDetailExam(null);
   467→      setSubmissionsStats(null);
   468→    }, 0);
   469→    void t1;
   470→    try {
   471→      const params = new URLSearchParams();
   472→      params.set('schoolId', schoolId);
   473→      params.set('teacherId', teacherId);
   474→
   475→      const [examRes, subsRes] = await Promise.allSettled([
   476→        fetch(`/api/exams/teacher/${examId}?${params.toString()}`),
   477→        fetch(`/api/exams/teacher/${examId}/submissions?${params.toString()}&limit=1`),
   478→      ]);
   479→
   480→      if (!mountedRef.current) return;
   481→
   482→      if (examRes.status === 'fulfilled') {
   483→        const examData = await examRes.value.json();
   484→        if (!examRes.value.ok) {
   485→          throw new Error(examData?.error || 'فشل جلب تفاصيل الامتحان');
   486→        }
   487→        const t = setTimeout(() => setDetailExam((examData as ExamDetailResponse).exam), 0);
   488→        void t;
   489→      } else {
   490→        throw new Error('فشل الاتصال بالخادم');
   491→      }
   492→
   493→      if (subsRes.status === 'fulfilled') {
   494→        const subsData = await subsRes.value.json();
   495→        if (subsRes.value.ok && subsData?.stats) {
   496→          const t = setTimeout(() => setSubmissionsStats(subsData.stats as SubmissionsStats), 0);
   497→          void t;
   498→        }
   499→      }
   500→    } catch (err) {
   501→      if (!mountedRef.current) return;
   502→      const msg = err instanceof Error ? err.message : 'فشل جلب تفاصيل الامتحان';
   503→      const t = setTimeout(() => setDetailError(msg), 0);
   504→      void t;
   505→      toast.error(msg);
   506→    } finally {
   507→      if (!mountedRef.current) return;
   508→      const t = setTimeout(() => setDetailLoading(false), 0);
   509→      void t;
   510→    }
   511→  }, [schoolId, teacherId]);
   512→
   513→  // ===== Create exam handlers =====
   514→  const updateCreateField = useCallback((patch: Partial<CreateExamForm>) => {
   515→    setCreateForm((prev) => ({ ...prev, ...patch }));
   516→  }, []);
   517→
   518→  const addQuestion = useCallback(() => {
   519→    setCreateForm((prev) => ({
   520→      ...prev,
   521→      questions: [...prev.questions, makeEmptyQuestion(prev.questions.length)],
   522→    }));
   523→  }, []);
   524→
   525→  const updateQuestion = useCallback((qid: string, patch: Partial<DraftQuestion>) => {
   526→    setCreateForm((prev) => ({
   527→      ...prev,
   528→      questions: prev.questions.map((q) => (q.id === qid ? { ...q, ...patch } : q)),
   529→    }));
   530→  }, []);
   531→
   532→  const removeQuestion = useCallback((qid: string) => {
   533→    setCreateForm((prev) => ({
   534→      ...prev,
   535→      questions: prev.questions.filter((q) => q.id !== qid),
   536→    }));
   537→  }, []);
   538→
   539→  const addOption = useCallback((qid: string) => {
   540→    setCreateForm((prev) => ({
   541→      ...prev,
   542→      questions: prev.questions.map((q) => {
   543→        if (q.id !== qid || q.options.length >= 8) return q;
   544→        return { ...q, options: [...q.options, ''] };
   545→      }),
   546→    }));
   547→  }, []);
   548→
   549→  const updateOption = useCallback((qid: string, optIndex: number, value: string) => {
   550→    setCreateForm((prev) => ({
   551→      ...prev,
   552→      questions: prev.questions.map((q) => {
   553→        if (q.id !== qid) return q;
   554→        const newOptions = [...q.options];
   555→        newOptions[optIndex] = value;
   556→        let newCorrect = q.correctAnswer;
   557→        if (q.correctAnswer === q.options[optIndex]) {
   558→          newCorrect = value;
   559→        }
   560→        return { ...q, options: newOptions, correctAnswer: newCorrect };
   561→      }),
   562→    }));
   563→  }, []);
   564→
   565→  const removeOption = useCallback((qid: string, optIndex: number) => {
   566→    setCreateForm((prev) => ({
   567→      ...prev,
   568→      questions: prev.questions.map((q) => {
   569→        if (q.id !== qid || q.options.length <= 2) return q;
   570→        const removedValue = q.options[optIndex];
   571→        const newOptions = q.options.filter((_, i) => i !== optIndex);
   572→        let newCorrect = q.correctAnswer;
   573→        if (q.correctAnswer === removedValue) newCorrect = '';
