   106→function HomePage() {
   107→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated } = useAdminStore()
   108→  const searchParams = useSearchParams()
   109→  const router = useRouter()
   110→  const pathname = usePathname()
   111→  const urlSubdomain = searchParams.get('school')
   112→  const hasAppliedUrlSchool = useRef(false)
   113→
   114→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   115→  const [news, setNews] = useState<NewsItem[]>([])
   116→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   117→  const [teachers, setTeachers] = useState<Teacher[]>([])
   118→  const [sliders, setSliders] = useState<SliderItem[]>([])
   119→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   120→  const [loading, setLoading] = useState(true)
   121→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
   122→  const [servicesOpen, setServicesOpen] = useState(false)
   123→  const [showAdminLogin, setShowAdminLogin] = useState(false)
   124→  const [showResultsPage, setShowResultsPage] = useState(false)
   125→  const [showStudentLife, setShowStudentLife] = useState(false)
   126→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   127→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   128→  const [showParentsPage, setShowParentsPage] = useState(false)
   129→  const [showTeacherExamsPage, setShowTeacherExamsPage] = useState(false)
   130→  const [showStudentExamsPage, setShowStudentExamsPage] = useState(false)
   131→  const [showParentExamsPage, setShowParentExamsPage] = useState(false)
   132→  const [showScrollTop, setShowScrollTop] = useState(false)
   133→  const [currentSlide, setCurrentSlide] = useState(0)
   134→  const logoClickCount = useRef(0)
   135→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   136→
   137→  const hydrated = _hasHydrated
   138→
   139→  // Resolve school from subdomain only (not from store's school list)
   140→  useEffect(() => {
   141→    if (!hydrated) return
   142→    let cancelled = false
   143→    async function resolveSchool() {
   144→      try {
   145→        if (urlSubdomain) {
   146→          // If subdomain is in URL, fetch by subdomain
   147→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   148→          if (res.ok && !cancelled) {
   149→            const data = await res.json()
   150→            if (data.school) {
   151→              setSchoolData(data)
   152→              setSelectedSchoolId(data.school.id)
   153→              return
   154→            }
   155→          }
   156→        }
   157→        // No subdomain in URL — fetch first available school
   158→        const schoolsRes = await fetch('/api/schools')
   159→        if (schoolsRes.ok && !cancelled) {
   160→          const schoolsList = await schoolsRes.json()
   161→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   162→            setSchools(schoolsList)
   163→            const firstSchool = schoolsList[0]
   164→            // Fetch full data for the first school
   165→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   166→            if (res.ok && !cancelled) {
   167→              const data = await res.json()
   168→              if (data.school) {
   169→                setSchoolData(data)
   170→                setSelectedSchoolId(data.school.id)
   171→              }
   172→            }
   173→          }
   174→        }
   175→      } catch {
   176→        // fallback to default data
   177→      }
   178→    }
   179→    resolveSchool()
   180→    return () => { cancelled = true }
   181→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   182→
   183→  // Fetch content for selected school
   184→  useEffect(() => {
   185→    if (!hydrated || !selectedSchoolId) return
   186→    let cancelled = false
   187→    const sid = selectedSchoolId
   188→    async function loadData() {
   189→      setLoading(true)
   190→      try {
   191→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   192→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   193→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   194→          fetch(`/api/teachers?schoolId=${sid}`),
   195→          fetch(`/api/sliders?schoolId=${sid}`),
   196→          fetch(`/api/custom-sections?schoolId=${sid}`),
   197→        ])
   198→        if (cancelled) return
   199→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   200→          const d = await newsRes.value.json()
   201→          if (Array.isArray(d) && !cancelled) setNews(d)
   202→        }
   203→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   204→          const d = await galleryRes.value.json()
   205→          if (Array.isArray(d) && !cancelled) setGallery(d)
   206→        }
   207→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   208→          const d = await teachersRes.value.json()
   209→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   210→        }
   211→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   212→          const d = await slidersRes.value.json()
   213→          if (Array.isArray(d) && !cancelled) setSliders(d)
   214→        }
   215→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   216→          const d = await csRes.value.json()
   217→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   218→        }
   219→      } catch {
   220→        // silently fail
   221→      } finally {
   222→        if (!cancelled) setLoading(false)
   223→      }
   224→    }
   225→    loadData()
   226→    return () => { cancelled = true }
   227→  }, [selectedSchoolId, hydrated])
   228→
   229→  // Auto-rotate slider
   230→  const activeSliders = sliders.filter(s => s.active)
   231→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   232→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   233→  useEffect(() => {
   234→    if (slideItems.length <= 1) return
   235→    const timer = setInterval(() => setCurrentSlide(i => (i + 1) % slideItems.length), 5000)
   236→    return () => clearInterval(timer)
   237→  }, [slideItems.length])
   238→
   239→  // Scroll to top
   240→  useEffect(() => {
   241→    const h = () => setShowScrollTop(window.scrollY > 500)
   242→    window.addEventListener('scroll', h)
   243→    return () => window.removeEventListener('scroll', h)
   244→  }, [])
   245→
   246→  const handleLogoClick = useCallback(() => {
   247→    logoClickCount.current += 1
   248→    if (logoClickTimer.current) clearTimeout(logoClickTimer.current)
   249→    logoClickTimer.current = setTimeout(() => { logoClickCount.current = 0 }, 3000)
   250→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   251→  }, [])
   252→
   253→  const handleServiceAction = (action: string) => {
   254→    setServicesOpen(false); setMobileMenuOpen(false)
   255→    if (action === 'results') setShowResultsPage(true)
   256→    if (action === 'schedules') setShowSchedulesPage(true)
   257→    if (action === 'library') setShowLibraryPage(true)
   258→    if (action === 'parents') setShowParentsPage(true)
   259→    if (action === 'teacher-exams') setShowTeacherExamsPage(true)
   260→    if (action === 'student-exams') setShowStudentExamsPage(true)
   261→    if (action === 'parent-exams') setShowParentExamsPage(true)
   262→  }
   263→
   264→  const school = schoolData?.school || defaultSchoolData.school
   265→  const settings = schoolData?.settings || defaultSchoolData.settings
   266→  const stats = schoolData?.stats || defaultSchoolData.stats
   267→  const alertNews = news.filter(n => n.category === 'تنبيه')
   268→
   269→  // Build ticker text for seamless scrolling
   270→  const tickerText = news.map(item => item.title).join('     ◆     ')
   271→
   272→  // Sub-page rendering
   273→  if (isAdminMode) return <AdminLayout />
   274→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   275→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   276→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   277→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />
   278→  if (showParentsPage) return <ParentsPortalPage onBack={() => setShowParentsPage(false)} schoolId={selectedSchoolId} />
   279→  if (showTeacherExamsPage) return <TeacherExamsPage onBack={() => setShowTeacherExamsPage(false)} schoolId={selectedSchoolId} />
   280→  if (showStudentExamsPage) return <StudentExamsPage onBack={() => setShowStudentExamsPage(false)} schoolId={selectedSchoolId} />
   281→  if (showParentExamsPage) return <ParentExamsPage onBack={() => setShowParentExamsPage(false)} schoolId={selectedSchoolId} />
   282→
   283→  if (!hydrated) {
   284→    return (
   285→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   286→        <div className="text-center">
   287→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />
   288→          <p className="text-gray-500 text-sm">جاري التحميل...</p>
   289→        </div>
   290→      </div>
   291→    )
   292→  }
   293→
   294→  return (
   295→    <div className="min-h-screen flex flex-col" dir="rtl">
   296→
   297→      {/* ===== STICKY TOP ===== */}
   298→      <div className="sticky top-0 z-50 shadow-lg">
   299→        {/* Top Bar - Contact info only, NO school switcher for visitors */}
   300→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">
   301→          <div className="max-w-[1280px] mx-auto px-4 py-1.5 flex items-center justify-between">
   302→            <div className="flex items-center gap-4">
   303→              {school.phone && <a href={`tel:${school.phone}`} className="hover:opacity-80 flex items-center gap-1">📞 {school.phone}</a>}
   304→              {school.email && <a href={`mailto:${school.email}`} className="hover:opacity-80 flex items-center gap-1">✉️ {school.email}</a>}
   305→            </div>
   306→            <div className="flex items-center gap-2">
   307→              {school.facebookUrl && (
   308→                <a href={school.facebookUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📘</a>
   309→              )}
   310→              {settings?.youtubeUrl && (
   311→                <a href={settings.youtubeUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📺</a>
   312→              )}
   313→            </div>
   314→          </div>
   315→        </div>
   316→
   317→        {/* Header */}
   318→        <header className="bg-white shadow-sm">
   319→          <div className="max-w-[1280px] mx-auto px-4 py-2 flex items-center gap-3">
   320→            <div onClick={handleLogoClick} className="cursor-pointer shrink-0" title="اضغط 5 مرات للإدارة">
   321→              {school.logoUrl ? (
   322→                <img src={school.logoUrl} alt={school.name} className="w-12 h-12 rounded-full object-cover border-2" style={{ borderColor: school.primaryColor }} />
   323→              ) : (
   324→                <div className="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl" style={{ backgroundColor: school.primaryColor }}>🏫</div>
   325→              )}
   326→            </div>
   327→            <div className="flex-1 min-w-0">
   328→              <h1 className="text-base md:text-xl font-bold truncate" style={{ color: school.primaryColor }}>{school.name}</h1>
   329→              {settings?.heroSubtitle && <p className="text-xs text-gray-500 truncate">{settings.heroSubtitle}</p>}
   330→            </div>
   331→            <button className="lg:hidden p-2 text-2xl text-gray-600" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   332→              {mobileMenuOpen ? '✕' : '☰'}
   333→            </button>
   334→          </div>
   335→        </header>
   336→
   337→        {/* Navigation */}
   338→        <nav style={{ backgroundColor: '#1a2332' }} className="text-white shadow-lg">
   339→          <div className="max-w-[1280px] mx-auto px-4">
   340→            {/* Desktop Nav */}
   341→            <div className="hidden lg:flex items-center h-11">
   342→              {navLinks.map((link) => (

   352→                            className="w-full flex items-center gap-3 px-4 py-3 text-sm text-gray-700 hover:bg-gray-50 transition-colors">
   353→                            <span className="text-lg">{item.icon}</span> {item.label}
   354→                          </button>
   355→                        ))}
   356→                      </div>
   357→                    )}
   358→                  </div>
   359→                ) : (
   360→                  <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   361→                    onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true) } : undefined}
   362→                    className="px-5 h-full text-sm font-medium hover:bg-white/10 flex items-center transition-colors">
   363→                    {link.label}
   364→                  </a>
   365→                )
   366→              ))}
   367→            </div>
   368→            {/* Mobile Nav */}
   369→            {mobileMenuOpen && (
   370→              <div className="lg:hidden py-2 border-t border-white/10">
   371→                {navLinks.map((link) => (
   372→                  link.isServices ? (
   373→                    <div key={link.href}>
   374→                      <div className="px-4 py-2 text-sm font-medium text-white/70">{link.label}</div>
   375→                      {serviceItems.map(item => (
   376→                        <button key={item.action} onClick={() => handleServiceAction(item.action)}
   377→                          className="w-full flex items-center gap-2 px-8 py-2 text-sm text-white/80 hover:bg-white/10">
   378→                          <span>{item.icon}</span> {item.label}
   379→                        </button>
   380→                      ))}
   381→                    </div>
   382→                  ) : (
   383→                    <a key={link.href} href={link.isStudentLife ? undefined : link.href}
   384→                      onClick={link.isStudentLife ? (e) => { e.preventDefault(); setShowStudentLife(true); setMobileMenuOpen(false) } : () => setMobileMenuOpen(false)}
   385→                      className="block px-4 py-2 text-sm hover:bg-white/10">{link.label}</a>

   388→              </div>

   390→          </div>

   392→
   393→        {/* News Ticker - Seamless scrolling */}
   394→        {settings?.showNewsTicker && news.length > 0 && (
   395→          <div className="bg-gray-100 border-b overflow-hidden">
   396→            <div className="flex items-center">
   397→              <span className="bg-red-600 text-white px-4 py-2 text-xs font-bold shrink-0 flex items-center gap-1 z-10">
   398→                🔴 عاجل
   399→              </span>
   400→              <div className="overflow-hidden flex-1 relative">
   401→                <div className="animate-news-ticker-scroll whitespace-nowrap py-2">
   402→                  {/* Duplicate content for seamless loop */}
   403→                  <span className="inline-block text-sm mx-4">
   404→                    {tickerText}
   405→                  </span>
   406→                  <span className="inline-block text-sm mx-4">
   407→                    {tickerText}
   408→                  </span>
   409→                </div>
   410→              </div>
   411→            </div>
   412→          </div>
   413→        )}
   414→      </div>
   415→
   416→      {/* ===== MAIN CONTENT ===== */}
   417→      <main className="flex-1">
   418→        {/* Hero Slider */}
   419→        {(settings?.showSlider ?? true) && (
   420→          <section className="max-w-[1280px] mx-auto px-4 py-6">
