    77→function timeAgo(dateStr: string): string {
    78→  const date = new Date(dateStr);
    79→  const diff = Math.max(0, Date.now() - date.getTime());
    80→  const mins = Math.floor(diff / 60000);
    81→  if (mins < 1) return 'الآن';
    82→  if (mins < 60) return `قبل ${mins} دقيقة`;
    83→  const hours = Math.floor(mins / 60);
    84→  if (hours < 24) return `قبل ${hours} ساعة`;
    85→  const days = Math.floor(hours / 24);
    86→  if (days < 7) return `قبل ${days} يوم`;
    87→  const weeks = Math.floor(days / 7);
    88→  if (weeks < 4) return `قبل ${weeks} أسبوع`;
    89→  return date.toLocaleDateString('ar-EG', { day: 'numeric', month: 'short' });
    90→}

   868→    if (!parentPhone.trim()) {
   869→      setError('هاتف ولي الأمر مطلوب');

   871→    }
   872→    if (!schoolIdInput.trim()) {
   873→      setError('معرّف المدرسة مطلوب');

   877→    try {

   889→        setError(data.error || 'فشل تسجيل الدخول');

   891→        return;
   892→      }
   893→      onLogin({
   894→        parentName: data.parentName,
   895→        parentPhone: data.parentPhone,
   896→        children: data.children,

   898→      });
   899→    } catch {
   900→      setError('تعذر الاتصال بالخادم. حاول مرة أخرى');

   902→    } finally {

   904→    }
   905→  };
   906→
   907→  return (

   933→                  autoComplete="off"
   934→                  dir="ltr"
   935→                />
   936→              </div>
   937→            </div>
   938→            <div className="space-y-2">
   939→              <Label htmlFor="parentPhone" className="text-sm font-medium text-gray-700">
   940→                هاتف ولي الأمر <span className="text-red-500">*</span>
   941→              </Label>

   943→                <Phone className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />

   945→                  id="parentPhone"
   946→                  value={parentPhone}
   947→                  onChange={(e) => setParentPhone(e.target.value)}
   948→                  placeholder="مثال: 01023456789"
   949→                  className="text-right pr-9"
   950→                  autoComplete="off"
   951→                  dir="ltr"
   952→                />
   953→              </div>
   954→            </div>
   955→            {error && (
   956→              <div className="bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 text-red-700 dark:text-red-400 text-sm rounded-lg p-3 flex items-start gap-2">
   957→                <AlertTriangle className="w-4 h-4 shrink-0 mt-0.5" />
   958→                <span>{error}</span>
   959→              </div>
   960→            )}
   961→            <Button
   962→              type="submit"
   963→              disabled={loading}
   964→              className="w-full bg-gradient-to-r from-emerald-500 to-teal-600 hover:from-emerald-600 hover:to-teal-700 text-white font-bold py-6 text-base shadow-lg hover:shadow-xl transition-all disabled:opacity-50"
   965→            >
   966→              {loading ? (
   967→                <>
   968→                  <RefreshCw className="w-5 h-5 ml-2 animate-spin" />
   969→                  جاري التحقق...
   970→                </>

   972→                <>
   973→                  <LogOut className="w-5 h-5 ml-2 rotate-180" />
   974→                  دخول البوابة
   975→                </>

   977→            </Button>
   978→          </form>
   979→
   980→          {/* Test mode hint */}
   981→          <div className="mt-4 p-3 bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800 rounded-lg">
   982→            <div className="flex items-start gap-2">
   983→              <Sparkles className="w-4 h-4 text-amber-600 shrink-0 mt-0.5" />
   984→              <div className="text-xs text-amber-800 dark:text-amber-300">
   985→                <p className="font-bold mb-1">للتجربة:</p>
