    93→// ===== Fetch helper مع مهلة زمنية + إعادة محاولة تلقائية + رسائل خطأ واضحة =====
    94→// المهلة: 30 ثانية (كافية لاستجابات Supabase البطيئة، لكنها تمنع التعليق الأبدي)
    95→// إعادة المحاولة: مرتان تلقائياً عند فشل الشبكة (NetworkError / 5xx) مع تأخير تصاعدي
    96→const FETCH_TIMEOUT_MS = 30000;
    97→const MAX_RETRIES = 2;
    98→
    99→async function parentFetchSingle<T = any>(
   100→  url: string,
   101→  options?: RequestInit & { headers?: Record<string, string> }
   102→): Promise<{ success: boolean; data?: T; error?: string; status: number; retryable: boolean }> {
   103→  const controller = new AbortController();
   104→  const timer = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);
   105→  try {
   106→    const res = await fetch(url, { ...options, signal: controller.signal });
   107→    // محاولة parse JSON بأمان (قد تفشل إن أعاد الخادم HTML أو نصاً)
   108→    let json: any = null;
   109→    try {
   110→      json = await res.json();
   111→    } catch {
   112→      return {
   113→        success: false,
   114→        error: `استجابة غير صالحة من الخادم (${res.status}). حاول مرة أخرى.`,
   115→        status: res.status,
   116→        retryable: res.status >= 500,
   117→      };
   118→    }
   119→    // أخطاء 5xx من Supabase — قابلة لإعادة المحاولة
   120→    if (res.status >= 500) {
   121→      return {
   122→        success: false,
   123→        error: json.error || `خطأ في الخادم (${res.status}). حاول مرة أخرى.`,
   124→        status: res.status,
   125→        retryable: true,
   126→      };
   127→    }
   128→    return { success: res.ok && json.success, data: json, error: json.error, status: res.status, retryable: false };
   129→  } catch (e: any) {
   130→    if (e?.name === 'AbortError') {
   131→      return {
   132→        success: false,
   133→        error: 'انتهت مهلة الاتصال — الخادم يستغرق وقتاً طويلاً. حاول مرة أخرى.',
   134→        status: 408,
   135→        retryable: true,
   136→      };
   137→    }
   138→    // NetworkError / فشل DNS / انقطاع الاتصال — قابل لإعادة المحاولة
   139→    return {
   140→      success: false,
   141→      error: 'تعذّر الاتصال بالخادم — تحقق من اتصالك وحاول مجدداً.',
   142→      status: 0,
   143→      retryable: true,
   144→    };
   145→  } finally {
   146→    clearTimeout(timer);
   147→  }
   148→}
   149→
   150→async function parentFetch<T = any>(
   151→  url: string,
   152→  options?: RequestInit & { headers?: Record<string, string> }
   153→): Promise<{ success: boolean; data?: T; error?: string; status: number }> {
   154→  let lastResult: { success: boolean; data?: T; error?: string; status: number; retryable: boolean } | null = null;
   155→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   156→    lastResult = await parentFetchSingle<T>(url, options);
   157→    if (lastResult.success || !lastResult.retryable) {
   158→      return lastResult;
   159→    }
   160→    // إعادة المحاولة مع تأخير تصاعدي (800ms ثم 1600ms)
   161→    if (attempt < MAX_RETRIES) {
   162→      await new Promise((r) => setTimeout(r, 800 * (attempt + 1)));
   163→    }
   164→  }
   165→  return lastResult!;
   166→}

   327→function ParentLogin({
   328→  schoolId, onLogin, onBackToSite,
   329→}: {
   330→  schoolId: string;
   331→  onLogin: (p: ParentInfo) => void;
   332→  onBackToSite: () => void;
   333→}) {
   334→  const [parentId, setParentId] = useState('');
   335→  const [parentName, setParentName] = useState('');
   336→  const [loading, setLoading] = useState(false);
   337→  const [error, setError] = useState<string | null>(null);
   338→
   339→  const handleSubmit = async (e: React.FormEvent) => {
   340→    e.preventDefault();
   341→    if (!parentId.trim() || !parentName.trim()) {
   342→      setError('يرجى إدخال معرّف ولي الأمر والاسم');
   343→      return;
   344→    }
   345→    setLoading(true);
   346→    setError(null);
   347→    try {
   348→      // تحقق سريع من API أن ولي الأمر صالح (ونرى عدد الأبناء)
   349→      const url = `/api/exams/parent?schoolId=${encodeURIComponent(schoolId)}&parentId=${encodeURIComponent(parentId.trim())}&parentName=${encodeURIComponent(parentName.trim())}`;
   350→      const result = await parentFetch(url, {
   351→        headers: { 'x-parent-id': parentId.trim(), 'x-parent-name': parentName.trim() },
   352→      });
   353→      if (!result.success) {
   354→        setError(result.error || 'فشل الدخول');
   355→        return;
   356→      }
   357→      onLogin({
   358→        parentId: parentId.trim(),
   359→        parentName: parentName.trim(),
   360→        isTestMode: parentId.trim().startsWith('test-'),
   361→      });
   362→    } catch {
   363→      setError('تعذّر الاتصال بالخادم');
   364→    } finally {
   365→      setLoading(false);
   366→    }
   367→  };
   368→
   369→  return (
   370→    <div className="flex items-center justify-center py-8">
   371→      <Card className="w-full max-w-md border-0 shadow-lg">
   372→        <CardContent className="p-6 sm:p-8">
   373→          <div className="text-center mb-6">
   374→            <div className="w-16 h-16 rounded-2xl flex items-center justify-center text-white mx-auto mb-3 shadow-md" style={{ backgroundColor: PARENT_COLOR }}>

   376→            </div>
   377→            <h2 className="text-xl font-bold text-gray-900">دخول بوابة ولي الأمر</h2>
   378→            <p className="text-sm text-gray-500 mt-1">تابع نتائج وأداء أبنائك في الامتحانات</p>

   381→          <form onSubmit={handleSubmit} className="space-y-4">

   383→              <Label htmlFor="parentId" className="text-sm font-medium">
   384→                معرّف ولي الأمر <span className="text-red-500">*</span>
   385→              </Label>
   386→              <Input
   387→                id="parentId"
   388→                type="text"

   494→function ParentDashboard({
