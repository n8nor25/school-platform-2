    84→    84→function parseTargetIds(raw: string | string[] | null): string[] {
    85→    85→  if (!raw) return []
    86→    86→  if (Array.isArray(raw)) return raw
    87→    87→  try {
    88→    88→    const parsed: unknown = JSON.parse(raw)
    89→    89→    if (Array.isArray(parsed)) {
    90→    90→      return parsed.filter((x): x is string => typeof x === 'string')
    91→    91→    }
    92→    92→    return []
    93→    93→  } catch {
    94→    94→    return []
    95→    95→  }
    96→    96→}
    97→    97→
    98→    98→function extractErrorMessage(data: unknown, fallback: string): string {
    99→    99→  if (data && typeof data === 'object' && 'error' in data) {
   100→   100→    const err = (data as { error?: unknown }).error
   101→   101→    if (typeof err === 'string') return err
   102→   102→  }
   103→   103→  return fallback
   104→   104→}
