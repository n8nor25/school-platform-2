   142→function timeAgo(iso: string): string {
   143→  try {
   144→    const date = new Date(iso);
   145→    const now = Date.now();
   146→    const diff = Math.max(0, now - date.getTime());
   147→    const seconds = Math.floor(diff / 1000);
   148→    const minutes = Math.floor(seconds / 60);
   149→    const hours = Math.floor(minutes / 60);
   150→    const days = Math.floor(hours / 24);
   151→    if (days >= 30) return date.toLocaleDateString('ar-EG', { year: 'numeric', month: 'long', day: 'numeric' });
   152→    if (days >= 1) return `قبل ${days} ${days === 1 ? 'يوم' : days === 2 ? 'يومين' : 'أيام'}`;
   153→    if (hours >= 1) return `قبل ${hours} ${hours === 1 ? 'ساعة' : hours === 2 ? 'ساعتين' : 'ساعات'}`;
   154→    if (minutes >= 1) return `قبل ${minutes} ${minutes === 1 ? 'دقيقة' : 'دقائق'}`;
   155→    return 'الآن';
   156→  } catch {
   157→    return '';
   158→  }
   159→}

   309→function ParentLogin({
   310→  schoolId, onLogin,
   311→}: {
   312→  schoolId: string;
   313→  onLogin: (s: ParentSession) => void;
   314→}) {
   315→  const [studentNumber, setStudentNumber] = useState('');
   316→  const [parentPhone, setParentPhone] = useState('');
   317→  const [loading, setLoading] = useState(false);
   318→  const [error, setError] = useState<string | null>(null);
   319→  const [fadeIn, setFadeIn] = useState(false);
   320→
   321→  useEffect(() => {
   322→    const t = setTimeout(() => setFadeIn(true), 100);
   323→    return () => clearTimeout(t);
   324→  }, []);
   325→
   326→  const handleSubmit = async (e: React.FormEvent) => {
   327→    e.preventDefault();
   328→    if (!studentNumber.trim() || !parentPhone.trim()) {
   329→      setError('يرجى إدخال رقم الطالب وهاتف ولي الأمر');
   330→      return;
   331→    }
   332→    setLoading(true);
   333→    setError(null);
   334→    try {
   335→      const res = await fetch(`/api/parent/login?schoolId=${encodeURIComponent(schoolId)}`, {
   336→        method: 'POST',
   337→        headers: { 'Content-Type': 'application/json' },
   338→        body: JSON.stringify({
   339→          schoolId,
   340→          studentNumber: studentNumber.trim(),
   341→          parentPhone: parentPhone.trim(),
   342→        }),
   343→      });
   344→      const data = await res.json();
   345→      if (!res.ok) {
   346→        const msg = data?.error || (res.status === 404 ? 'رقم الطالب أو هاتف ولي الأمر غير صحيح' : 'تعذّر تسجيل الدخول');
   347→        throw new Error(typeof msg === 'string' ? msg : 'تعذّر تسجيل الدخول');
   348→      }
   349→      if (!data?.children || !Array.isArray(data.children) || data.children.length === 0) {
   350→        throw new Error('لم يتم العثور على أبناء مرتبطين بهذا الحساب');
   351→      }
   352→      const sessionData: ParentSession = {
   353→        parentName: data.parentName || 'ولي الأمر',
   354→        parentPhone: data.parentPhone || parentPhone.trim(),
   355→        children: data.children,
   356→        testMode: !!data.testMode,
   357→        fakeStudent: !!data.fakeStudent,
   358→        originalStudentNumber: data.originalStudentNumber ?? null,
   359→        warning: data.warning ?? null,
   360→      };
   361→      onLogin(sessionData);
   362→      toast.success(`مرحباً ${sessionData.parentName}`);
   363→    } catch (err) {
   364→      const msg = err instanceof Error ? err.message : 'تعذّر تسجيل الدخول';
   365→      setError(msg);
   366→    } finally {
   367→      setLoading(false);
   368→    }
   369→  };
   370→
   371→  return (
   372→    <div className={`max-w-md mx-auto transition-all duration-700 ${fadeIn ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'}`}>
   373→      <Card className="overflow-hidden border-0 shadow-2xl">
   374→        {/* Gradient header */}
   375→        <div className="bg-gradient-to-l from-emerald-600 via-emerald-600 to-teal-600 p-6 text-white relative overflow-hidden">
   376→          <div className="absolute top-0 left-0 w-40 h-40 bg-white/10 rounded-full -translate-x-1/2 -translate-y-1/2" />
   377→          <div className="absolute bottom-0 right-0 w-32 h-32 bg-teal-300/10 rounded-full translate-x-1/4 translate-y-1/4" />
   378→          <div className="relative z-10 text-center">
   379→            <div className="w-16 h-16 mx-auto bg-white/20 rounded-full flex items-center justify-center mb-3 backdrop-blur">
   380→              <Users className="w-9 h-9 text-white" />
   381→            </div>
   382→            <h2 className="text-xl font-bold mb-1">تسجيل دخول ولي الأمر</h2>
   383→            <p className="text-emerald-50 text-sm">أدخل بيانات الطالب للوصول إلى لوحة المتابعة</p>
   384→          </div>
   385→        </div>
   386→
   387→        <CardContent className="p-6 space-y-4">
   388→          <form onSubmit={handleSubmit} className="space-y-4">
   389→            <div className="space-y-1.5">
   390→              <Label htmlFor="studentNumber" className="text-sm font-semibold">
   391→                رقم الطالب
   392→              </Label>
   393→              <Input
   394→                id="studentNumber"
   395→                value={studentNumber}
   396→                onChange={(e) => setStudentNumber(e.target.value)}
   397→                placeholder="مثال: 2024002"
   398→                disabled={loading}
   399→                autoComplete="off"
   400→                inputMode="numeric"
   401→                className="text-base h-11"
   402→              />
   403→            </div>
   404→            <div className="space-y-1.5">
   405→              <Label htmlFor="parentPhone" className="text-sm font-semibold">
   406→                هاتف ولي الأمر
   407→              </Label>
   408→              <Input
   409→                id="parentPhone"
   410→                value={parentPhone}
   411→                onChange={(e) => setParentPhone(e.target.value)}
   412→                placeholder="مثال: 01023456789"
   413→                disabled={loading}
   414→                autoComplete="off"
   415→                inputMode="tel"
   416→                className="text-base h-11"
   417→                dir="ltr"
   418→              />
   419→            </div>
   420→
   421→            {error && (
   422→              <div className="flex items-start gap-2 bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 rounded-lg p-3">
   423→                <AlertTriangle className="w-4 h-4 text-red-600 dark:text-red-400 shrink-0 mt-0.5" />
   424→                <p className="text-sm text-red-700 dark:text-red-300">{error}</p>
   425→              </div>
   426→            )}
   427→
   428→            <Button
   429→              type="submit"
   430→              disabled={loading}
   431→              className="w-full h-11 bg-gradient-to-l from-emerald-600 to-teal-600 hover:from-emerald-700 hover:to-teal-700 text-white font-semibold"
   432→            >
   433→              {loading ? (
   434→                <>
   435→                  <Loader2 className="w-5 h-5 animate-spin" />
   436→                  جاري التحقق...
   437→                </>
   438→              ) : (
   439→                <>
   440→                  <Shield className="w-5 h-5" />
   441→                  دخول البوابة
   442→                </>
   443→              )}
   444→            </Button>
   445→          </form>
   446→
   447→          {/* Test mode hint */}
   448→          <div className="bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800 rounded-lg p-3">
   449→            <div className="flex items-start gap-2">
   450→              <FlaskConical className="w-5 h-5 text-amber-600 dark:text-amber-400 shrink-0 mt-0.5" />
   451→              <div className="text-sm">
   452→                <p className="font-semibold text-amber-800 dark:text-amber-300 mb-1">وضع التجربة</p>
   453→                <p className="text-amber-700 dark:text-amber-400 text-xs leading-relaxed">
   454→                  لإضافة بادئة <code className="bg-amber-100 dark:bg-amber-900/40 px-1 rounded font-mono">test-</code> قبل رقم الطالب
   455→                  لعرض بيانات تجريبية. مثلاً:
   456→                </p>
   457→                <div className="mt-2 flex flex-wrap gap-1.5">
   458→                  <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-200 border-amber-200 dark:border-amber-800 font-mono">
   459→                    2024002 / 01023456789
   460→                  </Badge>
   461→                  <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-200 border-amber-200 dark:border-amber-800 font-mono">
   462→                    test-001
   463→                  </Badge>
   464→                </div>
   465→              </div>
   466→            </div>
   467→          </div>
   468→        </CardContent>
   469→      </Card>
   470→    </div>
   471→  );
   472→}
