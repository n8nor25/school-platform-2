     2→
     3→import React, { useState, useEffect } from 'react'
     4→import {
     5→  Bus, Plus, Edit, Trash2, Search, Phone, ShieldCheck,
     6→  User, Users, Route, AlertCircle, X, CheckCircle2,
     7→  XCircle, MapPin, Truck, FileText
     8→} from 'lucide-react'

    13→import { Badge } from '@/components/ui/badge'
    14→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from '@/components/ui/dialog'
    15→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    16→import { Textarea } from '@/components/ui/textarea'
    17→import { Skeleton } from '@/components/ui/skeleton'
    18→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    19→import { toast } from 'sonner'

    50→  plateNumber: string
    51→  driverName: string
    52→  driverPhone: string
    53→  driverLicense: string
    54→  supervisorName: string
    55→  supervisorPhone: string
    56→  capacity: number
    57→  model: string
    58→  color: string
    59→  active: boolean
    60→  notes: string
    61→}
    62→
    63→const defaultForm: BusForm = {
    64→  plateNumber: '',
    65→  driverName: '',
    66→  driverPhone: '',
    67→  driverLicense: '',
    68→  supervisorName: '',
    69→  supervisorPhone: '',
    70→  capacity: 30,
    71→  model: '',
    72→  color: '#610000',
    73→  active: true,
    74→  notes: '',
    75→}
    76→
    77→const predefinedColors = [
    78→  { label: 'أحمر', value: '#EF4444' },
    79→  { label: 'أزرق', value: '#3B82F6' },
    80→  { label: 'أخضر', value: '#22C55E' },
    81→  { label: 'أصفر', value: '#EAB308' },
    82→  { label: 'برتقالي', value: '#F97316' },
    83→  { label: 'أبيض', value: '#F8FAFC' },
    84→  { label: 'رمادي', value: '#6B7280' },
    85→  { label: 'أسود', value: '#1F2937' },
    86→]
    87→
    88→export function BusFleetManagement() {
    89→  const { selectedSchoolId } = useAdminStore()
    90→
    91→  // List state
    92→  const [buses, setBuses] = useState<Bus[]>([])
    93→  const [loading, setLoading] = useState(true)
    94→  const [searchQuery, setSearchQuery] = useState('')
    95→  const [activeFilter, setActiveFilter] = useState<string>('all')
    96→
    97→  // Tab state
    98→  const [activeTab, setActiveTab] = useState('fleet')
    99→
   100→  // Edit dialog state
   101→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   102→  const [editId, setEditId] = useState<string | null>(null)
   103→  const [editForm, setEditForm] = useState<BusForm>(defaultForm)
   104→  const [saving, setSaving] = useState(false)
   105→
   106→  // Add form state
   107→  const [addForm, setAddForm] = useState<BusForm>(defaultForm)
   108→
   109→  // Delete dialog state
   110→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   111→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null)
   112→
   113→  // Routes dialog state
   114→  const [routesDialogOpen, setRoutesDialogOpen] = useState(false)
   115→  const [routesTarget, setRoutesTarget] = useState<Bus | null>(null)
   116→
   117→  // Fetch buses
   118→  useEffect(() => {
   119→    let cancelled = false
   120→    async function load() {
   121→      if (!selectedSchoolId) {
   122→        if (!cancelled) {
   123→          setBuses([])
   124→          setLoading(false)
   125→        }
   126→        return
   127→      }
   128→      if (!cancelled) setLoading(true)
   129→      try {
   130→        const params = new URLSearchParams({ schoolId: selectedSchoolId })
   131→        if (activeFilter === 'active') params.set('active', 'true')
   132→        else if (activeFilter === 'inactive') params.set('active', 'false')
   133→        if (searchQuery.trim()) params.set('search', searchQuery.trim())
   134→
   135→        const res = await fetch(`/api/buses?${params.toString()}`)
   136→        if (!cancelled && res.ok) {
   137→          const data = await res.json()
   138→          setBuses(Array.isArray(data) ? data : [])
   139→        }
   140→      } catch {
   141→        if (!cancelled) toast.error('فشل في تحميل بيانات الأسطول')
   142→      } finally {
   143→        if (!cancelled) setLoading(false)
   144→      }
   145→    }
   146→    load()
   147→    return () => { cancelled = true }
   148→  }, [selectedSchoolId, searchQuery, activeFilter])
   149→
   150→  // Filter buses client-side for search (as backup if API doesn't support search)
   151→  const filteredBuses = buses.filter((bus) => {
   152→    if (!searchQuery.trim()) return true
   153→    const q = searchQuery.toLowerCase()
   154→    return (
   155→      bus.plateNumber.toLowerCase().includes(q) ||
   156→      bus.driverName.toLowerCase().includes(q)
   157→    )
   158→  })
   159→
   160→  // Open edit dialog
   161→  const openEdit = (bus: Bus) => {
   162→    setEditId(bus.id)
   163→    setEditForm({
   164→      plateNumber: bus.plateNumber,
   165→      driverName: bus.driverName,
   166→      driverPhone: bus.driverPhone || '',
   167→      driverLicense: bus.driverLicense || '',
   168→      supervisorName: bus.supervisorName || '',
   169→      supervisorPhone: bus.supervisorPhone || '',
   170→      capacity: bus.capacity,
   171→      model: bus.model || '',
   172→      color: bus.color || '#610000',
   173→      active: bus.active,
   174→      notes: bus.notes || '',
   175→    })
   176→    setEditDialogOpen(true)
   177→  }
   178→
   179→  // Handle edit save
   180→  const handleEditSave = async () => {
   181→    if (!editForm.plateNumber.trim() || !editForm.driverName.trim()) {
   182→      toast.error('يرجى إدخال رقم اللوحة واسم السائق')
   183→      return
   184→    }
   185→    setSaving(true)
   186→    try {
   187→      const body = {
   188→        schoolId: selectedSchoolId,
   189→        plateNumber: editForm.plateNumber.trim(),
   190→        driverName: editForm.driverName.trim(),
   191→        driverPhone: editForm.driverPhone.trim() || null,
   192→        driverLicense: editForm.driverLicense.trim() || null,
   193→        supervisorName: editForm.supervisorName.trim() || null,
   194→        supervisorPhone: editForm.supervisorPhone.trim() || null,
   195→        capacity: editForm.capacity,
   196→        model: editForm.model.trim() || null,
   197→        color: editForm.color || null,
   198→        active: editForm.active,
   199→        notes: editForm.notes.trim() || null,
   200→      }
   201→
   202→      const res = await fetch(`/api/buses/${editId}`, {
   203→        method: 'PUT',
   204→        headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify(body),
   206→      })
   207→
   208→      if (res.ok) {
   209→        toast.success('تم تحديث بيانات الباص بنجاح')
   210→        setEditDialogOpen(false)
   211→        // Refresh the list
   212→        refreshBuses()
   213→      } else {
   214→        toast.error('فشل في تحديث البيانات')
   215→      }
   216→    } catch {
   217→      toast.error('فشل في تحديث البيانات')
   218→    } finally {
   219→      setSaving(false)
   220→    }
   221→  }
   222→
   223→  // Handle add bus
   224→  const handleAddBus = async () => {
   225→    if (!addForm.plateNumber.trim() || !addForm.driverName.trim()) {
   226→      toast.error('يرجى إدخال رقم اللوحة واسم السائق')
   227→      return
   228→    }
   229→    if (addForm.capacity < 1) {
   230→      toast.error('يجب أن تكون سعة الباص أكبر من صفر')
   231→      return
   232→    }
   233→    setSaving(true)
   234→    try {
   235→      const body = {
   236→        schoolId: selectedSchoolId,
   237→        plateNumber: addForm.plateNumber.trim(),
   238→        driverName: addForm.driverName.trim(),
   239→        driverPhone: addForm.driverPhone.trim() || null,
   240→        driverLicense: addForm.driverLicense.trim() || null,
   241→        supervisorName: addForm.supervisorName.trim() || null,
   242→        supervisorPhone: addForm.supervisorPhone.trim() || null,
   243→        capacity: addForm.capacity,
   244→        model: addForm.model.trim() || null,
   245→        color: addForm.color || null,
   246→        active: addForm.active,
   247→        notes: addForm.notes.trim() || null,
   248→      }
   249→
   250→      const res = await fetch('/api/buses', {
   251→        method: 'POST',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify(body),
   254→      })
   255→
   256→      if (res.ok) {
   257→        toast.success('تم إضافة الباص بنجاح')
   258→        setAddForm(defaultForm)
   259→        setActiveTab('fleet')
   260→        refreshBuses()
   261→      } else {
   262→        const errData = await res.json().catch(() => ({}))
   263→        toast.error(errData.error || 'فشل في إضافة الباص')
   264→      }
   265→    } catch {
   266→      toast.error('فشل في إضافة الباص')
   267→    } finally {
   268→      setSaving(false)
   269→    }
   270→  }
   271→
   272→  // Handle delete (soft delete)
   273→  const handleDelete = async (id: string) => {
   274→    try {
   275→      const res = await fetch(`/api/buses/${id}`, { method: 'DELETE' })
   276→      if (res.ok) {
   277→        toast.success('تم تعطيل الباص بنجاح')
   278→        refreshBuses()
   279→      } else {
   280→        toast.error('فشل في تعطيل الباص')
   281→      }
   282→    } catch {
   283→      toast.error('فشل في تعطيل الباص')
   284→    } finally {
   285→      setDeleteDialogOpen(false)
   286→      setDeleteTarget(null)
   287→    }
   288→  }
   289→
   290→  // Refresh bus list
   291→  const refreshBuses = async () => {
   292→    if (!selectedSchoolId) return
   293→    try {
   294→      const params = new URLSearchParams({ schoolId: selectedSchoolId })
   295→      if (activeFilter === 'active') params.set('active', 'true')
   296→      else if (activeFilter === 'inactive') params.set('active', 'false')
   297→
   298→      const res = await fetch(`/api/buses?${params.toString()}`)
   299→      if (res.ok) {
   300→        const data = await res.json()
   301→        setBuses(Array.isArray(data) ? data : [])
   302→      }
   303→    } catch {
   304→      toast.error('فشل في تحديث البيانات')
   305→    }
   306→  }
   307→
   308→  // View routes
   309→  const openRoutes = (bus: Bus) => {
   310→    setRoutesTarget(bus)
   311→    setRoutesDialogOpen(true)
   312→  }
   313→
   314→  return (
   315→    <div className="space-y-6" dir="rtl">
   316→      {/* Header */}
   317→      <div className="flex justify-between items-center">
   318→        <h2 className="text-lg font-bold text-[#1a1a2e] flex items-center gap-2">
   319→          <Bus className="w-5 h-5 text-[#610000]" />
   320→          إدارة أسطول الباصات
   321→        </h2>
   322→      </div>
   323→
   324→      {/* Tabs */}
   325→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   326→        <TabsList className="h-11">
   327→          <TabsTrigger value="fleet" className="min-h-[44px] px-4 text-sm">
   328→            <Bus className="w-4 h-4 ml-1" />
   329→            الأسطول
   330→          </TabsTrigger>
   331→          <TabsTrigger value="add" className="min-h-[44px] px-4 text-sm">
   332→            <Plus className="w-4 h-4 ml-1" />
   333→            إضافة باص
   334→          </TabsTrigger>
   335→        </TabsList>
   336→
   337→        {/* ===== Tab 1: Fleet List ===== */}
   338→        <TabsContent value="fleet" className="space-y-4 mt-4">
   339→          {/* Search and Filter */}
   340→          <div className="flex flex-col sm:flex-row gap-3">
   341→            <div className="relative flex-1">
   342→              <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   343→              <Input
   344→                placeholder="البحث برقم اللوحة أو اسم السائق..."
   345→                value={searchQuery}
   346→                onChange={(e) => setSearchQuery(e.target.value)}
   347→                className="h-11 pr-10"
   348→              />
   349→            </div>
   350→            <Select value={activeFilter} onValueChange={setActiveFilter}>
   351→              <SelectTrigger className="h-11 w-full sm:w-[180px]">
   352→                <SelectValue placeholder="الحالة" />
   353→              </SelectTrigger>
   354→              <SelectContent>
   355→                <SelectItem value="all">الكل</SelectItem>
   356→                <SelectItem value="active">نشط</SelectItem>
   357→                <SelectItem value="inactive">معطل</SelectItem>
   358→              </SelectContent>
   359→            </Select>
   360→          </div>
   361→
   362→          {/* Bus Grid */}
   363→          {loading ? (
   364→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   365→              {Array.from({ length: 6 }).map((_, i) => (
   366→                <Skeleton key={i} className="h-64 rounded-xl" />
   367→              ))}
   368→            </div>
   369→          ) : filteredBuses.length === 0 ? (
   370→            <Card>
   371→              <CardContent className="py-12 text-center">
   372→                <AlertCircle className="w-12 h-12 mx-auto text-gray-300 mb-3" />
   373→                <p className="text-gray-400 text-sm">لا يوجد باصات مسجلة</p>
   374→                <Button
   375→                  className="mt-4 bg-[#610000] hover:bg-[#8B0000] text-white min-h-[44px]"
   376→                  onClick={() => setActiveTab('add')}
   377→                >
   378→                  <Plus className="w-4 h-4 ml-1" />
   379→                  إضافة باص جديد
   380→                </Button>
   381→              </CardContent>
   382→            </Card>
   383→          ) : (
   384→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   385→              {filteredBuses.map((bus) => (
   386→                <BusCard
   387→                  key={bus.id}
   388→                  bus={bus}
   389→                  onEdit={() => openEdit(bus)}
   390→                  onDelete={() => {
   391→                    setDeleteTarget(bus.id)
   392→                    setDeleteDialogOpen(true)
   393→                  }}
   394→                  onViewRoutes={() => openRoutes(bus)}
   395→                />
   396→              ))}
   397→            </div>
   398→          )}
   399→        </TabsContent>
   400→
   401→        {/* ===== Tab 2: Add Bus ===== */}
   402→        <TabsContent value="add" className="mt-4">
   403→          <div className="max-w-2xl mx-auto space-y-4">
   404→            {/* Bus Info Section */}
   405→            <Card>
   406→              <CardHeader className="pb-3">
   407→                <CardTitle className="text-base flex items-center gap-2">
   408→                  <Truck className="w-4 h-4 text-[#610000]" />
   409→                  بيانات الباص
   410→                </CardTitle>
   411→              </CardHeader>
   412→              <CardContent className="space-y-4">
   413→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   414→                  <div className="space-y-1.5">
   415→                    <Label>رقم اللوحة *</Label>
   416→                    <Input
   417→                      value={addForm.plateNumber}
   418→                      onChange={(e) => setAddForm({ ...addForm, plateNumber: e.target.value })}
   419→                      className="h-11"
   420→                      placeholder="أدخل رقم اللوحة"
   421→                    />
   422→                  </div>
   423→                  <div className="space-y-1.5">
   424→                    <Label>الموديل</Label>
   425→                    <Input
   426→                      value={addForm.model}
   427→                      onChange={(e) => setAddForm({ ...addForm, model: e.target.value })}
   428→                      className="h-11"
   429→                      placeholder="مثال: تويوتا كوستار"
   430→                    />
   431→                  </div>
   432→                </div>
   433→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   434→                  <div className="space-y-1.5">
   435→                    <Label>السعة *</Label>
   436→                    <Input
   437→                      type="number"
   438→                      min={1}
   439→                      value={addForm.capacity}
   440→                      onChange={(e) => setAddForm({ ...addForm, capacity: parseInt(e.target.value) || 0 })}
   441→                      className="h-11"
   442→                      placeholder="عدد الركاب"
   443→                    />
   444→                  </div>
   445→                  <div className="space-y-1.5">
   446→                    <Label>لون الباص</Label>
   447→                    <Select
   448→                      value={addForm.color}
   449→                      onValueChange={(value) => setAddForm({ ...addForm, color: value })}
   450→                    >
   451→                      <SelectTrigger className="h-11 w-full">
   452→                        <div className="flex items-center gap-2">
   453→                          <div
   454→                            className="w-4 h-4 rounded-full border border-gray-300 shrink-0"
   455→                            style={{ backgroundColor: addForm.color || '#610000' }}
   456→                          />
   457→                          <SelectValue placeholder="اختر اللون" />
   458→                        </div>
   459→                      </SelectTrigger>
   460→                      <SelectContent>
   461→                        {predefinedColors.map((c) => (
   462→                          <SelectItem key={c.value} value={c.value}>
   463→                            <div className="flex items-center gap-2">
   464→                              <div
   465→                                className="w-4 h-4 rounded-full border border-gray-300 shrink-0"
   466→                                style={{ backgroundColor: c.value }}
   467→                              />
   468→                              {c.label}
   469→                            </div>
   470→                          </SelectItem>
   471→                        ))}
   472→                      </SelectContent>
   473→                    </Select>
   474→                  </div>
   475→                </div>
   476→              </CardContent>
   477→            </Card>
   478→
   479→            {/* Driver Info Section */}
   480→            <Card>
   481→              <CardHeader className="pb-3">
   482→                <CardTitle className="text-base flex items-center gap-2">
   483→                  <User className="w-4 h-4 text-[#610000]" />
   484→                  بيانات السائق
   485→                </CardTitle>
   486→              </CardHeader>
   487→              <CardContent className="space-y-4">
   488→                <div className="space-y-1.5">
   489→                  <Label>اسم السائق *</Label>
   490→                  <Input
   491→                    value={addForm.driverName}
   492→                    onChange={(e) => setAddForm({ ...addForm, driverName: e.target.value })}
   493→                    className="h-11"
   494→                    placeholder="اسم السائق بالكامل"
   495→                  />
   496→                </div>
   497→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   498→                  <div className="space-y-1.5">
   499→                    <Label>رقم الهاتف</Label>
   500→                    <Input
   501→                      value={addForm.driverPhone}
   502→                      onChange={(e) => setAddForm({ ...addForm, driverPhone: e.target.value })}
   503→                      className="h-11"
   504→                      placeholder="05XXXXXXXX"
   505→                      dir="ltr"
   506→                    />
   507→                  </div>
   508→                  <div className="space-y-1.5">
   509→                    <Label>رقم الرخصة</Label>
   510→                    <Input
   511→                      value={addForm.driverLicense}
   512→                      onChange={(e) => setAddForm({ ...addForm, driverLicense: e.target.value })}
   513→                      className="h-11"
   514→                      placeholder="رقم رخصة القيادة"
   515→                      dir="ltr"
   516→                    />
   517→                  </div>
   518→                </div>
   519→              </CardContent>
   520→            </Card>
   521→
   522→            {/* Supervisor Info Section */}
   523→            <Card>
   524→              <CardHeader className="pb-3">
   525→                <CardTitle className="text-base flex items-center gap-2">
   526→                  <Users className="w-4 h-4 text-[#610000]" />
   527→                  بيانات المشرف
   528→                </CardTitle>
   529→              </CardHeader>
   530→              <CardContent className="space-y-4">
   531→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   532→                  <div className="space-y-1.5">
   533→                    <Label>اسم المشرف</Label>
   534→                    <Input
   535→                      value={addForm.supervisorName}
   536→                      onChange={(e) => setAddForm({ ...addForm, supervisorName: e.target.value })}
   537→                      className="h-11"
   538→                      placeholder="اسم المشرف"
   539→                    />
   540→                  </div>
   541→                  <div className="space-y-1.5">
   542→                    <Label>رقم هاتف المشرف</Label>
   543→                    <Input
   544→                      value={addForm.supervisorPhone}
   545→                      onChange={(e) => setAddForm({ ...addForm, supervisorPhone: e.target.value })}
   546→                      className="h-11"
   547→                      placeholder="05XXXXXXXX"
   548→                      dir="ltr"
   549→                    />
   550→                  </div>
   551→                </div>
   552→              </CardContent>
   553→            </Card>
   554→
   555→            {/* Other Info Section */}
   556→            <Card>
   557→              <CardHeader className="pb-3">
   558→                <CardTitle className="text-base flex items-center gap-2">
   559→                  <FileText className="w-4 h-4 text-[#610000]" />
   560→                  أخرى
   561→                </CardTitle>
   562→              </CardHeader>
   563→              <CardContent className="space-y-4">
   564→                <div className="space-y-1.5">
   565→                  <Label>ملاحظات</Label>
   566→                  <Textarea
   567→                    value={addForm.notes}
