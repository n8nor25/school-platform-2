   158→   158→  if (!dateStr) return '—'

   160→   160→  return d.toLocaleDateString('ar-EG', { year: 'numeric', month: 'short', day: 'numeric' })

   163→   163→// Calculate how many months a subscription should have payments for
   164→   164→function getExpectedMonths(startDate: string, endDate: string | null): number {
   165→   165→  const start = new Date(startDate)
   166→   166→  const end = endDate ? new Date(endDate) : new Date()
   167→   167→  const months = (end.getFullYear() - start.getFullYear()) * 12 + (end.getMonth() - start.getMonth()) + 1
   168→   168→  return Math.max(0, months)
   169→   169→}
