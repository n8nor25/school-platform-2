    78→    78→// ===== Helper: Format current date in Arabic =====

    88→    88→// ===== Helper: Direction badge color =====
    89→    89→function getDirectionBadge(direction: string) {
    90→    90→  switch (direction) {
    91→    91→    case 'ذهاب':
    92→    92→      return 'bg-emerald-100 text-emerald-800 border-emerald-200'
    93→    93→    case 'عودة':
    94→    94→      return 'bg-amber-100 text-amber-800 border-amber-200'

    96→    96→      return 'bg-sky-100 text-sky-800 border-sky-200'
    97→    97→    default:
    98→    98→      return 'bg-gray-100 text-gray-800 border-gray-200'
    99→    99→  }

   102→   102→// ===== Helper: Occupancy bar color =====
   103→   103→function getOccupancyColor(rate: number): string {
   104→   104→  if (rate >= 90) return 'bg-red-500'
   105→   105→  if (rate >= 70) return 'bg-amber-500'
   106→   106→  if (rate >= 40) return 'bg-emerald-500'
