   244→function formatArabicDate(iso: string): string {
   245→  try {
   246→    const d = new Date(iso);
   247→    if (isNaN(d.getTime())) return '—';
   248→    return d.toLocaleDateString('ar-EG', {
   249→      year: 'numeric',
   250→      month: 'long',
   251→      day: 'numeric',
   252→      hour: '2-digit',
   253→      minute: '2-digit',
   254→    });
   255→  } catch {
   256→    return '—';
   257→  }
   258→}

   413→export default function TeacherExamsPage({
   414→  onBack,
   415→  schoolId,
   416→  teacherId,
   417→  teacherName,
   418→}: TeacherExamsPageProps) {
   419→  // ===== State =====
   420→  const [fadeIn, setFadeIn] = useState(false);
   421→  const [exams, setExams] = useState<ExamListItem[]>([]);
   422→  const [loading, setLoading] = useState(false);
   423→  const [error, setError] = useState<string | null>(null);
   424→  const [search, setSearch] = useState('');
   425→  const [debouncedSearch, setDebouncedSearch] = useState('');
   426→  const [statusFilter, setStatusFilter] = useState<'all' | ExamStatus>('all');
   427→  const [categoryFilter, setCategoryFilter] = useState<'all' | ExamCategory>('all');
   428→
   429→  // Detail modal state
   430→  const [detailOpen, setDetailOpen] = useState(false);
   431→  const [detailLoading, setDetailLoading] = useState(false);
   432→  const [detailError, setDetailError] = useState<string | null>(null);
   433→  const [detailExam, setDetailExam] = useState<ExamDetail | null>(null);
   434→  const [submissionsStats, setSubmissionsStats] = useState<SubmissionsStats | null>(null);
   435→
   436→  // Create dialog state
   437→  const [showCreateDialog, setShowCreateDialog] = useState(false);
   438→  const [createSubmitting, setCreateSubmitting] = useState(false);
   439→  const [createForm, setCreateForm] = useState<CreateExamForm>(makeEmptyCreateForm());
   440→  const [createErrors, setCreateErrors] = useState<Record<string, string>>({});
   441→
   442→  // Mounted ref (prevent setState after unmount)
   443→  const mountedRef = useRef(true);
   444→
   445→  // ===== Restore fadeIn =====
   446→  useEffect(() => {
   447→    const t = setTimeout(() => setFadeIn(true), 0);
   448→    return () => clearTimeout(t);
   449→  }, []);
   450→
   451→  // ===== mountedRef lifecycle =====
   452→  useEffect(() => {
   453→    mountedRef.current = true;
   454→    return () => {
   455→      mountedRef.current = false;
   456→    };
   457→  }, []);
   458→
   459→  // ===== Debounced search (300ms) =====
   460→  useEffect(() => {
   461→    const t = setTimeout(() => {
   462→      if (mountedRef.current) setDebouncedSearch(search);
   463→    }, 300);
   464→    return () => clearTimeout(t);
   465→  }, [search]);
   466→
   467→  // ===== Fetch exams =====
   468→  const fetchExams = useCallback(async (silent: boolean = false) => {
   469→    if (!mountedRef.current) return;
   470→    if (!silent) {
   471→      const t = setTimeout(() => setLoading(true), 0);
   472→      void t;
   473→    }
   474→    if (!silent) {
   475→      const t = setTimeout(() => setError(null), 0);
   476→      void t;
   477→    }
   478→    try {
   479→      const params = new URLSearchParams();
   480→      params.set('schoolId', schoolId);
   481→      params.set('teacherId', teacherId);
   482→      if (debouncedSearch.trim()) params.set('search', debouncedSearch.trim());
   483→      if (statusFilter !== 'all') params.set('status', statusFilter);
   484→      // Note: category filtering is done client-side because the GET endpoint
   485→      // returns the category field for each exam; we filter the array directly.
   486→      params.set('limit', '100');
   487→
   488→      const res = await fetch(`/api/exams/teacher?${params.toString()}`);
   489→      const data = await res.json();
   490→      if (!res.ok) {
   491→        throw new Error(data?.error || 'فشل جلب الامتحانات');
   492→      }
   493→      if (!mountedRef.current) return;
   494→      const list = (data as ExamsListResponse).exams || [];
   495→      const t = setTimeout(() => setExams(list), 0);
   496→      void t;
   497→    } catch (err) {
   498→      if (!mountedRef.current) return;
   499→      const msg = err instanceof Error ? err.message : 'فشل جلب الامتحانات';
   500→      const t = setTimeout(() => setError(msg), 0);
   501→      void t;
   502→      if (!silent) toast.error(msg);
   503→    } finally {
   504→      if (!mountedRef.current) return;
   505→      const t = setTimeout(() => setLoading(false), 0);
   506→      void t;
   507→    }
   508→  }, [schoolId, teacherId, debouncedSearch, statusFilter]);
   509→
   510→  // ===== Initial fetch + auto-refresh + focus =====
   511→  useEffect(() => {
   512→    fetchExams(false);
   513→    const interval = setInterval(() => {
   514→      fetchExams(true);
   515→    }, 60_000);
   516→    const onFocus = () => fetchExams(true);
   517→    window.addEventListener('focus', onFocus);
   518→    return () => {
   519→      clearInterval(interval);
   520→      window.removeEventListener('focus', onFocus);
   521→    };
   522→  }, [debouncedSearch, statusFilter, schoolId, teacherId]);
   523→
   524→  // ===== Fetch exam detail =====
   525→  const openExamDetail = useCallback(async (examId: string) => {
   526→    if (!mountedRef.current) return;
   527→    const t1 = setTimeout(() => {
   528→      setDetailOpen(true);
   529→      setDetailLoading(true);
   530→      setDetailError(null);
   531→      setDetailExam(null);
   532→      setSubmissionsStats(null);
   533→    }, 0);
   534→    void t1;
   535→    try {
   536→      const params = new URLSearchParams();
   537→      params.set('schoolId', schoolId);
   538→      params.set('teacherId', teacherId);
   539→
   540→      const [examRes, subsRes] = await Promise.allSettled([
   541→        fetch(`/api/exams/teacher/${examId}?${params.toString()}`),
   542→        fetch(`/api/exams/teacher/${examId}/submissions?${params.toString()}&limit=1`),
   543→      ]);
   544→
   545→      if (!mountedRef.current) return;
   546→
   547→      if (examRes.status === 'fulfilled') {
   548→        const examData = await examRes.value.json();
   549→        if (!examRes.value.ok) {
   550→          throw new Error(examData?.error || 'فشل جلب تفاصيل الامتحان');
   551→        }
   552→        const t = setTimeout(() => setDetailExam((examData as ExamDetailResponse).exam), 0);
   553→        void t;
   554→      } else {
   555→        throw new Error('فشل الاتصال بالخادم');
   556→      }
   557→
