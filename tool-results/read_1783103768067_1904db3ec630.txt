    82→function normalizeStatus(raw: string | undefined | null): AttendanceStatus {
    83→  if (!raw) return 'UNKNOWN';
    84→  const s = String(raw).trim().toUpperCase();
    85→  if (s === 'PRESENT' || s === 'حاضر' || s === 'حاضر/' || s.includes('PRESENT') || s.includes('حاضر')) return 'PRESENT';
    86→  if (s === 'LATE' || s === 'متأخر' || s.includes('LATE') || s.includes('متأخر')) return 'LATE';
    87→  if (s === 'EXCUSED' || s === 'غائب بعذر' || s.includes('EXCUSED') || s.includes('بعذر')) return 'EXCUSED';
    88→  if (s === 'ABSENT' || s === 'غائب' || s.includes('ABSENT') || s.includes('غائب')) return 'ABSENT';
    89→  return 'UNKNOWN';
    90→}

   154→function formatDateKey(d: Date): string {
   155→  const y = d.getFullYear();
   156→  const m = String(d.getMonth() + 1).padStart(2, '0');
   157→  const day = String(d.getDate()).padStart(2, '0');
   158→  return `${y}-${m}-${day}`;
   159→}
   160→
   161→function parseDateKey(key: string): Date {
   162→  // Accept ISO strings and yyyy-mm-dd
   163→  return new Date(key.length > 10 ? key : key + 'T00:00:00');
   164→}
   165→
   166→function formatLongDate(d: Date): string {
   167→  try {
   168→    return d.toLocaleDateString('ar-EG', {
   169→      weekday: 'long', year: 'numeric', month: 'long', day: 'numeric',
   170→    });
   171→  } catch {
   172→    return d.toDateString();
   173→  }
   174→}
   175→
   176→function formatShortDate(d: Date): string {
   177→  try {
   178→    return d.toLocaleDateString('ar-EG', {
   179→      month: 'short', day: 'numeric',
   180→    });
   181→  } catch {
   182→    return formatDateKey(d);
   183→  }
   184→}
   185→
   186→function formatTime(iso?: string | null): string {
   187→  if (!iso) return '';
   188→  try {
   189→    return new Date(iso).toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' });
   190→  } catch {
   191→    return '';
   192→  }
   193→}

   197→// Build a 5-week (35-day) grid ending today, grouped by week (Sat..Fri).
   198→function buildHeatmap(records: AttendanceRecord[]): {
   199→  weeks: { date: Date; key: string; status: AttendanceStatus }[][];
   200→  todayKey: string;
   201→} {
   202→  const recordMap = new Map<string, AttendanceStatus>();
   203→  for (const r of records) {
   204→    if (!r?.date) continue;
   205→    const d = parseDateKey(r.date);
   206→    if (isNaN(d.getTime())) continue;
   207→    recordMap.set(formatDateKey(d), normalizeStatus(r.status));
   208→  }
   209→
   210→  const today = new Date();
   211→  today.setHours(0, 0, 0, 0);
   212→  const todayKey = formatDateKey(today);
   213→
   214→  // Find the Saturday of the week containing (today - 28 days).
   215→  // 35 days = 5 weeks ending on the current week.
   216→  const start = new Date(today);
   217→  start.setDate(start.getDate() - 34); // 35 days including today
   218→  // Shift to the Saturday on or before `start`
   219→  const startDay = start.getDay(); // 0=Sun .. 6=Sat
   220→  const satOffset = (startDay + 1) % 7; // days back to Saturday
   221→  start.setDate(start.getDate() - satOffset);
   222→
   223→  const weeks: { date: Date; key: string; status: AttendanceStatus }[][] = [];
   224→  for (let w = 0; w < 5; w++) {
   225→    const week: { date: Date; key: string; status: AttendanceStatus }[] = [];
   226→    for (let d = 0; d < 7; d++) {
   227→      const date = new Date(start);
   228→      date.setDate(start.getDate() + w * 7 + d);
   229→      const key = formatDateKey(date);
   230→      const status = recordMap.get(key) ?? (date > today ? 'UNKNOWN' : 'UNKNOWN');
   231→      week.push({ date, key, status });
   232→    }
   233→    weeks.push(week);
   234→  }
   235→  return { weeks, todayKey };
   236→}

   513→                  <div className="flex items-center gap-1.5 bg-white/15 rounded-full px-3 py-1.5 text-sm font-semibold backdrop-blur">
   514→                    <Flame className="w-4 h-4 text-amber-200" />
   515→                    <span>تتابع {stats.streak} يوم</span>
   516→                  </div>
   517→                )}
   518→                <Button
   519→                  size="sm"
   520→                  onClick={() => fetchData(true)}
   521→                  disabled={refreshing}
   522→                  className="bg-white/20 hover:bg-white/30 text-white border-0 backdrop-blur"
   523→                >
   524→                  {refreshing ? <Loader2 className="w-4 h-4 animate-spin" /> : <RefreshCw className="w-4 h-4" />}

   526→                </Button>
   527→              </div>
   528→            </div>
   529→            {lastUpdated && (
   530→              <p className="text-[11px] text-emerald-50/80 mt-2">
   531→                آخر تحديث: {lastUpdated.toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' })}
   532→              </p>
   533→            )}
   534→          </div>
   535→        </div>
   536→
   537→        {/* Today's status */}
   538→        <Card className={`border-2 ${todayMeta ? todayMeta.border : 'border-slate-200 dark:border-slate-800'}`}>
   539→          <CardContent className="p-4">
   540→            <div className="flex items-center gap-4">
   541→              <div className={`w-12 h-12 rounded-xl flex items-center justify-center shrink-0 ${todayMeta ? todayMeta.bg : 'bg-slate-100 dark:bg-slate-800'}`}>
   542→                <TodayIcon className={`w-6 h-6 ${todayMeta ? todayMeta.text : 'text-slate-400'}`} />
   543→              </div>
   544→              <div className="flex-1 min-w-0">
   545→                <p className="text-xs text-muted-foreground mb-0.5">حالة اليوم</p>
   546→                {todayMeta ? (
   547→                  <>
   548→                    <p className={`font-bold text-lg ${todayMeta.text}`}>{todayMeta.label}</p>
   549→                    {todayRecord?.arrivalTime && (
   550→                      <p className="text-xs text-muted-foreground">
   551→                        وقت الحضور: {formatTime(todayRecord.arrivalTime)}
   552→                      </p>
   553→                    )}
   554→                    {todayRecord?.notes && (
   555→                      <p className="text-xs text-muted-foreground mt-0.5 truncate">
   556→                        ملاحظة: {todayRecord.notes}
   557→                      </p>
   558→                    )}
   559→                  </>
   560→                ) : (
   561→                  <p className="font-bold text-base text-slate-500 dark:text-slate-400">لم يُسجَّل بعد</p>
   562→                )}
   563→              </div>
   564→              <div className="text-left shrink-0">
   565→                <p className="text-xs text-muted-foreground">{formatShortDate(today)}</p>
   566→              </div>
   567→            </div>
   568→          </CardContent>
   569→        </Card>
   570→
   571→        {/* KPIs */}
   572→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   573→          {kpis.map((kpi) => {
   574→            const Icon = kpi.icon;
   575→            return (
   576→              <Card key={kpi.label} className="overflow-hidden">
   577→                <CardContent className="p-4">
   578→                  <div className={`w-10 h-10 rounded-lg bg-gradient-to-br ${kpi.gradient} flex items-center justify-center mb-2 shadow-sm`}>
   579→                    <Icon className="w-5 h-5 text-white" />
   580→                  </div>
   581→                  <p className="text-2xl font-extrabold leading-tight">{kpi.value}</p>
   582→                  <p className="text-xs text-muted-foreground mt-0.5">{kpi.label}</p>
   583→                </CardContent>
   584→              </Card>
   585→            );
   586→          })}
   587→        </div>
   588→
   589→        {/* Heatmap */}
   590→        <Card>
   591→          <CardContent className="p-4">
   592→            <div className="flex items-center justify-between mb-3">
   593→              <div className="flex items-center gap-2">
   594→                <Calendar className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
   595→                <h4 className="font-semibold text-sm">خريطة الحضور (آخر 5 أسابيع)</h4>
   596→              </div>
   597→            </div>
   598→            <div className="overflow-x-auto -mx-1 px-1">
   599→              <div className="min-w-[320px]">
   600→                {/* Day cells (Sat..Fri columns) */}
   601→                <div className="grid grid-cols-5 gap-1.5">
   602→                  {weeks.map((week, wi) => (
   603→                    <div key={`wk-${wi}`} className="space-y-1">
   604→                      <div className="grid grid-cols-7 gap-1">
   605→                        {week.map((day, di) => {
   606→                          const meta = STATUS_META[day.status];
   607→                          const isToday = day.key === todayKey;
   608→                          const isFuture = day.date > today;
   609→                          return (

   611→                              key={`d-${wi}-${di}`}

   614→                            />
   615→                          );

   617→                      </div>
   618→                      {wi === 0 && (
   619→                        <div className="grid grid-cols-7 gap-1">
   620→                          {WEEKDAY_LABELS.map((label, li) => (
   621→                            <div key={`wl-${li}`} className="text-[8px] text-muted-foreground text-center leading-tight">
   622→                              {label.charAt(0)}
   623→                            </div>
   624→                          ))}
   625→                        </div>
   626→                      )}
   627→                    </div>
   628→                  ))}
   629→                </div>
   630→                {/* Legend */}
   631→                <div className="flex flex-wrap items-center gap-x-3 gap-y-1.5 mt-3 text-[11px] text-muted-foreground">
   632→                  <span className="flex items-center gap-1"><span className="w-3 h-3 rounded-sm bg-emerald-500" />حاضر</span>
   633→                  <span className="flex items-center gap-1"><span className="w-3 h-3 rounded-sm bg-amber-500" />متأخر</span>
   634→                  <span className="flex items-center gap-1"><span className="w-3 h-3 rounded-sm bg-red-500" />غائب</span>
   635→                  <span className="flex items-center gap-1"><span className="w-3 h-3 rounded-sm bg-sky-500" />بعذر</span>
   636→                  <span className="flex items-center gap-1"><span className="w-3 h-3 rounded-sm bg-slate-300 dark:bg-slate-700" />لا سجل</span>
   637→                </div>
   638→              </div>
   639→            </div>
   640→          </CardContent>
   641→        </Card>
   642→
   643→        {/* Records list */}
   644→        <Card>
   645→          <CardContent className="p-4">
   646→            <div className="flex items-center justify-between mb-3">
   647→              <h4 className="font-semibold text-sm">السجلات التفصيلية</h4>
   648→              <Badge variant="secondary" className="text-xs">{sortedRecords.length} سجل</Badge>
   649→            </div>
   650→            {sortedRecords.length === 0 ? (
   651→              <p className="text-sm text-muted-foreground text-center py-4">لا توجد سجلات لعرضها</p>
   652→            ) : (
   653→              <>
   654→                <div className="space-y-2">
   655→                  {visibleRecords.map((rec) => {

   657→                    const meta = STATUS_META[status];
   658→                    const Icon = meta.icon;
   659→                    const date = parseDateKey(rec.date);

   680→                          {rec.notes && (
   681→                            <p className="text-xs text-muted-foreground mt-0.5 truncate">

   685→                        </div>
   686→                      </div>
   687→                    );
   688→                  })}
   689→                </div>
   690→                {sortedRecords.length > 6 && (
   691→                  <Button
   692→                    variant="ghost"
   693→                    size="sm"
   694→                    onClick={() => setShowAllRecords((v) => !v)}
   695→                    className="w-full mt-3 text-emerald-700 dark:text-emerald-400 hover:bg-emerald-50 dark:hover:bg-emerald-900/10"

   697→                    {showAllRecords ? 'عرض أقل' : `عرض جميع السجلات (${sortedRecords.length})`}
   698→                    <ChevronDown className={`w-4 h-4 transition-transform ${showAllRecords ? 'rotate-180' : ''}`} />
   699→                  </Button>
   700→                )}
   701→              </>
   702→            )}
   703→          </CardContent>
   704→        </Card>
   705→      </div>
   706→    );
   707→  };
   708→
   709→  // ===== Main render =====
   710→
   711→  let content: React.ReactNode;
   712→  if (loading) {
   713→    content = renderSkeleton();
   714→  } else if (notFound) {
   715→    content = renderNotFound();
   716→  } else if (error) {
   717→    content = renderError();
   718→  } else if (data) {
   719→    content = renderSuccess();
   720→  } else {
   721→    content = renderSkeleton();
   722→  }
   723→
   724→  return (
   725→    <section className="space-y-3">
   726→      {content}
   727→    </section>
   728→  );
   729→}
