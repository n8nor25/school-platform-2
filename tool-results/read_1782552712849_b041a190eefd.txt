   192→              employeeId: r.employeeId,
   193→              status: r.status,
   194→              checkIn: r.checkIn || '',
   195→              checkOut: r.checkOut || '',
   196→              leaveType: r.leaveType || '',
   197→              leaveDuration: r.leaveDuration || 'يوم كامل',

   201→        })

   291→        toast.error('خطأ في حفظ الحضور')

   460→          {/* Quick Action Buttons + Summary */}
   461→          {Object.keys(attendanceEntries).length > 0 && (

   464→                <div className="flex flex-wrap items-center justify-between gap-3">
   465→                  <div className="flex flex-wrap items-center gap-2">

   494→
   495→          {/* Employee Attendance List */}
   496→          {loading ? (

   498→              <CardContent className="p-6 space-y-4">
   499→                {[...Array(5)].map((_, i) => (
