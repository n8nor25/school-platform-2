   163→                status: r.status,
   164→                arrivalTime: r.arrivalTime || '',

   175→          setAttendanceEntries(entries)
   176→        }
   177→      }
   178→    } catch (err) {
   179→      console.error('Error fetching students:', err)

   342→        const res = await fetch(`/api/attendance?${params}`)

   461→
   462→  // Compute summary for current recording
