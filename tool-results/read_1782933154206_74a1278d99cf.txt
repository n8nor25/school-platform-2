   142→function getCategoryMeta(category: string) {
   143→  return CATEGORY_META[category] || { icon: '📄', color: 'text-gray-700', bg: 'bg-gray-100 text-gray-700 dark:bg-gray-700/30 dark:text-gray-400', label: category || 'عام' };

   146→function timeAgo(dateStr: string): string {
   147→  const date = new Date(dateStr);
   148→  const diff = Math.max(0, Date.now() - date.getTime());
   149→  const mins = Math.floor(diff / 60000);
   150→  if (mins < 1) return 'الآن';
   151→  if (mins < 60) return `قبل ${mins} دقيقة`;
   152→  const hours = Math.floor(mins / 60);
   153→  if (hours < 24) return `قبل ${hours} ساعة`;
   154→  const days = Math.floor(hours / 24);
   155→  if (days < 7) return `قبل ${days} يوم`;
   156→  const weeks = Math.floor(days / 7);
   157→  if (weeks < 4) return `قبل ${weeks} أسبوع`;
   158→  return date.toLocaleDateString('ar-EG', { day: 'numeric', month: 'short' });
   159→}
