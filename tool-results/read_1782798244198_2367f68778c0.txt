   102→function HomePage() {
   103→  const { isAdminMode, selectedSchoolId, setSelectedSchoolId, setSchools, _hasHydrated } = useAdminStore()

   109→
   110→  const [schoolData, setSchoolData] = useState<SchoolData | null>(null)
   111→  const [news, setNews] = useState<NewsItem[]>([])
   112→  const [gallery, setGallery] = useState<GalleryItem[]>([])
   113→  const [teachers, setTeachers] = useState<Teacher[]>([])
   114→  const [sliders, setSliders] = useState<SliderItem[]>([])
   115→  const [customSections, setCustomSections] = useState<CustomSectionItem[]>([])
   116→  const [loading, setLoading] = useState(true)
   117→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false)
   118→  const [servicesOpen, setServicesOpen] = useState(false)
   119→  const [showAdminLogin, setShowAdminLogin] = useState(false)
   120→  const [showResultsPage, setShowResultsPage] = useState(false)
   121→  const [showStudentLife, setShowStudentLife] = useState(false)
   122→  const [showSchedulesPage, setShowSchedulesPage] = useState(false)
   123→  const [showLibraryPage, setShowLibraryPage] = useState(false)
   124→  const [showParentsPage, setShowParentsPage] = useState(false)
   125→  const [showTeacherExamsPage, setShowTeacherExamsPage] = useState(false)
   126→  const [showScrollTop, setShowScrollTop] = useState(false)
   127→  const [currentSlide, setCurrentSlide] = useState(0)
   128→  const logoClickCount = useRef(0)
   129→  const logoClickTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   130→
   131→  const hydrated = _hasHydrated
   132→
   133→  // Resolve school from subdomain only (not from store's school list)
   134→  useEffect(() => {
   135→    if (!hydrated) return
   136→    let cancelled = false
   137→    async function resolveSchool() {
   138→      try {
   139→        if (urlSubdomain) {
   140→          // If subdomain is in URL, fetch by subdomain
   141→          const res = await fetch(`/api/school/by-subdomain?subdomain=${urlSubdomain}`)
   142→          if (res.ok && !cancelled) {
   143→            const data = await res.json()
   144→            if (data.school) {
   145→              setSchoolData(data)
   146→              setSelectedSchoolId(data.school.id)
   147→              return
   148→            }
   149→          }
   150→        }
   151→        // No subdomain in URL — fetch first available school
   152→        const schoolsRes = await fetch('/api/schools')
   153→        if (schoolsRes.ok && !cancelled) {
   154→          const schoolsList = await schoolsRes.json()
   155→          if (Array.isArray(schoolsList) && schoolsList.length > 0) {
   156→            setSchools(schoolsList)
   157→            const firstSchool = schoolsList[0]
   158→            // Fetch full data for the first school
   159→            const res = await fetch(`/api/school?schoolId=${firstSchool.id}`)
   160→            if (res.ok && !cancelled) {
   161→              const data = await res.json()
   162→              if (data.school) {
   163→                setSchoolData(data)
   164→                setSelectedSchoolId(data.school.id)
   165→              }
   166→            }
   167→          }
   168→        }
   169→      } catch {
   170→        // fallback to default data
   171→      }
   172→    }
   173→    resolveSchool()
   174→    return () => { cancelled = true }
   175→  }, [hydrated, urlSubdomain, setSelectedSchoolId, setSchools])
   176→
   177→  // Fetch content for selected school
   178→  useEffect(() => {
   179→    if (!hydrated || !selectedSchoolId) return
   180→    let cancelled = false
   181→    const sid = selectedSchoolId
   182→    async function loadData() {
   183→      setLoading(true)
   184→      try {
   185→        const [newsRes, galleryRes, teachersRes, slidersRes, csRes] = await Promise.allSettled([
   186→          fetch(`/api/news?limit=10&schoolId=${sid}`),
   187→          fetch(`/api/gallery?limit=8&schoolId=${sid}`),
   188→          fetch(`/api/teachers?schoolId=${sid}`),
   189→          fetch(`/api/sliders?schoolId=${sid}`),
   190→          fetch(`/api/custom-sections?schoolId=${sid}`),
   191→        ])
   192→        if (cancelled) return
   193→        if (newsRes.status === 'fulfilled' && newsRes.value.ok) {
   194→          const d = await newsRes.value.json()
   195→          if (Array.isArray(d) && !cancelled) setNews(d)
   196→        }
   197→        if (galleryRes.status === 'fulfilled' && galleryRes.value.ok) {
   198→          const d = await galleryRes.value.json()
   199→          if (Array.isArray(d) && !cancelled) setGallery(d)
   200→        }
   201→        if (teachersRes.status === 'fulfilled' && teachersRes.value.ok) {
   202→          const d = await teachersRes.value.json()
   203→          if (Array.isArray(d) && !cancelled) setTeachers(d)
   204→        }
   205→        if (slidersRes.status === 'fulfilled' && slidersRes.value.ok) {
   206→          const d = await slidersRes.value.json()
   207→          if (Array.isArray(d) && !cancelled) setSliders(d)
   208→        }
   209→        if (csRes.status === 'fulfilled' && csRes.value.ok) {
   210→          const d = await csRes.value.json()
   211→          if (Array.isArray(d) && !cancelled) setCustomSections(d)
   212→        }
   213→      } catch {
   214→        // silently fail
   215→      } finally {
   216→        if (!cancelled) setLoading(false)
   217→      }
   218→    }
   219→    loadData()
   220→    return () => { cancelled = true }
   221→  }, [selectedSchoolId, hydrated])
   222→
   223→  // Auto-rotate slider
   224→  const activeSliders = sliders.filter(s => s.active)
   225→  const newsWithImage = news.filter(n => n.image).slice(0, 5)
   226→  const slideItems = activeSliders.length > 0 ? activeSliders : newsWithImage
   227→  useEffect(() => {
   228→    if (slideItems.length <= 1) return

   232→
   233→  // Scroll to top

   238→  }, [])
   239→
   240→  const handleLogoClick = useCallback(() => {

   244→    if (logoClickCount.current >= 5) { logoClickCount.current = 0; setShowAdminLogin(true) }
   245→  }, [])
   246→
   247→  const handleServiceAction = (action: string) => {
   248→    setServicesOpen(false); setMobileMenuOpen(false)
   249→    if (action === 'results') setShowResultsPage(true)
   250→    if (action === 'schedules') setShowSchedulesPage(true)
   251→    if (action === 'library') setShowLibraryPage(true)
   252→    if (action === 'parents') setShowParentsPage(true)
   253→    if (action === 'teacher-exams') setShowTeacherExamsPage(true)
   254→  }
   255→
   256→  const school = schoolData?.school || defaultSchoolData.school
   257→  const settings = schoolData?.settings || defaultSchoolData.settings
   258→  const stats = schoolData?.stats || defaultSchoolData.stats
   259→  const alertNews = news.filter(n => n.category === 'تنبيه')
   260→
   261→  // Build ticker text for seamless scrolling

   263→
   264→  // Sub-page rendering
   265→  if (isAdminMode) return <AdminLayout />
   266→  if (showStudentLife) return <StudentLifePage onBack={() => setShowStudentLife(false)} />
   267→  if (showResultsPage) return <ResultsPage onBack={() => setShowResultsPage(false)} schoolId={selectedSchoolId} />
   268→  if (showSchedulesPage) return <SchedulesPage onBack={() => setShowSchedulesPage(false)} schoolId={selectedSchoolId} />
   269→  if (showLibraryPage) return <DigitalLibraryPage onBack={() => setShowLibraryPage(false)} schoolId={selectedSchoolId} />
   270→  if (showParentsPage) return <ParentsPortalPage onBack={() => setShowParentsPage(false)} schoolId={selectedSchoolId} />
   271→  if (showTeacherExamsPage) return <TeacherExamsPage onBack={() => setShowTeacherExamsPage(false)} schoolId={selectedSchoolId} />
   272→
   273→  if (!hydrated) {
   274→    return (
   275→      <div className="min-h-screen flex items-center justify-center" style={{ backgroundColor: '#f5f5f5' }}>
   276→        <div className="text-center">
   277→          <div className="w-10 h-10 border-4 border-[#610000] border-t-transparent rounded-full animate-spin mx-auto mb-3" />

   279→        </div>

   281→    )
   282→  }
   283→
   284→  return (
   285→    <div className="min-h-screen flex flex-col" dir="rtl">
   286→
   287→      {/* ===== STICKY TOP ===== */}

   289→        {/* Top Bar - Contact info only, NO school switcher for visitors */}
   290→        <div style={{ backgroundColor: school.primaryColor }} className="text-white text-xs">
   291→          <div className="max-w-[1280px] mx-auto px-4 py-1.5 flex items-center justify-between">
   292→            <div className="flex items-center gap-4">
   293→              {school.phone && <a href={`tel:${school.phone}`} className="hover:opacity-80 flex items-center gap-1">📞 {school.phone}</a>}
   294→              {school.email && <a href={`mailto:${school.email}`} className="hover:opacity-80 flex items-center gap-1">✉️ {school.email}</a>}
   295→            </div>
   296→            <div className="flex items-center gap-2">
   297→              {school.facebookUrl && (
   298→                <a href={school.facebookUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📘</a>
   299→              )}
   300→              {settings?.youtubeUrl && (
   301→                <a href={settings.youtubeUrl} target="_blank" rel="noopener noreferrer" className="hover:opacity-80">📺</a>

   304→          </div>
   305→        </div>
   306→
   307→        {/* Header */}
   308→        <header className="bg-white shadow-sm">
   309→          <div className="max-w-[1280px] mx-auto px-4 py-2 flex items-center gap-3">
   310→            <div onClick={handleLogoClick} className="cursor-pointer shrink-0" title="اضغط 5 مرات للإدارة">
   311→              {school.logoUrl ? (
   312→                <img src={school.logoUrl} alt={school.name} className="w-12 h-12 rounded-full object-cover border-2" style={{ borderColor: school.primaryColor }} />

   314→                <div className="w-12 h-12 rounded-full flex items-center justify-center text-white text-xl" style={{ backgroundColor: school.primaryColor }}>🏫</div>
   315→              )}
   316→            </div>
   317→            <div className="flex-1 min-w-0">
   318→              <h1 className="text-base md:text-xl font-bold truncate" style={{ color: school.primaryColor }}>{school.name}</h1>
   319→              {settings?.heroSubtitle && <p className="text-xs text-gray-500 truncate">{settings.heroSubtitle}</p>}
   320→            </div>
   321→            <button className="lg:hidden p-2 text-2xl text-gray-600" onClick={() => setMobileMenuOpen(!mobileMenuOpen)}>
   322→              {mobileMenuOpen ? '✕' : '☰'}
   323→            </button>
   324→          </div>
   325→        </header>
   326→
   327→        {/* Navigation */}
   328→        <nav style={{ backgroundColor: '#1a2332' }} className="text-white shadow-lg">
   329→          <div className="max-w-[1280px] mx-auto px-4">
   330→            {/* Desktop Nav */}
   331→            <div className="hidden lg:flex items-center h-11">
   332→              {navLinks.map((link) => (
   333→                link.isServices ? (
   334→                  <div key={link.href} className="relative" onMouseEnter={() => setServicesOpen(true)} onMouseLeave={() => setServicesOpen(false)}>

   383→        {/* News Ticker - Seamless scrolling */}

   394→                    {tickerText}
   395→                  </span>
   396→                  <span className="inline-block text-sm mx-4">
   397→                    {tickerText}
   398→                  </span>
   399→                </div>
   400→              </div>
   401→            </div>

   404→      </div>
   405→
   406→      {/* ===== MAIN CONTENT ===== */}
   407→      <main className="flex-1">
   408→        {/* Hero Slider */}
   409→        {(settings?.showSlider ?? true) && (
