    90→export function BusFleetManagement() {
    91→  const { selectedSchoolId } = useAdminStore()
    92→
    93→  // List state
    94→  const [buses, setBuses] = useState<Bus[]>([])
    95→  const [loading, setLoading] = useState(true)
    96→  const [searchQuery, setSearchQuery] = useState('')
    97→  const [activeFilter, setActiveFilter] = useState<string>('all')
    98→
    99→  // Tab state
   100→  const [activeTab, setActiveTab] = useState('fleet')
   101→
   102→  // Edit dialog state
   103→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   104→  const [editId, setEditId] = useState<string | null>(null)
   105→  const [editForm, setEditForm] = useState<BusForm>(defaultForm)
   106→  const [saving, setSaving] = useState(false)
   107→
   108→  // Add form state
   109→  const [addForm, setAddForm] = useState<BusForm>(defaultForm)
   110→
   111→  // Delete dialog state
   112→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   113→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null)
   114→
   115→  // Routes dialog state
   116→  const [routesDialogOpen, setRoutesDialogOpen] = useState(false)
   117→  const [routesTarget, setRoutesTarget] = useState<Bus | null>(null)
   118→
   119→  // Passenger report dialog state
   120→  const [reportDialogOpen, setReportDialogOpen] = useState(false)
   121→  const [reportBusId, setReportBusId] = useState<string | null>(null)
   122→
   123→  // Fetch buses
   124→  useEffect(() => {
   125→    let cancelled = false
   126→    async function load() {
   127→      if (!selectedSchoolId) {
   128→        if (!cancelled) {
   129→          setBuses([])
   130→          setLoading(false)
   131→        }
   132→        return
   133→      }
   134→      if (!cancelled) setLoading(true)
   135→      try {
   136→        const params = new URLSearchParams({ schoolId: selectedSchoolId })
   137→        if (activeFilter === 'active') params.set('active', 'true')
   138→        else if (activeFilter === 'inactive') params.set('active', 'false')
   139→        if (searchQuery.trim()) params.set('search', searchQuery.trim())
   140→
   141→        const res = await fetch(`/api/buses?${params.toString()}`)
   142→        if (!cancelled && res.ok) {
   143→          const data = await res.json()
   144→          setBuses(Array.isArray(data) ? data : [])
   145→        }
   146→      } catch {
   147→        if (!cancelled) toast.error('فشل في تحميل بيانات الأسطول')
   148→      } finally {
   149→        if (!cancelled) setLoading(false)
   150→      }
   151→    }
   152→    load()
   153→    return () => { cancelled = true }
   154→  }, [selectedSchoolId, searchQuery, activeFilter])
   155→
   156→  // Filter buses client-side for search (as backup if API doesn't support search)
   157→  const filteredBuses = buses.filter((bus) => {
   158→    if (!searchQuery.trim()) return true
   159→    const q = searchQuery.toLowerCase()
   160→    return (
   161→      bus.plateNumber.toLowerCase().includes(q) ||
   162→      bus.driverName.toLowerCase().includes(q)
   163→    )
   164→  })
   165→
   166→  // Open edit dialog
   167→  const openEdit = (bus: Bus) => {
   168→    setEditId(bus.id)
   169→    setEditForm({
   170→      plateNumber: bus.plateNumber,
   171→      driverName: bus.driverName,
   172→      driverPhone: bus.driverPhone || '',
   173→      driverLicense: bus.driverLicense || '',
   174→      supervisorName: bus.supervisorName || '',
   175→      supervisorPhone: bus.supervisorPhone || '',
   176→      capacity: bus.capacity,
   177→      model: bus.model || '',
   178→      color: bus.color || '#610000',
   179→      active: bus.active,
   180→      notes: bus.notes || '',
   181→    })
   182→    setEditDialogOpen(true)
   183→  }
   184→
   185→  // Handle edit save
   186→  const handleEditSave = async () => {
   187→    if (!editForm.plateNumber.trim() || !editForm.driverName.trim()) {
   188→      toast.error('يرجى إدخال رقم اللوحة واسم السائق')
   189→      return
   190→    }
   191→    setSaving(true)
   192→    try {
   193→      const body = {
   194→        schoolId: selectedSchoolId,
   195→        plateNumber: editForm.plateNumber.trim(),
   196→        driverName: editForm.driverName.trim(),
   197→        driverPhone: editForm.driverPhone.trim() || null,
   198→        driverLicense: editForm.driverLicense.trim() || null,
   199→        supervisorName: editForm.supervisorName.trim() || null,
   200→        supervisorPhone: editForm.supervisorPhone.trim() || null,
   201→        capacity: editForm.capacity,
   202→        model: editForm.model.trim() || null,
   203→        color: editForm.color || null,
   204→        active: editForm.active,
   205→        notes: editForm.notes.trim() || null,
   206→      }
   207→
   208→      const res = await fetch(`/api/buses/${editId}`, {
   209→        method: 'PUT',
   210→        headers: { 'Content-Type': 'application/json' },
   211→        body: JSON.stringify(body),
   212→      })
   213→
   214→      if (res.ok) {
   215→        toast.success('تم تحديث بيانات الباص بنجاح')
   216→        setEditDialogOpen(false)
   217→        // Refresh the list
   218→        refreshBuses()
   219→      } else {
   220→        toast.error('فشل في تحديث البيانات')
   221→      }
   222→    } catch {
   223→      toast.error('فشل في تحديث البيانات')
   224→    } finally {
   225→      setSaving(false)
   226→    }
   227→  }
   228→
   229→  // Handle add bus
   230→  const handleAddBus = async () => {
   231→    if (!addForm.plateNumber.trim() || !addForm.driverName.trim()) {
   232→      toast.error('يرجى إدخال رقم اللوحة واسم السائق')
   233→      return
   234→    }
   235→    if (addForm.capacity < 1) {
   236→      toast.error('يجب أن تكون سعة الباص أكبر من صفر')
   237→      return
   238→    }
   239→    setSaving(true)
   240→    try {
   241→      const body = {
   242→        schoolId: selectedSchoolId,
   243→        plateNumber: addForm.plateNumber.trim(),
   244→        driverName: addForm.driverName.trim(),
   245→        driverPhone: addForm.driverPhone.trim() || null,
   246→        driverLicense: addForm.driverLicense.trim() || null,
   247→        supervisorName: addForm.supervisorName.trim() || null,
   248→        supervisorPhone: addForm.supervisorPhone.trim() || null,
   249→        capacity: addForm.capacity,
   250→        model: addForm.model.trim() || null,
   251→        color: addForm.color || null,
   252→        active: addForm.active,
   253→        notes: addForm.notes.trim() || null,
   254→      }
   255→
   256→      const res = await fetch('/api/buses', {
   257→        method: 'POST',
   258→        headers: { 'Content-Type': 'application/json' },
   259→        body: JSON.stringify(body),
   260→      })
   261→
   262→      if (res.ok) {
   263→        toast.success('تم إضافة الباص بنجاح')
   264→        setAddForm(defaultForm)
   265→        setActiveTab('fleet')
   266→        refreshBuses()

   268→        const errData = await res.json().catch(() => ({}))
   269→        toast.error(errData.error || 'فشل في إضافة الباص')
   270→      }
   271→    } catch {
   272→      toast.error('فشل في إضافة الباص')
   273→    } finally {
   274→      setSaving(false)
   275→    }
   276→  }
   277→

   279→  const handleDelete = async (id: string) => {
   280→    try {
   281→      const res = await fetch(`/api/buses/${id}`, { method: 'DELETE' })
   282→      if (res.ok) {
   283→        toast.success('تم تعطيل الباص بنجاح')
   284→        refreshBuses()
   285→      } else {
   286→        toast.error('فشل في تعطيل الباص')
   287→      }
   288→    } catch {
   289→      toast.error('فشل في تعطيل الباص')
   290→    } finally {
   291→      setDeleteDialogOpen(false)
   292→      setDeleteTarget(null)
   293→    }
   294→  }
   295→
   296→  // Refresh bus list
   297→  const refreshBuses = async () => {
   298→    if (!selectedSchoolId) return
   299→    try {
   300→      const params = new URLSearchParams({ schoolId: selectedSchoolId })
   301→      if (activeFilter === 'active') params.set('active', 'true')
   302→      else if (activeFilter === 'inactive') params.set('active', 'false')
   303→
   304→      const res = await fetch(`/api/buses?${params.toString()}`)
   305→      if (res.ok) {
   306→        const data = await res.json()
   307→        setBuses(Array.isArray(data) ? data : [])
   308→      }
   309→    } catch {
   310→      toast.error('فشل في تحديث البيانات')
   311→    }
   312→  }
   313→
   314→  // View routes
   315→  const openRoutes = (bus: Bus) => {
   316→    setRoutesTarget(bus)
   317→    setRoutesDialogOpen(true)
   318→  }
   319→
   320→  // Open passenger report
   321→  const openReport = (bus: Bus) => {
   322→    setReportBusId(bus.id)
   323→    setReportDialogOpen(true)
   324→  }
   325→
   326→  return (
   327→    <div className="space-y-6" dir="rtl">
   328→      {/* Header */}
   329→      <div className="flex justify-between items-center">
   330→        <h2 className="text-lg font-bold text-[#1a1a2e] flex items-center gap-2">
   331→          <Bus className="w-5 h-5 text-[#610000]" />
   332→          إدارة أسطول الباصات
   333→        </h2>
   334→      </div>
   335→
   336→      {/* Tabs */}
   337→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   338→        <TabsList className="h-11">
   339→          <TabsTrigger value="fleet" className="min-h-[44px] px-4 text-sm">
   340→            <Bus className="w-4 h-4 ml-1" />
   341→            الأسطول
   342→          </TabsTrigger>
   343→          <TabsTrigger value="add" className="min-h-[44px] px-4 text-sm">
   344→            <Plus className="w-4 h-4 ml-1" />
   345→            إضافة باص
   346→          </TabsTrigger>
   347→        </TabsList>
   348→
   349→        {/* ===== Tab 1: Fleet List ===== */}
   350→        <TabsContent value="fleet" className="space-y-4 mt-4">
   351→          {/* Search and Filter */}
   352→          <div className="flex flex-col sm:flex-row gap-3">
   353→            <div className="relative flex-1">
   354→              <Search className="absolute right-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   355→              <Input
   356→                placeholder="البحث برقم اللوحة أو اسم السائق..."
   357→                value={searchQuery}
   358→                onChange={(e) => setSearchQuery(e.target.value)}
   359→                className="h-11 pr-10"
   360→              />
   361→            </div>
   362→            <Select value={activeFilter} onValueChange={setActiveFilter}>
   363→              <SelectTrigger className="h-11 w-full sm:w-[180px]">
   364→                <SelectValue placeholder="الحالة" />
   365→              </SelectTrigger>
   366→              <SelectContent>
   367→                <SelectItem value="all">الكل</SelectItem>
   368→                <SelectItem value="active">نشط</SelectItem>
   369→                <SelectItem value="inactive">معطل</SelectItem>
   370→              </SelectContent>
   371→            </Select>
   372→          </div>
   373→
   374→          {/* Bus Grid */}
   375→          {loading ? (
   376→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   377→              {Array.from({ length: 6 }).map((_, i) => (
   378→                <Skeleton key={i} className="h-64 rounded-xl" />
   379→              ))}
   380→            </div>
   381→          ) : filteredBuses.length === 0 ? (
   382→            <Card>
   383→              <CardContent className="py-12 text-center">
   384→                <AlertCircle className="w-12 h-12 mx-auto text-gray-300 mb-3" />
   385→                <p className="text-gray-400 text-sm">لا يوجد باصات مسجلة</p>
   386→                <Button
   387→                  className="mt-4 bg-[#610000] hover:bg-[#8B0000] text-white min-h-[44px]"
   388→                  onClick={() => setActiveTab('add')}
   389→                >
   390→                  <Plus className="w-4 h-4 ml-1" />
   391→                  إضافة باص جديد
   392→                </Button>
   393→              </CardContent>
   394→            </Card>
   395→          ) : (
   396→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   397→              {filteredBuses.map((bus) => (
   398→                <BusCard
   399→                  key={bus.id}
   400→                  bus={bus}
   401→                  onEdit={() => openEdit(bus)}
   402→                  onDelete={() => {
   403→                    setDeleteTarget(bus.id)
   404→                    setDeleteDialogOpen(true)
   405→                  }}
   406→                  onViewRoutes={() => openRoutes(bus)}
   407→                  onViewReport={() => openReport(bus)}
   408→                />
   409→              ))}
   410→            </div>
   411→          )}
   412→        </TabsContent>
   413→
   414→        {/* ===== Tab 2: Add Bus ===== */}
   415→        <TabsContent value="add" className="mt-4">
   416→          <div className="max-w-2xl mx-auto space-y-4">
   417→            {/* Bus Info Section */}
   418→            <Card>
   419→              <CardHeader className="pb-3">
   420→                <CardTitle className="text-base flex items-center gap-2">
   421→                  <Truck className="w-4 h-4 text-[#610000]" />
   422→                  بيانات الباص
   423→                </CardTitle>
   424→              </CardHeader>
   425→              <CardContent className="space-y-4">
   426→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   427→                  <div className="space-y-1.5">
   428→                    <Label>رقم اللوحة *</Label>
   429→                    <Input
   430→                      value={addForm.plateNumber}
   431→                      onChange={(e) => setAddForm({ ...addForm, plateNumber: e.target.value })}
   432→                      className="h-11"
   433→                      placeholder="أدخل رقم اللوحة"
   434→                    />
   435→                  </div>
   436→                  <div className="space-y-1.5">
   437→                    <Label>الموديل</Label>
   438→                    <Input
   439→                      value={addForm.model}
   440→                      onChange={(e) => setAddForm({ ...addForm, model: e.target.value })}
   441→                      className="h-11"
   442→                      placeholder="مثال: تويوتا كوستار"
   443→                    />
   444→                  </div>
   445→                </div>
   446→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   447→                  <div className="space-y-1.5">
   448→                    <Label>السعة *</Label>
   449→                    <Input
   450→                      type="number"
   451→                      min={1}
   452→                      value={addForm.capacity}
   453→                      onChange={(e) => setAddForm({ ...addForm, capacity: parseInt(e.target.value) || 0 })}
   454→                      className="h-11"
   455→                      placeholder="عدد الركاب"
   456→                    />
   457→                  </div>
   458→                  <div className="space-y-1.5">
   459→                    <Label>لون الباص</Label>
   460→                    <Select
   461→                      value={addForm.color}
   462→                      onValueChange={(value) => setAddForm({ ...addForm, color: value })}
   463→                    >
   464→                      <SelectTrigger className="h-11 w-full">
   465→                        <div className="flex items-center gap-2">
   466→                          <div
   467→                            className="w-4 h-4 rounded-full border border-gray-300 shrink-0"
   468→                            style={{ backgroundColor: addForm.color || '#610000' }}
   469→                          />
   470→                          <SelectValue placeholder="اختر اللون" />
   471→                        </div>
   472→                      </SelectTrigger>
   473→                      <SelectContent>
   474→                        {predefinedColors.map((c) => (
   475→                          <SelectItem key={c.value} value={c.value}>
   476→                            <div className="flex items-center gap-2">
   477→                              <div
   478→                                className="w-4 h-4 rounded-full border border-gray-300 shrink-0"
   479→                                style={{ backgroundColor: c.value }}
   480→                              />
   481→                              {c.label}
   482→                            </div>
   483→                          </SelectItem>
   484→                        ))}
   485→                      </SelectContent>
   486→                    </Select>
   487→                  </div>
   488→                </div>
   489→              </CardContent>
   490→            </Card>
   491→
   492→            {/* Driver Info Section */}
   493→            <Card>
   494→              <CardHeader className="pb-3">
   495→                <CardTitle className="text-base flex items-center gap-2">
   496→                  <User className="w-4 h-4 text-[#610000]" />
   497→                  بيانات السائق
   498→                </CardTitle>
   499→              </CardHeader>
   500→              <CardContent className="space-y-4">
   501→                <div className="space-y-1.5">
   502→                  <Label>اسم السائق *</Label>
   503→                  <Input
   504→                    value={addForm.driverName}
   505→                    onChange={(e) => setAddForm({ ...addForm, driverName: e.target.value })}
   506→                    className="h-11"
   507→                    placeholder="اسم السائق بالكامل"
   508→                  />
   509→                </div>
   510→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   511→                  <div className="space-y-1.5">
   512→                    <Label>رقم الهاتف</Label>
   513→                    <Input
   514→                      value={addForm.driverPhone}
   515→                      onChange={(e) => setAddForm({ ...addForm, driverPhone: e.target.value })}
   516→                      className="h-11"
   517→                      placeholder="05XXXXXXXX"
   518→                      dir="ltr"
   519→                    />
   520→                  </div>
   521→                  <div className="space-y-1.5">
   522→                    <Label>رقم الرخصة</Label>
   523→                    <Input

   543→              <CardContent className="space-y-4">
   544→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   545→                  <div className="space-y-1.5">
   546→                    <Label>اسم المشرف</Label>
   547→                    <Input
   548→                      value={addForm.supervisorName}
