   210→function categoryBadgeClass(category: string): string {
   211→  const c = (category || '').trim();
   212→  const map: Record<string, string> = {
   213→    'تنبيه': 'bg-rose-100 text-rose-700 dark:bg-rose-900/30 dark:text-rose-300',
   214→    'أخبار': 'bg-violet-100 text-violet-700 dark:bg-violet-900/30 dark:text-violet-300',
   215→    'إعلان': 'bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-300',
   216→    'فعالية': 'bg-fuchsia-100 text-fuchsia-700 dark:bg-fuchsia-900/30 dark:text-fuchsia-300',
   217→  };
   218→  return map[c] || 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-300';
   219→}

   221→function formatDate(iso: string): string {
   222→  try {
   223→    const d = new Date(iso);
   224→    return d.toLocaleDateString('ar-EG', {
   225→      year: 'numeric',
   226→      month: 'long',
   227→      day: 'numeric',
   228→    });
   229→  } catch {
   230→    return '';
   231→  }
   232→}

   252→
   253→  // ===== Restore session on mount =====

   255→    let restored: LoginResponse | null = null;
   256→    try {
   257→      const raw = sessionStorage.getItem(SESSION_KEY);
   258→      if (raw) restored = JSON.parse(raw) as LoginResponse;
   259→    } catch {
   260→      restored = null;
   261→    }

   263→    const t = setTimeout(() => {
   264→      if (restored) setSession(restored);

   272→    if (!session) {
   273→      const t = setTimeout(() => setDashboard(null), 0);
   274→      return () => clearTimeout(t);

   277→    const t1 = setTimeout(() => setDashboardLoading(true), 0);

   279→      try {
   280→        const res = await fetch(
   281→          `/api/teacher/dashboard?schoolId=${encodeURIComponent(
   282→            schoolId
   283→          )}&teacherId=${encodeURIComponent(session.teacherId)}`
   284→        );
   285→        const data = await res.json();
   286→        if (!res.ok) {
   287→          throw new Error(data?.error || 'فشل تحميل البيانات');
   288→        }
   289→        if (cancelled) return;
   290→        const t = setTimeout(() => setDashboard(data as DashboardData), 0);
   291→        void t;
   292→      } catch (err) {
   293→        const msg = err instanceof Error ? err.message : 'فشل تحميل البيانات';
   294→        toast.error(msg);
   295→      } finally {
   296→        if (cancelled) return;
   297→        const t = setTimeout(() => setDashboardLoading(false), 0);
   298→        void t;
   299→      }
   300→    }
   301→    run();
   302→    return () => {
   303→      cancelled = true;
   304→      clearTimeout(t1);
   305→    };
   306→  }, [session, schoolId]);
   307→
   308→  // Staggered card animation for quick actions
   309→  useEffect(() => {
   310→    if (!session) return;
   311→    const timeouts: ReturnType<typeof setTimeout>[] = [];
   312→    quickActions.forEach((_, index) => {
   313→      timeouts.push(
   314→        setTimeout(() => {
   315→          setVisibleCards((prev) =>
   316→            prev.includes(index) ? prev : [...prev, index]
   317→          );
   318→        }, 300 + index * 120)
   319→      );
   320→    });
   321→    return () => timeouts.forEach(clearTimeout);
   322→  }, [session]);
   323→
   324→  // ===== Handlers =====
   325→  const handleLogin = async (e: React.FormEvent) => {
   326→    e.preventDefault();
   327→    setLoginError(null);
   328→    if (!teacherCode.trim() || !phone.trim()) {
   329→      const t = setTimeout(() => setLoginError('يرجى إدخال كود المعلم ورقم الهاتف'), 0);
   330→      return () => clearTimeout(t);
   331→    }
   332→    setSubmitting(true);
   333→    try {
   334→      const res = await fetch('/api/teacher/login', {
   335→        method: 'POST',
   336→        headers: { 'Content-Type': 'application/json' },
   337→        body: JSON.stringify({ schoolId, teacherCode: teacherCode.trim(), phone: phone.trim() }),
   338→      });
   339→      const data = await res.json();
   340→      if (!res.ok) {
   341→        throw new Error(data?.error || 'فشل تسجيل الدخول');
   342→      }
   343→      const loginData = data as LoginResponse;
   344→      try {
   345→        sessionStorage.setItem(SESSION_KEY, JSON.stringify(loginData));
   346→      } catch {
   347→        // ignore quota errors
   348→      }
   349→      const t = setTimeout(() => {
   350→        setSession(loginData);
   351→        setVisibleCards([]);
   352→      }, 0);
   353→      void t;
   354→      if (loginData.testMode) {
   355→        toast.success('تم تسجيل الدخول في وضع التجربة');
   356→      } else {
   357→        toast.success(`مرحباً ${loginData.name}`);
   358→      }
   359→      if (loginData.warning) {
   360→        toast.info(loginData.warning);
   361→      }
   362→    } catch (err) {
   363→      const msg = err instanceof Error ? err.message : 'فشل تسجيل الدخول';
   364→      const t = setTimeout(() => setLoginError(msg), 0);
   365→      void t;
   366→    } finally {
   367→      const t = setTimeout(() => setSubmitting(false), 0);
   368→      void t;
   369→    }
   370→  };

   381→      setTeacherCode('');
