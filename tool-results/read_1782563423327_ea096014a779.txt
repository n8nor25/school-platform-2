   152→function getYearOptions(): number[] {

   157→function formatDate(dateStr: string | null): string {
   158→  if (!dateStr) return '—'
   159→  const d = new Date(dateStr)
   160→  return d.toLocaleDateString('ar-EG', { year: 'numeric', month: 'short', day: 'numeric' })
   161→}

   163→// Calculate how many months a subscription should have payments for
   164→function getExpectedMonths(startDate: string, endDate: string | null): number {
   165→  const start = new Date(startDate)
   166→  const end = endDate ? new Date(endDate) : new Date()
   167→  const months = (end.getFullYear() - start.getFullYear()) * 12 + (end.getMonth() - start.getMonth()) + 1
   168→  return Math.max(0, months)
   169→}

   640→  // ===== Bulk payments =====
   641→  async function handleBulkPayment() {
   642→    if (selectedSubscriptionIds.length === 0) {
   643→      toast.error('يرجى اختيار طالب واحد على الأقل')
   644→      return
   645→    }
   646→    setBulkPaymentLoading(true)
   647→    let successCount = 0
   648→    let errorCount = 0
   649→    for (const subId of selectedSubscriptionIds) {
   650→      try {
   651→        const sub = paymentSubscriptions.find(s => s.id === subId)
   652→        const res = await fetch('/api/transport', {
   653→          method: 'POST',
   654→          headers: { 'Content-Type': 'application/json' },
   655→          body: JSON.stringify({
   656→            action: 'payment',
   657→            schoolId: selectedSchoolId,
   658→            subscriptionId: subId,
   659→            month: paymentMonth,
   660→            year: paymentYear,
   661→            amount: sub?.monthlyFee || 0,
   662→            paymentDate: new Date().toISOString().split('T')[0],
   663→            paymentMethod: 'نقدي',
   664→          }),
   665→        })
   666→        if (res.ok) {
   667→          successCount++
   668→        } else {
   669→          errorCount++
   670→        }
   671→      } catch {
   672→        errorCount++
   673→      }

   678→    setBulkPaymentOpen(false)
   679→    // Trigger refresh
   680→    setPaymentMonth(prev => prev)
   681→  }
